\frontside
\begin{bigidea}
A clean trial shows what the machine did; permission to operate is a claim
about every condition it may meet next. Release decides, \hl{claim by claim}, what the evidence licenses. Work across the \hl{causal boundary} cannot be recalled, so an unknown premise refuses.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{A trial supports only the conditions it tested.}{Demonstrating a safety-grade failure rate from clean
  hours alone takes impractically long.}
\idea{Release criteria are physical inequalities.}{Stop, contact force, timing. A \emph{warrant}
  says why the evidence supports each claim.}
\idea{A claim is no stronger than its weakest premise.}{Look there first.
  Only truly independent, sufficient legs add confidence.}
\idea{A defeater breaks a warrant.}{A real condition the case never covered undercuts its evidence;
  exclude it or show the claim holds there.}
\idea{A verdict is a signed permit that expires.}{It holds while standing conditions are known-true and
  the manifest's digests match; the policy still only proposes.}
\idea{An update reopens what it touches.}{A bigger model can starve the enforcer on shared buses.
  Shadow mode never closes the loop; a contained canary must.}
\end{ideas}

\fcard{Stopping budget (jerk-limited stop)}%
  {\begin{gathered}\StoppingDistance = \SpeedAtDetection\PrebrakeDelay + \frac{\SpeedAtDetection^2}{2\CredibleBrakingDeceleration}
     + \frac{\SpeedAtDetection\CredibleBrakingDeceleration}{2\JerkLimit}\\ {}+ \LocalizationBound + \TrackingErrorBound + \ProtectiveClearanceMargin \le \ClearDistance\\[2pt]
   \text{spare} = \ClearDistance - \StoppingDistance, \qquad \text{reserve} = \text{spare} - \SpeedAtDetection\DispatchAdmissionInterval\end{gathered}}%
  {$\PrebrakeDelay$: observation age + renewal to brake onset; $\JerkLimit$: jerk bound; then allowances for
   localization, tracking, margin. $\DispatchAdmissionInterval$: dispatch to admission. Sign only a positive reserve.}%
  {}{}
\fcard{Confidence of an argument}%
  {\Pr(\AssuranceClaim) = \prod_{i=1}^{\RequiredPremiseCount} \RequiredPremiseConfidence_i}%
  {$\RequiredPremiseCount$ independent required premises with confidences $\RequiredPremiseConfidence_i$:
   each one added can only lower the product.}%
  {A stipulated model, not a hazard rate.}{}
\fcard{Time to brake onset}%
  {\begin{gathered}\text{silent proposer: lease} + \text{tick} + \text{bus} + \text{drive}\\
   \text{flagged premise: } \FailureFlagLatency + \text{tick} + \text{bus} + \text{drive}\end{gathered}}%
  {Lease \val{lease}, tick \val{tick}, bus \val{bus}, drive \val{onset}\,ms: \val{tsilent}\,ms,
   versus \val{tflag}\,ms plus flag latency $\FailureFlagLatency$. Blind travel = speed $\times$ time.}%
  {A monitor helps only if $\FailureFlagLatency$ stays below the lease.}{}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{How much of the clear distance does the aisle stop leave?}
The loaded base runs at \val{v}\,m/s toward a person \val{clear}\,m ahead.
Delay $\PrebrakeDelay$ = \val{age} + \val{onsetdelay} = \val{tau}\,ms; $\CredibleBrakingDeceleration$ = \val{abrake}\,m/s$^2$;
$\JerkLimit$ = \val{jerk}\,m/s$^3$; allowances \val{loc} + \val{track} + \val{margin}\,mm; $\DispatchAdmissionInterval$ = \val{dda}\,ms.\par\vspace{5pt}

  \pgfmathsetmacro\xc{\xb+\val{pallow}}
  \pgfmathsetmacro\xd{\xc+\val{pda}}
  \pgfmathsetmacro\xe{\xd+\val{preserve}}
  \draw[nbox] (0,0) rectangle (\xa,12);
  \draw[accbox] (\xa,0) rectangle (\xb,12);

  \draw[hotbox] (\xc,0) rectangle (\xd,12);
  \draw[nbox, dashed] (\xd,0) rectangle (\xe,12);
  \node[small, anchor=south] at ({\xa/2},13) {reaction};
  \node[small, anchor=south] at ({(\xa+\xb)/2},13) {braking};
  \node[small, anchor=south] at ({(\xb+\xc)/2},13) {allowances};
  \draw[dim, dashed] (\xe,-4) -- (\xe,26);
  \node[small, anchor=south east] at (\xe,26) {person, \val{clear}\,m};
  \node[small, anchor=north east] at ({\xe-4},-3) {admission, then reserve \val{reserve}\,mm};
\end{tikzpicture}\par}\vspace{6pt}
\begin{steps}
\item Reaction: \val{v}\,m/s $\times$ \val{tau}\,ms = \res{\val{react}\,mm}.
\item Braking: $\val{v}^2/(2\times\val{abrake}) + \val{v}\times\val{abrake}/(2\times\val{jerk})$ = \val{quad} + \val{jerkterm} = \res{\val{brake}\,mm}.
\item Budget: plus \val{allow}\,mm, \res{\val{total}\,mm}; spare \val{clearmm} $-$ \val{total} = \res{\val{spare}\,mm}.
\item Admission: \val{v} $\times$ \val{dda}\,ms = \val{travelda}\,mm; reserve \res{\val{reserve}\,mm}.
\end{steps}
\takeaway{The budget closes on paper, but no loaded trial backs \val{abrake}\,m/s$^2$: REFUSE.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``A thousand clean runs prove the machine is safe.''}%
  {Runs sample nominal conditions. Showing $\ConstantHazardRate$ = $10^{\val{lamexp}}$ failures per hour at 95\% confidence takes
   $\val{wall}/\ConstantHazardRate$, about \val{wallbillion} billion failure-free hours.}{}
\mistake{``If the update misbehaves, roll back and the old permit returns.''}%
  {Image and calibration roll back together, but no image repairs a worn brake; the retained image
   must pass current standing checks.}{}

\sheetsec{Check yourself}
\begin{qs}
\item \assumed{} a stopping claim rests on \val{qk} independent required premises, each at \val{qp}. Its confidence?
  Adding a seventh at \val{qpseven}: stronger or weaker? Where does the assessor look?
\item \assumed{} the proposer falls silent at \val{v}\,m/s and a monitor flags it with
  $\FailureFlagLatency$ = \val{qdflag}\,ms. Which route brakes first, and how far does each travel?
\item Loaded stops on dry concrete ($\CoulombFrictionCoefficient \approx$ \val{mudry}) back the \val{abrake}\,m/s$^2$
  premise. A COND mode admits inspected floors ($\CoulombFrictionCoefficient \ge$ \val{muinsp}) at \val{vrestricted}\,m/s,
  with no loaded stop there. (a) Verdict? (b) What lifts it, with which task measures? (c) Clear oil
  ($\CoulombFrictionCoefficient \approx$ \val{muoil}) is reported: does slowing help?
\item A retrained policy uses a larger backbone; the enforcer binary is unchanged. Which claims reopen? Why
  can't a month of near-zero shadow-mode discrepancy qualify it? A calibration file changes after signing: what does the gate do?
\end{qs}
