\begin{bigidea}
A plan safe at admission can turn unsafe while the machine follows it. So the permission path
re-checks every proposal \hl{on every control tick}: pass it, correct it minimally, or run its own
stop. Work across the \hl{causal boundary} cannot be recalled.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Permission is renewed every tick.}{Admission judges the predicted state; the enforcer,
  the state actually reached.}
\idea{Clearance is a finished budget.}{Delay grows with speed, braking with its square, plus jerk
  ramps and tracking bound; each is a watched premise.}
\idea{Gravity spends braking authority.}{Moving with the load, only the torque left over
  brakes, and stopping distance grows as one over it.}
\idea{A barrier must carry the cost to stop.}{On raw distance, torque has no authority until
  too late; subtracting the stopping distance restores it.}
\idea{Correct minimally; veto the incoherent.}{Project a sound proposal onto the boundary;
  per-axis clamps can still break the joint constraint.}
\idea{The certified floor carries the claim.}{The filter is unrated; rated drive,
  scanner and watchdog functions beneath it carry the protective claim.}
\end{ideas}

\fcard{Gravity braking tax}%
  {\NetAngularBrakingDeceleration = \frac{\JointTorqueLimit - \JointBiasTorque}{\EffectiveJointInertia},
   \qquad \text{tool travel} = \frac{\LeverArmLength\,\JointAngularSpeed^2}{2\NetAngularBrakingDeceleration}}%
  {Peak torque less gravity torque pushing with the motion, over joint inertia.}%
  {}{}
\fcard{Kinetic barrier}%
  {\BarrierFunction = \AngularClearance - \frac{\JointAngularSpeed^2}{2\NetAngularBrakingDeceleration} \ge 0}%
  {$\AngularClearance$: tool clearance over $\LeverArmLength$. Unlike raw distance, its rate contains torque,
   giving a per-tick torque limit. At $\BarrierFunction = 0$: full peak braking; closer, more than the motor has.}%
  {}{}
\fcard{Least-change projection}%
  {\CertifiedControlAction = \NominalControlCommand - \BarrierProjectionMultiplier\BarrierConstraintNormal,
   \quad \BarrierProjectionMultiplier = \max\!\left(0, \frac{\BarrierConstraintNormal^\top\NominalControlCommand - \BarrierConstraintBound}{\|\BarrierConstraintNormal\|^2}\right)}%
  {Constraint $\BarrierConstraintNormal^\top \ControlInput \le \BarrierConstraintBound$. A safe proposal
   ($\BarrierProjectionMultiplier = 0$) passes untouched; a breach lands exactly on the boundary.}%
  {Holds only inside actuator limits.}{}

\sheetsec{Worked example}
\begin{practicebox}\raggedright\boxtitle{Does the aisle speed fit the finished budget?}
Base at \val{wv}\,m/s, rack end \val{wclear}\,m away. $\PrebrakeDelay$ = \val{wtau}\,ms,
$\CredibleBrakingDeceleration$ = \val{wa}\,m/s$^2$, $\JerkLimit$ = \val{wj}\,m/s$^3$ (so $\CredibleBrakingDeceleration/2\JerkLimit$ =
\val{wramp}\,ms). Allowances: $\LocalizationBound$ = \val{wloc}, $\ProtectiveClearanceMargin$ = \val{wmar},
$\TrackingErrorBound$ = \val{weps}\,mm. $\DispatchAdmissionInterval$ = \val{wdda}\,ms.
\begin{steps}
\item Delay: \val{wv}\,m/s $\times$ \val{wtau}\,ms = \res{\val{wblind}\,mm}.
\item Stop: $\val{wv}^2/(2\times\val{wa})$ = \val{wbrake}\,mm plus \val{wv}\,m/s $\times$ \val{wramp}\,ms = \val{wrampt}\,mm: \res{\val{wstop}\,mm}.
\item Plus \val{wallow}\,mm: \res{\val{wtotal}\,mm}, spare \val{wspare}\,mm; less \val{wv}\,m/s $\times$ \val{wdda}\,ms =
  \val{wdatravel}\,mm dispatch travel: reserve \res{\val{wreserve}\,mm}.
\item $\EffectiveBrakingDelay$ = \val{wteff}\,ms, $\CredibleBrakingDeceleration\EffectiveBrakingDelay$ = \val{watau}\,m/s, $\InsetAvailableBrakingDistance$ = \val{wavail}\,mm:
  $\sqrt{\val{watau}^2 + 2\times\val{wa}\times\val{wavailm}}$ = \val{wroot}\,m/s; less \val{watau}: \res{$\SpeedBound \approx$ \val{wvmax}\,m/s}.
\end{steps}
\takeaway{It fits while every premise holds; the \val{vdrive}\,m/s limit does not.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``A distance barrier will stop the descending arm.''}%
  {Raw distance's rate holds no torque: it acts too late. Under \val{biasch}\% gravity load,
   the kinetic barrier asks full peak braking at \val{toolch}\,mm, more than the motor has at \val{bnear}\,mm.}{}
\mistake{``The speed passed the force criterion, so every handover is safe.''}%
  {Peak force = speed $\times\sqrt{\text{stiffness}\times\text{mass}}$; hand stiffness \val{khand}\,kN/m. A free
   \val{mhand}\,kg hand recoils, so the reduced mass (1/\val{mhand} + 1/\val{meff})$^{-1}$ = \val{mred}\,kg acts: \val{ffree}\,N at
   \val{vhand}\,m/s, inside \val{crit}\,N. Clamped against a rack it meets the arm's full \val{meff}\,kg: \val{fclamp}\,N.
   Only clamp-free geometry passes.}{}
\mistake{``Clearance checks positive every tick, so the base never crosses.''}%
  {The drive holds each command for a \val{ztick}\,ms tick. At \val{zv}\,m/s and \val{za}\,m/s$^2$ the base drifts
   speed $\times$ tick + acceleration $\times$ tick$^2$/2 = \val{zdrift}\,mm unchecked: shrink the checked clearance by that.
   A kinetic barrier must bound its change within the tick.}{}
