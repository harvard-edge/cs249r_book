\begin{bigidea}
A better prediction strengthens only one link of a machine's argument; the rest of the machine must supply
what the model lacks. At the \hl{causal boundary}, \hl{irreversibility} makes the \hl{weakest open premise}, not the best
prediction, the measure of what a machine may be trusted to do.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{A restriction opens an untested regime.}{The slower mode was never evaluated; test it or record it open.}
\idea{An unobserved premise is not an unknown.}{A monitored unknown triggers a stop; an unmonitored
  premise never reports, so only a standing restriction contains it.}
\idea{Every open premise gets a register entry.}{Missing observable, restriction, owner, and a closure test
  declared in advance; the restriction holds until it passes.}
\idea{Identical histories defeat any model.}{If states needing different actions read alike to the deadline, add a channel,
  act safely for both, or withhold the mode.}
\idea{A restriction has a price.}{Leg time scales as $1/\SpeedAtDetection$; weigh it against the campaign that would
  retire it.}
\end{ideas}

\fcard{Shielded hazard rate}%
  {\SystemHazardRate \le \HazardousProposalRate(1-\ShieldInterceptionCoverage) + \HardwareCommonCauseRate}%
  {Hazardous proposals per hour $\HazardousProposalRate$, coverage $\ShieldInterceptionCoverage$ measured on those

  {No credit for hazards the shield cannot sense.}{}
\fcard{Available deceleration: the friction floor}%
  {\AvailableDeceleration = \min(\CredibleBrakingDeceleration,\ \CoulombFrictionCoefficient\Gravity)}%
  {The brake binds only while $\CoulombFrictionCoefficient \ge \CredibleBrakingDeceleration/\Gravity$: \val{mumin} for
   \val{abrake}\,m/s$^2$.}%
  {Braking on time yet decelerating weakly points to the floor.}{}

\fcard{Stopping distance and speed ceiling}%
  {\begin{gathered}\StoppingDistance = \SpeedAtDetection\PrebrakeDelay + \frac{\SpeedAtDetection^2}{2\CredibleBrakingDeceleration}
     \le \AvailableBrakingDistance\\[2pt]
   \SpeedBound = -\CredibleBrakingDeceleration\PrebrakeDelay + \sqrt{(\CredibleBrakingDeceleration\PrebrakeDelay)^2
     + 2\CredibleBrakingDeceleration\AvailableBrakingDistance}\end{gathered}}%
  {Delay travel over the chain $\PrebrakeDelay$ (image to brake onset), then braking at
   $\CredibleBrakingDeceleration$; budget $\AvailableBrakingDistance$ = clearance less a reserve.}%
  {Delay is linear in speed, braking quadratic.}{}

\sheetsec{Worked example}
\begin{practicebox}\raggedright\boxtitle{Can the packaging carriage stop short of the gate?}
A carriage at \val{cv}\,m/s is \val{cclear}\,mm from the stop line as the gate closes; less a
\val{creserve}\,mm reserve, $\AvailableBrakingDistance$ = \val{cbudget}\,mm. Image to braking, $\PrebrakeDelay$ =
\val{ctau}\,ms; $\CredibleBrakingDeceleration$ = \val{ca}\,m/s$^2$.\par\vspace{5pt}
{\centering
\begin{tikzpicture}[sheet]
  \node[lab, anchor=east] at (36,26) {\val{cv}\,m/s};
  \draw[hotbox] (40,20) rectangle ({40+\val{pblind}},32);
  \draw[accbox] ({40+\val{pblind}},20) rectangle ({40+\val{pblind}+\val{pbrake}},32);
  \node[small, anchor=south west] at (40,33) {delay \val{cblind}};
  \node[small, anchor=south] at ({40+\val{pblind}+\val{pbrake}/2},33) {braking \val{cbrake}};
  \node[lab, anchor=east] at (36,6) {\val{cvm}\,m/s};
  \draw[hotbox] (40,0) rectangle ({40+\val{pblindm}},12);
  \draw[accbox] ({40+\val{pblindm}},0) rectangle ({40+\val{pblindm}+\val{pbrakem}},12);
  \draw[dim, dashed] ({40+\val{pbudget}},-4) -- ({40+\val{pbudget}},36);
  \draw[dim] ({40+\val{pclear}},-4) -- ({40+\val{pclear}},36);
  \node[small, anchor=north east] at ({40+\val{pbudget}},-5) {budget};
  \node[small, anchor=north west] at ({40+\val{pclear}},-5) {stop line};
  \node[note, anchor=west] at (122,6) {to scale, mm};
\end{tikzpicture}\par}\vspace{6pt}
\begin{steps}
\item Delay: \val{cv} $\times$ \val{ctau}\,ms = \res{\val{cblind}\,mm}, past the line.
\item Braking: (\val{cv}\,m/s)$^2/(2\times\val{ca}$\,m/s$^2$) = \val{cbrake}\,mm; stop \val{cblind} + \val{cbrake} =
  \res{\val{cstop}\,mm}: refuse.
\item Ceiling: $\CredibleBrakingDeceleration\PrebrakeDelay$ = \val{catau}\,m/s, $2\CredibleBrakingDeceleration\AvailableBrakingDistance$
  = \val{ctwoad}\,m$^2$/s$^2$; $\sqrt{\val{catau}^2+\val{ctwoad}}$ = \val{croot}, so $\SpeedBound$ = \res{\val{cceiling}\,m/s}.
\item Candidate \val{cvm}\,m/s: \val{cblindm} + \val{cbrakem} = \res{\val{cstopm}\,mm}, \val{cinside}\,mm to spare.
\end{steps}
\takeaway{Delay, not the brake, spends the clearance.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``The shield intercepts almost everything, so the proposer's error rate no longer matters.''}%
  {A fleet log of $\val{loghours}$ proposal-hours with none hazardous bounds the rate only at
   $\val{lamprop}$\,h$^{-1}$ (\val{conf}\% confidence); at coverage \val{shieldc} the residual bound is
   $\val{lamres}$\,h$^{-1}$, above a $\val{lamtarget}$\,h$^{-1}$ target. Coverage needs
   \val{covtrials} miss-free seeded trials.}{}
\mistake{``The governor trimmed the speed, so the machine failed its task.''}%
  {A trim shows an intervention, not a failure; the task succeeded if delivery met its deadline.
   Track autonomy, task success and interventions separately.}{}
