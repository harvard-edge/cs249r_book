\begin{document}

\begin{tikzpicture}[tdplot_main_coords, font=\sffamily, >=Stealth]
    
    % Floor grid
    \begin{scope}[canvas is xy plane at z=0]
        \fill[cardbg] (-4,-4) rectangle (7,4);
        \draw[step=1cm, ink!20, thin] (-4,-4) grid (7,4);
    \end{scope}

    % 1. High Speed Envelope (Outer, Red)
    \begin{scope}[canvas is xy plane at z=0]
        \fill[harvardcrimson, fill opacity=0.15] (0, 3.5) arc (90:270:3.5) -- (2.5, -3.5) arc (270:450:3.5) -- cycle;
        \draw[harvardcrimson, thick, dashed] (0, 3.5) arc (90:270:3.5) -- (2.5, -3.5) arc (270:450:3.5) -- cycle;

        \draw[harvardcrimson, thick, dotted] (0,3.5) -- (2.5,3.5);
        \draw[harvardcrimson, thick, dotted] (0,-3.5) -- (2.5,-3.5);
        \draw[harvardcrimson, thick, dotted] (0, 3.5) arc (90:-90:3.5);
    \end{scope}
    
    % 2. Low Speed Envelope (Inner, Green)
    \begin{scope}[canvas is xy plane at z=0]
        \fill[emeraldpine, fill opacity=0.3] (0, 1.5) arc (90:270:1.5) -- (0.5, -1.5) arc (270:450:1.5) -- cycle;

        \draw[emeraldpine, thick, dotted] (0,1.5) -- (0.5,1.5);
        \draw[emeraldpine, thick, dotted] (0,-1.5) -- (0.5,-1.5);
        \draw[emeraldpine, thick, dotted] (0, 1.5) arc (90:-90:1.5);
    \end{scope}

    % Blind travel indicator
    \begin{scope}[canvas is xy plane at z=0]
        \draw[|<->|, ink, thick] (0, -3.8) -- (2.5, -3.8) node[midway, below, font=\small, sloped] {Blind travel ($v \cdot \tau$)};
    \end{scope}
    
    % Robot (simplified 3D cylinder + box on top)
    \begin{scope}[shift={(0,0,0)}]
        % Base
        \tdplotdrawarc[fill=ethdarkblue!80, draw=ethdarkblue, thick]{(0,0,0)}{0.4}{0}{360}{}{}
        \draw[ethdarkblue, thick, fill=ethdarkblue!20] (0,-0.4,0) -- (0,-0.4,0.6) arc (-90:90:0.4) -- (0,0.4,0) arc (90:-90:0.4) -- cycle;

        \draw[ethpetrol, line width=4pt, line cap=round] (0,0,0.6) -- (0.5, 0.5, 1.5) -- (1, 0, 1);

    \draw[->, ethdarkblue, ultra thick] (0, 0, 0.3) -- (2, 0, 0.3) node[above, font=\small\bfseries] {$v$};

    % Human (simplified 3D)
    \begin{scope}[shift={(3.5, -2, 0)}]

        \draw[ethbronze, line width=6pt, line cap=round] (0,0,0.5) -- (0,0,1.4);

            $r = \sqrt{2 \text{KE} / F_{\text{safe}}}$
        };
    \end{scope}
