\begin{tikzpicture}[tdplot_main_coords, >={Stealth[length=2.5mm, width=1.8mm]}, font=\small]

    % Helper macro to draw a triad with custom label placements
    \newcommand{\drawtriadbase}[4]{
        \begin{scope}[shift={(#1,#2,#3)}]
            \draw[very thick, red!85!black, ->] (0,0,0) -- (#4,0,0) node[anchor=north east, font=\scriptsize\bfseries, text=red!85!black, inner sep=1pt] {X};
            \draw[very thick, green!60!black, ->] (0,0,0) -- (0,#4,0) node[anchor=west, font=\scriptsize\bfseries, text=green!60!black, inner sep=1pt] {Y};
            \draw[very thick, ethdarkblue, ->] (0,0,0) -- (0,0,#4) node[anchor=east, font=\scriptsize\bfseries, text=ethdarkblue, inner sep=1pt] {Z};
        \end{scope}
    }

    % =========================================================
    % 1. Map Frame (World Datum Origin)
    % =========================================================
    \coordinate (O_map) at (0,0,0);
    \drawtriadbase{0}{0}{0}{1.8}
    \node[anchor=north, font=\small\bfseries, text=ink, yshift=-4pt] at (0,0,0) {$\mathcal{F}_{\text{map}}$};
    \node[anchor=north, font=\scriptsize, text=softink, yshift=-16pt] at (0,0,0) {Global Map Datum};

    % =========================================================
    % 2. Odom Frame (Smooth Local Dead-Reckoning)
    % =========================================================
    \coordinate (O_odom) at (1.0, 7.0, 0);
    
    % Small covariance at odom (leader line points up-left into open space)
    \begin{scope}[shift={(O_odom)}]
        \shade[ball color=ethpetrol, opacity=0.25] (0,0,0) circle[radius=0.45cm];
        \draw[ethpetrol, dashed, thin] (0,0,0) circle[radius=0.45cm];
        \draw[ethpetrol, thin] (-0.3, -0.3, 0.3) -- (-1.0, -1.0, 1.2) 
            node[anchor=south east, font=\scriptsize\bfseries, align=center] {Small cov.\\$\boldsymbol{\Sigma}_{\text{odom}}$};
    \end{scope}
    
    \drawtriadbase{1.0}{7.0}{0}{1.4}
    \node[anchor=north, font=\small\bfseries, text=ink, yshift=-5pt] at (1.0,7.0,0) {$\mathcal{F}_{\text{odom}}$};

    % Transform Map -> Odom (Discontinuous Loop Closure Jump)
    \draw[->, ethbronze, thick, dashed] (O_map) to[bend right=22] 
        node[midway, below, font=\scriptsize, align=center, yshift=-2pt] {
            $\mathbf{T}_{\text{map}\to\text{odom}} \in \mathrm{SE}(3)$\\[1pt]
            \textcolor{ethbronze}{\textbf{Discontinuous}}\\[1pt]
            \textcolor{softink}{(relocalization jump)}
        } (O_odom);

    % =========================================================
    % 3. Base Frame & Robot Chassis
    % =========================================================
    \coordinate (O_base) at (3.0, 16.5, 0.6);

    % Draw Robot Chassis (translucent with clean edges)
    \begin{scope}[shift={(3.0, 16.5, 0)}]
        % Wheels (subtle black cylinders/boxes)
        \fill[softink!60] (-2.2, -1.9, -0.1) rectangle (-1.2, -1.7, 0.4);
        \fill[softink!60] ( 1.2, -1.9, -0.1) rectangle ( 2.2, -1.7, 0.4);
        \fill[softink!60] (-2.2,  1.7, -0.1) rectangle (-1.2,  1.9, 0.4);
        \fill[softink!60] ( 1.2,  1.7, -0.1) rectangle ( 2.2,  1.9, 0.4);

        % Chassis main body
        % Bottom face
        \draw[thick, draw=paleink!70!ink, fill=cardbg, fill opacity=0.55] 
            (-2.4,-1.8,0.2) -- (2.4,-1.8,0.2) -- (2.4,1.8,0.2) -- (-2.4,1.8,0.2) -- cycle;
        % Top face
        \draw[thick, draw=paleink!70!ink, fill=shadecolor, fill opacity=0.55] 
            (-2.4,-1.8,1.2) -- (2.4,-1.8,1.2) -- (2.4,1.8,1.2) -- (-2.4,1.8,1.2) -- cycle;
        % Vertical corner pillars
        \draw[thick, draw=paleink!70!ink] (-2.4,-1.8,0.2) -- (-2.4,-1.8,1.2);
        \draw[thick, draw=paleink!70!ink] ( 2.4,-1.8,0.2) -- ( 2.4,-1.8,1.2);
        \draw[thick, draw=paleink!70!ink] ( 2.4, 1.8,0.2) -- ( 2.4, 1.8,1.2);
        \draw[thick, draw=paleink!70!ink] (-2.4, 1.8,0.2) -- (-2.4, 1.8,1.2);

        % Sensor mounting mast for LiDAR
        \draw[very thick, paleink!80!ink] (0, 0, 1.2) -- (0, 0, 3.3);
        % Sensor front bracket for Camera
        \draw[very thick, paleink!80!ink] (2.4, 0, 0.6) -- (3.6, 0, 0.6);
    \end{scope}

    % Growing covariance at base_link (leader line points to open space to the right)
    \begin{scope}[shift={(O_base)}]
        \shade[ball color=ethbronze, opacity=0.25] (0,0,0) circle[radius=0.8cm];
        \draw[ethbronze, dashed, thin] (0,0,0) circle[radius=0.8cm];
        \draw[ethbronze, thin] (0, 0.8, 0) -- (0, 2.7, 0) 
            node[anchor=west, font=\scriptsize\bfseries, align=left] {Growing drift cov.\\$\boldsymbol{\Sigma}_{\text{drift}}$};
    \end{scope}

    \drawtriadbase{3.0}{16.5}{0.6}{0.85}
    \node[anchor=south east, font=\small\bfseries, text=ink, xshift=-4pt, yshift=6pt] at (3.0, 16.5, 0.6) {$\mathcal{F}_{\text{body}}$};

    % Transform Odom -> Base (approaches from rear/top to avoid camera collision)
    \draw[->, ethdarkblue, thick] (O_odom) to[bend left=18] 
        node[midway, above left, font=\scriptsize, align=center, yshift=2pt] {
            $\mathbf{T}_{\text{odom}\to\text{body}} \in \mathrm{SE}(3)$\\[1pt]
            \textcolor{ethdarkblue}{\textbf{Continuous $\text{C}^1$}}\\[1pt]
            \textcolor{softink}{(odometric drift accumulation)}
        } (O_base);

    % =========================================================
    % 4. Camera Frame (Front bracket: +X is front)
    % =========================================================
    \coordinate (O_cam) at (6.6, 16.5, 0.6);

    % Compounded covariance at Camera (leader line points straight down in Z into open space)
    \begin{scope}[shift={(O_cam)}]
        \shade[ball color=harvardcrimson, opacity=0.20] (0,0,0) circle[radius=0.75cm];
        \draw[harvardcrimson, dashed, thin] (0,0,0) circle[radius=0.75cm];
        \draw[harvardcrimson, thin] (0, 0, -0.75) -- (0, 0, -1.8) 
            node[anchor=north, font=\scriptsize\bfseries, align=center] {Compounded cov.\\$\boldsymbol{\Sigma}_{\text{cam}}$};
    \end{scope}

    \drawtriadbase{6.6}{16.5}{0.6}{0.95}
    \node[anchor=south west, font=\small\bfseries, text=ink, xshift=6pt, yshift=12pt] at (6.6, 16.5, 0.6) {$\mathcal{F}_{\text{cam}}$};

    % Transform Base -> Camera (Rigid Extrinsic)
    \draw[->, emeraldpine, thick] (O_base) to[bend left=25] 
        node[midway, below right=2pt, font=\scriptsize, align=center, fill=white, fill opacity=0.85, text opacity=1, inner sep=1.5pt, rounded corners=1pt] {
            $\mathbf{T}_{\text{body}\to\text{cam}}$\\ \textcolor{emeraldpine}{\tiny(rigid)}
        } (O_cam);

    % 5. LiDAR Frame (Roof Mast: +Z is up)
    % =========================================================

    \begin{scope}[shift={(O_lidar)}]

            node[anchor=south west, font=\scriptsize\bfseries, align=left] {Compounded cov.\\$\boldsymbol{\Sigma}_{\text{lidar}} \approx \boldsymbol{\Sigma}_{\text{base}} + \mathbf{J}\boldsymbol{\Sigma}_{\text{ext}}\mathbf{J}^\top$};
    \end{scope}

    \drawtriadbase{3.0}{16.5}{3.3}{0.9}
    \node[anchor=south, font=\small\bfseries, text=ink, yshift=6pt] at (3.0, 16.5, 4.4) {$\mathcal{F}_{\text{lidar}}$};

    % Transform Base -> LiDAR (Rigid Extrinsic)
    \draw[->, emeraldpine, thick] (O_base) to[bend left=25] 
        node[midway, left=4pt, font=\scriptsize, align=center] {
            $\mathbf{T}_{\text{body}\to\text{lidar}}$\\ \textcolor{emeraldpine}{\scriptsize(rigid)}
        } (O_lidar);
