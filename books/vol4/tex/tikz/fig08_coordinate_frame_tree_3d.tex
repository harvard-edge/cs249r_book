% Color Palette
\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ethpurple}{HTML}{5B4B8A}
\definecolor{ink}{HTML}{1A202C}

\begin{document}

\tdplotsetmaincoords{70}{110}
\begin{tikzpicture}[tdplot_main_coords, >={Stealth[length=3mm, width=2mm]}, font=\small]

    % Helper macro to draw a triad
    \newcommand{\drawtriad}[7]{
        \begin{scope}[shift={(#1,#2,#3)}]
            \draw[thick, red, ->] (0,0,0) -- (#4,0,0) node[anchor=#5, font=\scriptsize, inner sep=1pt] {X};
            \draw[thick, green!70!black, ->] (0,0,0) -- (0,#4,0) node[anchor=#6, font=\scriptsize, inner sep=1pt] {Y};
            \draw[thick, blue, ->] (0,0,0) -- (0,0,#4) node[anchor=#7, font=\scriptsize, inner sep=1pt] {Z};
        \end{scope}
    }

    % 1. Map Frame (World Origin)
    \coordinate (O_map) at (0,0,0);
    \drawtriad{0}{0}{0}{2}{west}{west}{south}
    \node[font=\small\bfseries, text=ink] at (0,0,-0.5) {$F_{map}$};

    % 2. Odom Frame
    \coordinate (O_odom) at (1,6,0);
    \begin{scope}[shift={(O_odom)}]
        \draw[ethpetrol, dashed] (0,0,0) circle[radius=0.4cm] node[above right, xshift=0.3cm, yshift=0.3cm, font=\scriptsize, ethpetrol] {small cov.};
        \shade[ball color=ethpetrol, opacity=0.2] (0,0) circle[radius=0.4cm];
    \end{scope}
    \drawtriad{1}{6}{0}{1.5}{west}{west}{south}
    \node[font=\small\bfseries, text=ink] at (1,6,-0.5) {$F_{odom}$};

    % Transform Map -> Odom
    \draw[->, ethdarkblue, thick, dashed] (O_map) to[bend right=20] node[midway, below, font=\scriptsize, align=center] {$\mathbf{T}_{map \to odom}$\\ (discontinuous)} (O_odom);


    % 3. Base Frame (Robot Body)
    \coordinate (O_base) at (3,16,0);
    \begin{scope}[shift={(O_base)}]
        \draw[ethbronze, dashed] (0,0,0) circle[radius=1.3cm];
        % Use a line to point to the covariance to avoid overlap
        \draw[ethbronze, thin] (-0.9,-0.9,0) -- (-2.5,-2,0) node[anchor=north east, font=\scriptsize, align=center] {growing cov.};
        \shade[ball color=ethbronze, opacity=0.2] (0,0) circle[radius=1.3cm];
        
        % Enlarged robot chassis (box)
        \begin{scope}[shift={(0,0,0)}]
            \draw[thick, fill=cardbg, opacity=0.9] (-2.5,-2,0) -- (2.5,-2,0) -- (2.5,2,0) -- (-2.5,2,0) -- cycle; % bottom
            \draw[thick, fill=cardbg, opacity=0.9] (-2.5,-2,1) -- (2.5,-2,1) -- (2.5,2,1) -- (-2.5,2,1) -- cycle; % top
            \draw[thick, fill=cardbg, opacity=0.9] (-2.5,-2,0) -- (-2.5,-2,1) -- (-2.5,2,1) -- (-2.5,2,0) -- cycle; % back
            \draw[thick, fill=cardbg, opacity=0.9] (2.5,-2,0) -- (2.5,-2,1) -- (2.5,2,1) -- (2.5,2,0) -- cycle; % front
            \draw[thick, fill=cardbg, opacity=0.9] (-2.5,-2,0) -- (2.5,-2,0) -- (2.5,-2,1) -- (-2.5,-2,1) -- cycle; % side
            \draw[thick, fill=cardbg, opacity=0.9] (-2.5,2,0) -- (2.5,2,0) -- (2.5,2,1) -- (-2.5,2,1) -- cycle; % side
        \end{scope}
    \end{scope}
    \drawtriad{3}{16}{0.5}{1.5}{west}{west}{south}
    \node[font=\small\bfseries, text=ink] at (2.5,15,0.5) {$F_{base}$};

    % Transform Odom -> Base
    \draw[->, ethdarkblue, thick] (O_odom) to[bend right=15] node[midway, below right, font=\scriptsize, align=center] {$\mathbf{T}_{odom \to base}$\\ (continuous)} (O_base);


    % 4. Camera Frame (Front)
    \coordinate (O_cam) at (6.5,16,0.5); % further front of the chassis (X is forward)
    \begin{scope}[shift={(O_cam)}]
        \draw[harvardcrimson, dashed] (0,0,0) circle[radius=1.8cm];
        \draw[harvardcrimson, thin] (0,-1.8,0) -- (1,-2.5,0) node[anchor=north west, font=\scriptsize] {large cov.};
        \shade[ball color=harvardcrimson, opacity=0.15] (0,0) circle[radius=1.8cm];
    \end{scope}
    \drawtriad{6.5}{16}{0.5}{1}{west}{west}{south}
    \node[font=\small\bfseries, text=ink] at (6.5,15,0.5) {$F_{camera}$};

    % Transform Base -> Camera
    \draw[->, emeraldpine, thick] (O_base) to[bend right=20] node[midway, below, yshift=-0.2cm, font=\scriptsize, align=center] {$\mathbf{T}_{base \to cam}$\\ (static)} (O_cam);


    % 5. LiDAR Frame (Top)
    \coordinate (O_lidar) at (3,16,3); % further top of the chassis
    \begin{scope}[shift={(O_lidar)}]
        \draw[emeraldpine, dashed] (0,0,0) circle[radius=2cm];
        \draw[emeraldpine, thin] (1.4,1.4,0) -- (2.5,2.5,0) node[anchor=south west, font=\scriptsize] {large cov.};
        \shade[ball color=emeraldpine, opacity=0.15] (0,0) circle[radius=2cm];
    \end{scope}
    \drawtriad{3}{16}{3}{1}{west}{west}{south}
    \node[font=\small\bfseries, text=ink] at (2.5,16,4) {$F_{lidar}$};

    % Transform Base -> Lidar
    \draw[->, emeraldpine, thick] (O_base) to[bend left=30] node[midway, left, xshift=-0.2cm, font=\scriptsize, align=center] {$\mathbf{T}_{base \to lidar}$\\ (static)} (O_lidar);

\end{tikzpicture}
