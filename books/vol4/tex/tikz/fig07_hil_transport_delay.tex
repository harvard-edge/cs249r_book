% Scaling factor: 1 ms = 0.15 cm. Total = 80 ms = 12 cm.

% Coordinates for the start of each block
\coordinate (x0) at (0, 0);

% Sensor (20 ms) = 3 cm
\draw[fill=ethpetrol] (x0) rectangle ++(20*\scale, \h) node[midway, white, align=center] {\footnotesize Sensor\\[-0.5ex] \footnotesize Capture\\[0.5ex] \scriptsize \textbf{20 ms}};
\coordinate (x1) at ($(x0) + (20*\scale, 0)$);

% ISP (5 ms) = 0.75 cm
\draw[fill=ethpetrol!70!white] (x1) rectangle ++(5*\scale, \h) node[midway, white, rotate=90, font=\scriptsize] {ISP \textbf{5\,ms}};
\coordinate (x2) at ($(x1) + (5*\scale, 0)$);

% Net1 (2 ms) = 0.3 cm
\draw[fill=ethbronze] (x2) rectangle ++(2*\scale, \h) coordinate[midway] (net1_m);
\node[above=0.5cm of net1_m, text=ethbronze, font=\scriptsize, align=center] {Net\\[-0.5ex] \textbf{2\,ms}};
\draw[ethbronze] (net1_m) -- ++(0, 0.7);

% Perception (30 ms) = 4.5 cm
\draw[fill=ethdarkblue] (x3) rectangle ++(30*\scale, \h) node[midway, white, align=center] {\footnotesize Perception\\[-0.5ex] \footnotesize Inference\\[0.5ex] \scriptsize \textbf{30 ms}};

% Planning (10 ms) = 1.5 cm

\coordinate (x5) at ($(x4) + (10*\scale, 0)$);

% Control (1 ms) = 0.15 cm
\draw[fill=ethdarkblue!60!white] (x5) rectangle ++(1*\scale, \h) coordinate[midway] (ctrl_m);
\node[above=0.7cm of ctrl_m, text=ethdarkblue!80!black, font=\scriptsize, align=center] {Ctrl\\[-0.5ex] \textbf{1\,ms}};
