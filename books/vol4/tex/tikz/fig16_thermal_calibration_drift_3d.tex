\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usetikzlibrary{calc, positioning, arrows.meta, shapes.geometric, intersections, decorations.pathmorphing}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{softink}{HTML}{4A5568}
\definecolor{paleink}{HTML}{A0AEC0}
\definecolor{cardbg}{HTML}{F8FAFC}
\definecolor{shadecolor}{HTML}{F4F6F9}

\begin{document}
\tdplotsetmaincoords{70}{110}
\begin{tikzpicture}[tdplot_main_coords, font=\sffamily, >=Stealth, scale=1.0, line join=round, line cap=round]

% Chassis
\def\chMinX{-1.0}
\def\chMaxX{1.0}
\def\chMinY{-2.0}
\def\chMaxY{0.5}
\def\chH{1.0}

% Draw chassis box (simplified)
\draw[thick, fill=shadecolor, draw=ethdarkblue] (\chMinX, \chMaxY, 0) -- (\chMaxX, \chMaxY, 0) -- (\chMaxX, \chMaxY, \chH) -- (\chMinX, \chMaxY, \chH) -- cycle;
\draw[thick, fill=shadecolor, draw=ethdarkblue] (\chMaxX, \chMinY, 0) -- (\chMaxX, \chMaxY, 0) -- (\chMaxX, \chMaxY, \chH) -- (\chMaxX, \chMinY, \chH) -- cycle;
\draw[thick, fill=cardbg, draw=ethdarkblue] (\chMinX, \chMinY, \chH) -- (\chMaxX, \chMinY, \chH) -- (\chMaxX, \chMaxY, \chH) -- (\chMinX, \chMaxY, \chH) -- cycle;

\node[ethdarkblue, font=\bfseries] at (0, -0.75, \chH/2 + 0.5) {Robot Chassis};

% Crossbar base
\coordinate (CBBase) at (0, 0, \chH);
% Crossbar length
\def\cbL{3.5}
\coordinate (CBEndOrig) at (0, \cbL, \chH);

% Original crossbar
\draw[thick, ethpetrol!60, dashed] (CBBase) -- (CBEndOrig);

% Warped crossbar end point
\coordinate (CBEndWarp) at (0.3, \cbL, \chH + 0.5);
% Draw warped crossbar as a curve
\draw[ultra thick, ethbronze] (CBBase) to[out=90, in=200] (CBEndWarp);

% Texts for crossbars
\node[ethpetrol, font=\small, anchor=west] at (0, 1.2, \chH-0.2) {Original crossbar};
\node[ethbronze, font=\bfseries\small, anchor=west] at (0.3, 1.5, \chH + 0.4) {Aluminum crossbar};

% Annotations for thermal expansion
\node[harvardcrimson, font=\bfseries\small, anchor=west] at (-3, 1.5, \chH + 1.2) {$\Delta T$ thermal expansion};
\draw[->, harvardcrimson, thick] (-1.0, 1.5, \chH + 1.1) to[bend left=10] (0.15, 1.5, \chH + 0.35);

% Frustums
\def\fL{5.0} % length of frustum
\def\fW{1.2} % half width at end

% Original frustum (points purely in +X)
\coordinate (FO) at (0, \cbL, \chH);
\coordinate (FE1) at ($(FO) + (\fL, -\fW, -\fW)$);
\coordinate (FE2) at ($(FO) + (\fL, \fW, -\fW)$);
\coordinate (FE3) at ($(FO) + (\fL, \fW, \fW)$);
\coordinate (FE4) at ($(FO) + (\fL, -\fW, \fW)$);
\coordinate (FC) at ($(FO) + (\fL, 0, 0)$);

\draw[thick, ethpetrol!60, dashed] (FO) -- (FE1);
\draw[thick, ethpetrol!60, dashed] (FO) -- (FE2);
\draw[thick, ethpetrol!60, dashed] (FO) -- (FE3);
\draw[thick, ethpetrol!60, dashed] (FO) -- (FE4);
\draw[thick, ethpetrol!60, dashed, fill=ethpetrol!5, opacity=0.5] (FE1) -- (FE2) -- (FE3) -- (FE4) -- cycle;
\draw[ethpetrol, dash dot] (FO) -- (FC);

% Warped frustum (points mostly in +X, but tilted in Y and Z)
\coordinate (FW) at (CBEndWarp);
% Tilted center
\coordinate (FWC) at ($(FW) + (\fL, -0.6, -1.0)$);
\coordinate (FWE1) at ($(FWC) + (0, -\fW, -\fW)$); 
\coordinate (FWE2) at ($(FWC) + (0, \fW, -\fW)$);
\coordinate (FWE3) at ($(FWC) + (0, \fW, \fW)$);
\coordinate (FWE4) at ($(FWC) + (0, -\fW, \fW)$);

\draw[thick, harvardcrimson!80] (FW) -- (FWE1);
\draw[thick, harvardcrimson!80] (FW) -- (FWE2);
\draw[thick, harvardcrimson!80] (FW) -- (FWE3);
\draw[thick, harvardcrimson!80] (FW) -- (FWE4);
\draw[thick, harvardcrimson!80, fill=harvardcrimson!10, opacity=0.8] (FWE1) -- (FWE2) -- (FWE3) -- (FWE4) -- cycle;
\draw[harvardcrimson, dash dot, thick] (FW) -- (FWC);

% Annotate angular tilt
\node[harvardcrimson, font=\small\bfseries, anchor=west] at (1.0, \cbL-1.0, \chH+1.5) {$0.095^\circ$ angular tilt};
\draw[->, harvardcrimson, thick] (1.0, \cbL-1.0, \chH+1.3) to[bend left=10] (0.2, \cbL, \chH+0.6);

% Spatial offset at 15m
\draw[<->, ultra thick, ink] (FC) -- (FWC) node[midway, right, align=left, font=\bfseries\small] {2.5 cm spatial error\\at 15 m};

\end{tikzpicture}
\end{document}
