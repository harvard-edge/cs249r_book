\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{amssymb}
\usetikzlibrary{calc, arrows.meta, positioning, shapes.geometric, decorations.pathreplacing, intersections, decorations.markings}

% Define colors
\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ethpurple}{HTML}{5B4B8A}
\definecolor{ink}{HTML}{1A202C}
\definecolor{softink}{HTML}{4A5568}
\definecolor{paleink}{HTML}{A0AEC0}
\definecolor{cardbg}{HTML}{F8FAFC}
\definecolor{shadecolor}{HTML}{F4F6F9}

\begin{document}
\tdplotsetmaincoords{70}{120} % viewing angle

\begin{tikzpicture}[tdplot_main_coords, font=\sffamily, >={Stealth[length=5pt,width=4pt]}]

% Camera frame origin
\coordinate (C) at (0,0,0);
\coordinate (Cx) at (1.5,0,0);
\coordinate (Cy) at (0,1.5,0);
\coordinate (Cz) at (0,0,1.5);

% Image plane at x=2
\def\focal{2}
\def\iw{1.2}
\def\ih{1.2}
\coordinate (I_tl) at (\focal,-\iw,\ih);
\coordinate (I_tr) at (\focal,\iw,\ih);
\coordinate (I_bl) at (\focal,-\iw,-\ih);
\coordinate (I_br) at (\focal,\iw,-\ih);

% Draw image plane
\draw[thick, ink, fill=cardbg, fill opacity=0.85] (I_tl) -- (I_tr) -- (I_br) -- (I_bl) -- cycle;
\node[ink, above=5pt, anchor=south] at (I_tl) {Image Plane ($\mathbb{R}^2$)};

% Draw camera axes (drawn after image plane so it appears in front)
\draw[->, thick, harvardcrimson] (C) -- (Cx) node[anchor=north east] {$x_{cam}$};
\draw[->, thick, emeraldpine] (C) -- (Cy) node[anchor=north west] {$y_{cam}$};
\draw[->, thick, ethdarkblue] (C) -- (Cz) node[anchor=south] {$z_{cam}$};
\node[ethdarkblue, anchor=east, xshift=-20pt, yshift=-10pt] at (C) {\textbf{Camera Frame}};

% Draw semantic mask on image plane
\def\mw{0.4}
\def\mh{0.4}
\coordinate (M_tl) at (\focal,-\mw,\mh);
\coordinate (M_tr) at (\focal,\mw,\mh);
\coordinate (M_bl) at (\focal,-\mw,-\mh);
\coordinate (M_br) at (\focal,\mw,-\mh);
\draw[thick, ethpurple, fill=ethpurple, fill opacity=0.6] (M_tl) -- (M_tr) -- (M_br) -- (M_bl) -- cycle;
\node[ethpurple, right=5pt, font=\small] at (M_tr) {VLM Mask};

% Depth Intersect at x=6
\def\depth{6}
\def\dw{1.2}
\def\dh{1.2}
\coordinate (D_tl) at (\depth,-\dw,\dh);
\coordinate (D_tr) at (\depth,\dw,\dh);
\coordinate (D_bl) at (\depth,-\dw,-\dh);
\coordinate (D_br) at (\depth,\dw,-\dh);

% Frustum rays from C through M to D
\draw[thick, dashed, softink] (C) -- (D_tl);
\draw[thick, dashed, softink] (C) -- (D_tr);
\draw[thick, dashed, softink] (C) -- (D_bl);
\draw[thick, dashed, softink] (C) -- (D_br);

% Draw surface intersection (3D points x_cam)
% Curved surface simulating depth
\draw[thick, ethpetrol, fill=ethpetrol, fill opacity=0.4] 
  (D_tl) to[bend left=10] (D_tr) to[bend right=10] (D_br) to[bend right=10] (D_bl) to[bend left=10] cycle;
  
\node[ethpetrol, below=10pt, anchor=north] at (D_bl) {Depth Intersect ($\mathbf{x}_{cam} \in \mathbb{R}^3$)};

% Robot Base Frame Origin
\coordinate (R) at (8, 6, -2);
\coordinate (Rx) at ($(R) + (1.5,0,0)$);
\coordinate (Ry) at ($(R) + (0,1.5,0)$);
\coordinate (Rz) at ($(R) + (0,0,1.5)$);

% Draw robot axes
\draw[->, thick, harvardcrimson] (R) -- (Rx) node[anchor=north east] {$x_{base}$};
\draw[->, thick, emeraldpine] (R) -- (Ry) node[anchor=north west] {$y_{base}$};
\draw[->, thick, ethdarkblue] (R) -- (Rz) node[anchor=south] {$z_{base}$};
\node[ethdarkblue, below=25pt, anchor=north] at (R) {\textbf{Robot Base Frame}};

% Transformed surface (workspace)
% Let's create a rotated/translated version of the depth patch
\coordinate (W_tl) at ($(R) + (-1, 0, 1.5)$);
\coordinate (W_tr) at ($(R) + (0, 1.5, 1.5)$);
\coordinate (W_bl) at ($(R) + (-1, 0, -0.5)$);
\coordinate (W_br) at ($(R) + (0, 1.5, -0.5)$);

\draw[thick, ethbronze, fill=ethbronze, fill opacity=0.5] 
  (W_tl) to[bend left=10] (W_tr) to[bend right=10] (W_br) to[bend right=10] (W_bl) to[bend left=10] cycle;
\node[ethbronze, above=10pt] at (W_tr) {Affordance Volume ($\mathbf{x}_{base}$)};

% SE(3) transformation arrow
\coordinate (MidD) at (\depth, 0.5, 0.5);
\coordinate (MidW) at ($(R) + (-0.5, 0.75, 0.5)$);

\draw[->, ultra thick, harvardcrimson] (MidD) to[bend right=25] node[midway, below left, xshift=-5pt, yshift=-5pt, font=\bfseries] {$\mathbf{T}_{base}^{cam} \in SE(3)$} (MidW);

\end{tikzpicture}
\end{document}
