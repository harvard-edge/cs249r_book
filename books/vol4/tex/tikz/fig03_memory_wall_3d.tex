\def\drawbox#1#2#3#4#5#6#7#8{
    % Left face (y = #3)
    \filldraw[fill=#1!60, draw=ink, thick, fill opacity=#8, line join=round] (#2,#3,#4) -- (#2+#5,#3,#4) -- (#2+#5,#3,#4+#7) -- (#2,#3,#4+#7) -- cycle;

}
