% ==================== 1. 3D Chirp Stack ====================
\begin{scope}[tdplot_main_coords]
    % 3D Axes
    \draw[axis, paleink] (0,0,0) -- (5.6,0,0) node[anchor=north east, text=ink] {Fast Time ($t$)};
    \draw[axis, paleink] (0,0,0) -- (0,5.2,0) node[anchor=north west, text=ethdarkblue, font=\bfseries] {Slow Time (Chirp $n$)};
    \draw[axis, paleink] (0,0,0) -- (0,0,5.4) node[anchor=south, text=ink] {Frequency ($f$)};

    \def\chirplen{4.8}
    \def\fmin{1.0}
    \def\fmax{4.6}
    \def\delay{0.9}

    % Chirps stacked along slow-time y
    \foreach \y in {3.75, 2.5, 1.25, 0} {
        \draw[paleink, very thin, opacity=0.4] (0,\y,0) -- (\chirplen,\y,0);
        \draw[paleink, very thin, opacity=0.4] (0,\y,\fmin) -- (\chirplen,\y,\fmin);
        
        % Transmitted chirp ramp
        \draw[chirp] (0,\y,\fmin) -- (\chirplen,\y,\fmax);
        \draw[chirp, opacity=0.35, thin] (\chirplen,\y,\fmax) -- (\chirplen,\y,\fmin);
        \ifdim \y pt > 0 pt
            \draw[chirp, opacity=0.3, thin, dotted] (\chirplen,\y,\fmin) -- (0,{\y-1.25},\fmin);
        \fi
    }

    % Chirp index labels along slow-time axis
    \node[ethdarkblue, font=\scriptsize\bfseries, anchor=north] at (0, 3.75, 0.4) {$n{=}N{-}1$};
    \node[ethdarkblue, font=\scriptsize\bfseries, anchor=north] at (0, 2.5, 0.4) {$n{=}2$};
    \node[ethdarkblue, font=\scriptsize\bfseries, anchor=north] at (0, 1.25, 0.4) {$n{=}1$};
    \node[ethdarkblue, font=\scriptsize\bfseries, anchor=north] at (0, 0, 0.4) {$n{=}0$};

    % Received chirp on chirp 0
    \draw[rxchirp] (\delay, 0, \fmin) -- (\chirplen, 0, {\fmin + (\chirplen-\delay)*(\fmax-\fmin)/\chirplen});

    % Clean labels on Chirp 0
    \node[ethpetrol, font=\footnotesize\bfseries, anchor=south, align=center] at (2.4, 0, 4.4) {TX Chirp Ramp ($S = B/T_c$)};
    \node[ethbronze, font=\scriptsize\bfseries, anchor=north] at (3.5, 0, 1.7) {RX Chirp ($\tau = 2R/c$)};

    % Beat frequency highlight with direct formula callout
    \def\xb{2.0}
    \def\ytx{{\fmin + \xb*(\fmax-\fmin)/\chirplen}}
    \def\yrx{{\fmin + (\xb-\delay)*(\fmax-\fmin)/\chirplen}}
    \draw[<->, >=stealth, harvardcrimson, very thick] (\xb, 0, \ytx) -- (\xb, 0, \yrx)
        node[midway, left=3pt, font=\footnotesize, fill=white, inner sep=2pt, rounded corners=2pt, align=right] {
            \textcolor{harvardcrimson}{\textbf{Beat Freq:}}\\[1pt]
            $f_b = S \cdot \tau = \frac{2 R B}{c \, T_c} \propto R$
        };

    % Delay tau indicator
    \draw[<->, >=stealth, ethbronze, thin] (0, 0, 0.65) -- (\delay, 0, 0.65)
        node[midway, below=-1pt, font=\scriptsize] {$\tau$};

    % Frequency axis ticks on the right of the axis
    \draw[thick, softink] (0, -0.08, \fmin) -- (0, 0.08, \fmin) node[anchor=west, font=\scriptsize, text=ink] {$f_{\min}$};
    \draw[thick, softink] (0, -0.08, \fmax) -- (0, 0.08, \fmax) node[anchor=west, font=\scriptsize, text=ink] {$f_{\max}$};
    \draw[decorate, decoration={brace, amplitude=4pt}, softink, thin]
        (0, 0.25, \fmax) -- (0, 0.25, \fmin)
        node[midway, right=5pt, font=\scriptsize\bfseries, text=ethdarkblue] {Bandwidth $B$};
\end{scope}

% Chirp duration Tc dimension marker
\draw[<->, >=stealth, softink, thin] (-0.15, 0.70) -- (-2.18, -0.79)
    node[midway, below left=-1pt, font=\scriptsize\bfseries, text=ethdarkblue] {Duration $T_c$};

% ==================== 2. 2D FFT Processing Block ====================
\node[box, text width=4.7cm] at (7.7, 1.9) (fft2d) {
    \textbf{\large 2D FFT Processing}\\[6pt]
    \textbf{\textcolor{ethpetrol}{1st FFT: Fast-Time (Range)}}\\[2pt]
    \footnotesize Resolves beat frequency along $t$:\\[1pt]
    $f_b \propto R \implies \text{Range Bins}$\\[6pt]
    \normalsize $\boldsymbol{\Downarrow}$\\[4pt]
    \textbf{\textcolor{ethdarkblue}{2nd FFT: Slow-Time (Doppler)}}\\[2pt]
    \footnotesize Resolves phase shift across chirps $n$:\\[1pt]
    $\Delta\phi = \frac{4\pi v_r T_c}{\lambda} \implies \text{Doppler Bins}$
};

% Connection from 3D chirps to 2D FFT
\draw[->, >=stealth, very thick, ethdarkblue] (3.6, 1.9) -- node[above=3pt, font=\footnotesize\bfseries, align=center] {Sampled IF\\Beat Signals} node[below=2pt, font=\scriptsize, softink] {$S[k, n]$} (fft2d.west);
