\documentclass[10pt,tikz,border=10pt]{standalone}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{tikz-3dplot}
\usetikzlibrary{arrows.meta, positioning, calc, shapes.geometric, decorations.pathreplacing, backgrounds}

% Colors from theme
\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ethpurple}{HTML}{5B4B8A}
\definecolor{ink}{HTML}{1A202C}
\definecolor{softink}{HTML}{4A5568}
\definecolor{paleink}{HTML}{A0AEC0}
\definecolor{cardbg}{HTML}{F8FAFC}
\definecolor{shadecolor}{HTML}{F4F6F9}

\begin{document}

\tdplotsetmaincoords{70}{115}

\begin{tikzpicture}[
    scale=1.2,
    thick,
    font=\sffamily,
    line/.style={draw, thick},
    axis/.style={draw, ->, >=stealth, thick},
    chirp/.style={draw=ethpetrol, very thick},
    rxchirp/.style={draw=ethbronze, thick, dashed},
    box/.style={draw=ethdarkblue, fill=cardbg, rounded corners=3pt, align=center, inner sep=8pt, text=ink},
]

\begin{scope}[tdplot_main_coords]
    % 3D Grid Axes
    % x: Fast time (0 to 6)
    % y: Slow time / Chirp index (0 to 4)
    % z: Frequency (0 to 5)

    % Draw axes
    \draw[axis, paleink] (0,0,0) -- (6.5,0,0) node[anchor=north east, text=ink] {Fast Time ($t$)};
    \draw[axis, paleink] (0,0,0) -- (0,5.5,0) node[anchor=north west, text=ink] {Slow Time (Chirp $n$)};
    \draw[axis, paleink] (0,0,0) -- (0,0,5.5) node[anchor=south, text=ink] {Frequency ($f$)};

    % Draw background grid lines (optional) for clarity
    \foreach \y in {0,1.25,2.5,3.75} {
        \draw[paleink, very thin, opacity=0.3] (0,\y,0) -- (5,\y,0);
        \draw[paleink, very thin, opacity=0.3] (0,\y,1) -- (5,\y,1);
    }

    \def\chirplen{5}
    \def\fmin{1}
    \def\fmax{5}
    \def\delay{0.8}

    % Draw chirps stacked
    \foreach \y in {3.75, 2.5, 1.25, 0} {
        % Transmitted chirp
        \draw[chirp] (0,\y,\fmin) -- (\chirplen,\y,\fmax);
        
        \ifdim \y pt > 0 pt
            % Vertical drops and reset
            \draw[chirp, opacity=0.5, thin] (\chirplen,\y,\fmax) -- (\chirplen,\y,\fmin);
            \draw[chirp, opacity=0.5, thin] (\chirplen,\y,\fmin) -- (0,{\y-1.25},\fmin);
        \else
            \draw[chirp, opacity=0.5, thin] (\chirplen,\y,\fmax) -- (\chirplen,\y,\fmin);
        \fi
    }

    % Received chirp on the first one (y=0)
    \draw[rxchirp] (\delay, 0, \fmin) -- ({\chirplen}, 0, {\fmin + (\chirplen-\delay)*(\fmax-\fmin)/\chirplen}) node[pos=0.8, right=2pt, text=ethbronze] {RX Chirp};
    \node[ethpetrol, anchor=south, align=center] at (1.5, 0, 3.8) {TX Chirp Ramp};

    % Beat frequency highlight
    \draw[<->, >=stealth, harvardcrimson, very thick] (2.5, 0, 3.0) -- (2.5, 0, 2.36) node[midway, left=2pt, fill=white, inner sep=1pt, rounded corners=1pt] {$f_{\text{beat}}$};
    
    \node[harvardcrimson, font=\footnotesize, fill=white, inner sep=1pt] at (3.5, -0.5, 1.5) {$f_{\text{beat}} \propto \text{range}$};
    \draw[harvardcrimson, thin, ->] (3.5, -0.5, 1.7) -- (2.6, -0.1, 2.5);
    
    % Coordinate nodes for connecting later
    \coordinate (fast_time_out) at (6, 0, 2.5);
    \coordinate (slow_time_out) at (0, 4.5, 3);
\end{scope}

% Normal 2D coordinates for the rest
\node[box] at (8, 2.5) (fft2d) {\textbf{2D FFT}\\[4pt]Fast-time: Range\\Slow-time: Doppler};

% Connections
\draw[->, >=stealth, ethpetrol, thick] (fast_time_out) to[out=0, in=180] node[below=4pt, midway, font=\footnotesize, align=center] {Fast-time FFT\\(Range)} (fft2d.west);
\draw[->, >=stealth, ethdarkblue, thick] (slow_time_out) to[out=45, in=90] node[above=4pt, midway, font=\footnotesize, align=center] {Slow-time FFT\\(Doppler)} (fft2d.north);

% Range Doppler Map Box
\begin{scope}[shift={(11.5, 0.5)}]
    % Axes for RD map
    \draw[->, >=stealth, thick] (0,0) -- (4,0) node[anchor=north, text=ink] {Doppler (Velocity)};
    \draw[->, >=stealth, thick] (0,0) -- (0,4) node[anchor=east, text=ink] {Range};
    
    % Draw some grid/heatmap like structure
    \fill[shadecolor] (0,0) rectangle (3.5, 3.5);
    \draw[step=0.5, paleink, thin] (0,0) grid (3.5, 3.5);
    
    % Heatmap spots (noise)
    \fill[ethbronze, opacity=0.2] (0.5, 1.0) rectangle (1.0, 1.5);
    \fill[ethpetrol, opacity=0.3] (1.5, 2.5) rectangle (2.0, 3.0);
    \fill[emeraldpine, opacity=0.1] (2.5, 0.5) rectangle (3.0, 1.0);
    
    % Target detection
    \fill[emeraldpine] (2.5, 2.5) rectangle (3.0, 3.0);
    \fill[harvardcrimson] (2.75, 2.75) circle (3pt) node[anchor=south west, text=harvardcrimson, font=\bfseries] {Target};
    
    % Label
    \node[anchor=north] at (1.75, -0.7) {\textbf{Range-Doppler Map}};
\end{scope}

\draw[->, >=stealth, thick, ethdarkblue] (fft2d.east) -- (11.5, 2.5);

\end{tikzpicture}
\end{document}
