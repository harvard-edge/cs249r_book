\documentclass{standalone}
\usepackage[T1]{fontenc}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usetikzlibrary{calc, positioning, arrows.meta, decorations.markings}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{softink}{HTML}{4A5568}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\tdplotsetmaincoords{60}{110}
\begin{tikzpicture}[tdplot_main_coords, scale=1.5, >=Stealth, font=\sffamily\small]

  % Define styles
  \tikzset{
    coordinate_axis/.style={->, thick},
    valid_pose/.style={color=emeraldpine, fill=emeraldpine},
    invalid_pose/.style={color=harvardcrimson, thick},
    base/.style={fill=ethdarkblue!80, draw=ethdarkblue},
  }

  % Draw axes
  \draw[coordinate_axis, softink] (0,0,0) -- (1.5,0,0) node[anchor=north east]{$x$};
  \draw[coordinate_axis, softink] (0,0,0) -- (0,1.5,0) node[anchor=north west]{$y$};
  \draw[coordinate_axis, softink] (0,0,0) -- (0,0,1.5) node[anchor=south]{$z$};

  % Robot Base
  \begin{scope}[shift={(0,0,0)}]
    % Simple cylinder for base
    \draw[base] (0,0,0) circle (0.4);
    \fill[base] (0,0,0.3) circle (0.4);
    \foreach \a in {0, 15, ..., 345} {
      \draw[ethdarkblue, very thin, opacity=0.5] ({\a}:0.4) -- ++(0,0,0.3);
    }
  \end{scope}

  % Dexterous workspace (Inner shell)
  \begin{scope}[tdplot_screen_coords]
    \fill[ethpetrol, opacity=0.10] (0,0,0) circle (2.2);
    \draw[ethpetrol, dashed, thick] (0,0,0) circle (2.2);
  \end{scope}
  % Equator for Dexterous
  \draw[ethpetrol, dashed, thick, opacity=0.4] (0,0,0) circle (2.2);

  % Outer reachable workspace
  \begin{scope}[tdplot_screen_coords]
    \fill[emeraldpine, opacity=0.10] (0,0,0) circle (3.5);
    \draw[emeraldpine, dashed, thick] (0,0,0) circle (3.5);
  \end{scope}
  % Equator for Reachable
  \draw[emeraldpine, dashed, thick, opacity=0.4] (0,0,0) circle (3.5);

  % Draw Valid Goal Pose inside dexterous workspace
  \def\vx{1.0}
  \def\vy{1.0}
  \def\vz{1.0}
  \coordinate (Valid) at (\vx,\vy,\vz);
  \fill[valid_pose] (Valid) circle (1.5pt);
  % Coordinate frame at valid goal
  \draw[coordinate_axis, emeraldpine] (Valid) -- ++(0.4,0,0);
  \draw[coordinate_axis, emeraldpine] (Valid) -- ++(0,0.4,0);
  \draw[coordinate_axis, emeraldpine] (Valid) -- ++(0,0,0.4);
  \node[emeraldpine, anchor=north west] at (Valid) {Valid goal};

  % Draw Invalid Goal Pose outside reachable workspace
  \def\ix{1.5}
  \def\iy{3.5}
  \def\iz{1.0}
  \coordinate (Invalid) at (\ix,\iy,\iz);
  \draw[invalid_pose] ($(Invalid) + (-0.1,-0.1,0)$) -- ($(Invalid) + (0.1,0.1,0)$);
  \draw[invalid_pose] ($(Invalid) + (-0.1,0.1,0)$) -- ($(Invalid) + (0.1,-0.1,0)$);
  % Coordinate frame at invalid goal
  \draw[coordinate_axis, harvardcrimson] (Invalid) -- ++(0.4,0,0);
  \draw[coordinate_axis, harvardcrimson] (Invalid) -- ++(0,0.4,0);
  \draw[coordinate_axis, harvardcrimson] (Invalid) -- ++(0,0,0.4);
  \node[harvardcrimson, anchor=north west] at (Invalid) {Rejected (infeasible)};

  % Trajectory exiting the workspace
  \def\sx{1.0}
  \def\sy{-1.0}
  \def\sz{1.5}
  \coordinate (Start) at (\sx,\sy,\sz);
  \fill[ethbronze] (Start) circle (1.5pt);
  
  \def\ex{3.0}
  \def\ey{-3.0}
  \def\ez{2.0}
  \coordinate (End) at (\ex,\ey,\ez);
  
  % Trajectory curve
  \draw[ethbronze, thick, ->] (Start) .. controls (2.0, -1.5, 1.8) and (2.5, -2.5, 1.9) .. (End);
  
  % Intersection point (approximate calculation for radius=3.5)
  \coordinate (Cross) at (2.25, -2.35, 1.85);
  \draw[harvardcrimson, thick] ($(Cross) + (-0.1,-0.1,0)$) -- ($(Cross) + (0.1,0.1,0)$);
  \draw[harvardcrimson, thick] ($(Cross) + (-0.1,0.1,0)$) -- ($(Cross) + (0.1,-0.1,0)$);
  \node[harvardcrimson, anchor=north, font=\sffamily\scriptsize, align=center, yshift=-4pt] at (Cross) {Boundary\\violation};

  % Labels in screen coordinates for stable positioning
  \begin{scope}[tdplot_screen_coords]
    \node[ethpetrol, font=\sffamily\small, align=center] at (-1.2, -1.5) {Dexterous\\workspace};
    \node[emeraldpine, font=\sffamily\small, align=center] at (-2.2, 3.6) {Reachable\\workspace};
  \end{scope}

\end{tikzpicture}
\end{document}
