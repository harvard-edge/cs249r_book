% 6. Gripper Macro
% -------------------------------------------------------------
\newcommand{\drawgripper}[4]{
  % #1=X, #2=Y, #3=Z, #4=color
  \begin{scope}[shift={(#1,#2,#3)}]
    % Crossbar (base)
    \filldraw[#4!35, draw=#4, thick] (-0.35,-0.1,0.5) -- (0.35,-0.1,0.5) -- (0.35,0.1,0.5) -- (-0.35,0.1,0.5) -- cycle;
    % Stem
    \filldraw[#4!35, draw=#4, thick] (-0.08,-0.08,0.5) -- (0.08,-0.08,0.5) -- (0.08,0.08,1.0) -- (-0.08,0.08,1.0) -- cycle;
    % Left finger
    \filldraw[#4!35, draw=#4, thick] (-0.35,-0.07,0.5) -- (-0.23,-0.07,0.5) -- (-0.23,0.07,0.02) -- (-0.35,0.07,0.02) -- cycle;
    % Right finger
    \filldraw[#4!35, draw=#4, thick] (0.35,-0.07,0.5) -- (0.23,-0.07,0.5) -- (0.23,0.07,0.02) -- (0.35,0.07,0.02) -- cycle;
  \end{scope}
}
