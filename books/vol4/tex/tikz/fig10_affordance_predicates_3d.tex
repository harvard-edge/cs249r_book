\documentclass[tikz,border=15pt]{standalone}
\usepackage{tikz-3dplot}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usetikzlibrary{calc,positioning,shapes.geometric,arrows.meta,decorations.markings,fadings,3d}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\tdplotsetmaincoords{65}{110}
\begin{tikzpicture}[tdplot_main_coords, >={Latex[length=3mm,width=2mm]}, line join=round, line cap=round]

% Tabletop
\begin{scope}[canvas is xy plane at z=0]
\filldraw[fill=cardbg, draw=ink!20, thick] (-4,-4) rectangle (4,4);
% Grid lines optional
\foreach \x in {-3,-2,...,3} {
  \draw[ink!10, very thin] (\x,-4) -- (\x,4);
  \draw[ink!10, very thin] (-4,\x) -- (4,\x);
}
\end{scope}

% Object (Cylinder)
\def\objR{0.6}
\def\objH{1.8}

% Cylinder bottom
\begin{scope}[canvas is xy plane at z=0]
\filldraw[ethdarkblue, fill=ethdarkblue!50] (0,0) circle[radius=\objR];
\end{scope}

% Cylinder body (drawn using rectangle in rotated plane to align with viewing angle)
\tdplotsetrotatedcoords{-20}{0}{0}
\begin{scope}[tdplot_rotated_coords, canvas is zx plane at y=0]
\filldraw[ethdarkblue, thick, fill=ethdarkblue!40] (0,-\objR) rectangle (\objH,\objR);
\end{scope}
  
% Cylinder top
\begin{scope}[canvas is xy plane at z=\objH]
\filldraw[ethdarkblue, thick, fill=ethdarkblue!60] (0,0) circle[radius=\objR];
\end{scope}

% Grasp point on top surface (center)
\coordinate (GraspPoint) at (0,0,\objH);

% Clearance Sphere
\def\clearR{3.5}
% Clearance sphere background fill
\node[circle, minimum size=2*\clearR cm, inner sep=0pt, fill=ethpetrol!5, opacity=0.8, draw=ethpetrol, thick, dashed] (sphere) at (0, 0, \objH) {};

\node[ethpetrol, font=\small\bfseries, anchor=north] at (2.5, -1, \objH - 0.5) {Clearance sphere $r_{clear}$};

% Approach Cone
% Cone apex at (0,0, \objH), base at (0,0, \objH+3.5)
\def\coneH{4.0}
\def\coneR{2.0}
\coordinate (ConeBaseC) at (0,0,\objH+\coneH);

\begin{scope}[canvas is xy plane at z=\objH+\coneH]
% Cone base back curve
\draw[emeraldpine, thick, fill=emeraldpine!10, opacity=0.6] (0,0) circle[radius=\coneR];
\end{scope}

% Cone sides
\draw[emeraldpine, thick, fill=emeraldpine!20, opacity=0.4] 
  (GraspPoint) -- (\coneR, 0, \objH+\coneH) 
  arc(0:-180:\coneR) -- cycle;
\draw[emeraldpine, thick] (GraspPoint) -- (\coneR, 0, \objH+\coneH);
\draw[emeraldpine, thick] (GraspPoint) -- (-\coneR, 0, \objH+\coneH);

\begin{scope}[canvas is xy plane at z=\objH+\coneH]
% Cone base front curve (to give 3D depth)
\draw[emeraldpine, thick] (\coneR,0) arc(0:-180:\coneR);
\end{scope}

% Contact normal
\draw[->, harvardcrimson, line width=2pt] (GraspPoint) -- (0,0,\objH+1.5) node[anchor=north west, xshift=2mm, yshift=2mm, font=\small\bfseries] {Contact normal $\hat{n}$};

% Gripper function
\def\drawgripper#1#2#3#4{
  % #1=X, #2=Y, #3=Z, #4=color
  \begin{scope}[shift={(#1,#2,#3)}]
    % Base
    \draw[#4, thick, fill=#4!30] (-0.5,-0.2,0.8) -- (0.5,-0.2,0.8) -- (0.5,0.2,0.8) -- (-0.5,0.2,0.8) -- cycle;
    % Shaft
    \draw[#4, thick, fill=#4!30] (-0.1,-0.1,0.8) -- (0.1,-0.1,0.8) -- (0.1,0.1,1.5) -- (-0.1,0.1,1.5) -- cycle;
    % Fingers
    \draw[#4, thick, fill=#4!30] (-0.5,-0.1,0.8) -- (-0.3,-0.1,0.8) -- (-0.3,0.1,0) -- (-0.5,0.1,0) -- cycle;
    \draw[#4, thick, fill=#4!30] (0.5,-0.1,0.8) -- (0.3,-0.1,0.8) -- (0.3,0.1,0) -- (0.5,0.1,0) -- cycle;
  \end{scope}
}

% Valid TCP pose (inside cone, aligned with Z)
\drawgripper{0}{0}{\objH+1.0}{emeraldpine}
\node[emeraldpine, font=\small\bfseries, align=center, anchor=east] at (-0.3, 0, \objH+2.5) {Valid TCP\\pose};

% Invalid TCP pose (outside cone, crossed out)
% Placed at angle
\begin{scope}[shift={(-2.5, 0, \objH+1.5)}, rotate around y=45]
  \drawgripper{0}{0}{0}{harvardcrimson}
  % Cross out
  \draw[harvardcrimson, line width=2pt] (-0.8, -0.5, 0.5) -- (0.8, 0.5, 1.2);
  \draw[harvardcrimson, line width=2pt] (-0.8, 0.5, 1.2) -- (0.8, -0.5, 0.5);
\end{scope}
\node[harvardcrimson, font=\small\bfseries, align=center, anchor=north] at (-3.0, 0, \objH+3.0) {Invalid TCP\\pose};

% Labels
\node[emeraldpine, font=\small\bfseries, anchor=south west] at (0.3, 0.3, \objH+\coneH+0.2) {Approach cone ($\alpha_{max}$)};

\end{tikzpicture}
\end{document}
