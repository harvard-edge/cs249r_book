\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}

\begin{document}
\tdplotsetmaincoords{65}{115}
\begin{tikzpicture}[tdplot_main_coords, scale=1.5, >=stealth]

% Axes
\draw[->, thick, ink] (0,0,0) -- (6,0,0) node[anchor=north east]{$x$};
\draw[->, thick, ink] (0,0,0) -- (0,5,0) node[anchor=north west]{$y$};
\draw[->, thick, ink] (0,0,0) -- (0,0,8) node[anchor=south]{$t$ (time)};

% Obstacle
\def\ox{3.5}
\def\oy{3.0}
\def\obstdx{1.5}
\def\obstdy{1.5}
\def\ot{7.5}

% Draw back faces of obstacle
\draw[harvardcrimson, fill=harvardcrimson!10, opacity=0.5] (\ox,\oy,0) -- (\ox+\obstdx,\oy,0) -- (\ox+\obstdx,\oy,\ot) -- (\ox,\oy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!10, opacity=0.5] (\ox,\oy,0) -- (\ox,\oy+\obstdy,0) -- (\ox,\oy+\obstdy,\ot) -- (\ox,\oy,\ot) -- cycle;

% Tube
\foreach \t in {0,0.2,...,7.5} {
    \pgfmathsetmacro{\x}{1 + 0.4*\t}
    \pgfmathsetmacro{\y}{1 + 0.2*\t + 0.03*\t*\t}
    \pgfmathsetmacro{\r}{0.1 + 0.1*\t}
    \tdplotdrawarc[tdplot_main_coords, shift={(\x,\y,\t)}, color=emeraldpine, fill=emeraldpine!10, opacity=0.3, draw opacity=0.3]{(0,0,0)}{\r}{0}{360}{}{}
}

% Cross section at t=5
\pgfmathsetmacro{\tc}{5}
\pgfmathsetmacro{\xc}{1 + 0.4*\tc}
\pgfmathsetmacro{\yc}{1 + 0.2*\tc + 0.03*\tc*\tc}
\pgfmathsetmacro{\rc}{0.1 + 0.1*\tc}
\tdplotdrawarc[tdplot_main_coords, shift={(\xc,\yc,\tc)}, color=ethpetrol, fill=ethpetrol!30, opacity=0.7, thick]{(0,0,0)}{\rc}{0}{360}{}{}
\draw[ethpetrol, thick, ->] (\xc,\yc,\tc) -- (\xc+\rc,\yc,\tc) node[anchor=south west, inner sep=1pt] {$\sigma_{\mbox{\tiny sim}} \!+\! \sigma_{\mbox{\tiny exec}}$};

% Nominal trajectory
\pgfmathsetmacro{\prevx}{1}
\pgfmathsetmacro{\prevy}{1}
\pgfmathsetmacro{\prevt}{0}
\foreach \t in {0.1,0.2,...,7.5} {
    \pgfmathsetmacro{\x}{1 + 0.4*\t}
    \pgfmathsetmacro{\y}{1 + 0.2*\t + 0.03*\t*\t}
    \draw[emeraldpine, thick] (\prevx, \prevy, \prevt) -- (\x, \y, \t);
    \xdef\prevx{\x}
    \xdef\prevy{\y}
    \xdef\prevt{\t}
}

% Obstacle front faces
\draw[harvardcrimson, fill=harvardcrimson!20, opacity=0.3, thick] (\ox+\obstdx,\oy,0) -- (\ox+\obstdx,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox+\obstdx,\oy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!20, opacity=0.3, thick] (\ox,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox,\oy+\obstdy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!20, opacity=0.3, thick] (\ox,\oy,\ot) -- (\ox+\obstdx,\oy,\ot) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox,\oy+\obstdy,\ot) -- cycle;

% Labels
\node[emeraldpine, font=\small, anchor=east] at (1.5, 0.5, 2) {Nominal trajectory};
\draw[->, emeraldpine, thick] (1.5, 0.5, 2) -- (1.5, 1.4, 2);

\node[ethpetrol, font=\small, anchor=east] at (\xc-0.5, \yc-0.5, \tc+1.5) {Uncertainty tube (sim2real gap)};
\draw[->, ethpetrol, thick] (\xc-0.5, \yc-0.5, \tc+1.5) -- (\xc, \yc, \tc+0.2);

\node[harvardcrimson, font=\small] at (\ox+2.5, \oy-0.5, 2) {Obstacle};
\draw[->, harvardcrimson, thick] (\ox+2.5, \oy-0.5, 2) -- (\ox+1.5, \oy, 2);

\node[harvardcrimson, font=\small, anchor=west] at (\xc+1.2, \yc, \tc) {Critical intersection};
\draw[->, harvardcrimson, thick] (\xc+1.2, \yc, \tc) -- (\ox, \oy, \tc);

\end{tikzpicture}
