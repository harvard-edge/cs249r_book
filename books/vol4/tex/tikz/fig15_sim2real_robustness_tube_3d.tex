\documentclass[tikz,border=12pt]{standalone}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[T1]{fontenc}
\usepackage{textcomp}
\usepackage[EULERGREEK]{sansmath}
\sansmath
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, calc}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\tdplotsetmaincoords{65}{115}
\begin{tikzpicture}[tdplot_main_coords, scale=1.5, >=Stealth]

% ── Coordinate Axes ──────────────────────────────────────────────────────
\draw[->, thick, ink] (0,0,0) -- (6,0,0) node[anchor=north east, font=\bfseries\small]{$x$};
\draw[->, thick, ink] (0,0,0) -- (0,5.2,0) node[anchor=north west, font=\bfseries\small]{$y$};
\draw[->, thick, ink] (0,0,0) -- (0,0,8.2) node[anchor=south, font=\bfseries\small]{$t$ (time)};

% ── Obstacle parameters ──────────────────────────────────────────────────
\def\ox{3.5}
\def\oy{3.0}
\def\obstdx{1.5}
\def\obstdy{1.5}
\def\ot{7.5}

% Obstacle back faces
\draw[harvardcrimson, fill=harvardcrimson!6, opacity=0.4] (\ox,\oy,0) -- (\ox+\obstdx,\oy,0) -- (\ox+\obstdx,\oy,\ot) -- (\ox,\oy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!6, opacity=0.4] (\ox,\oy,0) -- (\ox,\oy+\obstdy,0) -- (\ox,\oy+\obstdy,\ot) -- (\ox,\oy,\ot) -- cycle;

% ── Robustness Tube (widens with time: distribution widening) ────────────
\foreach \t in {0,0.2,...,7.5} {
    \pgfmathsetmacro{\x}{1 + 0.4*\t}
    \pgfmathsetmacro{\y}{1 + 0.2*\t + 0.03*\t*\t}
    \pgfmathsetmacro{\r}{0.12 + 0.09*\t}
    \tdplotdrawarc[tdplot_main_coords, shift={(\x,\y,\t)}, color=ethpetrol, fill=ethpetrol!6, opacity=0.20, draw opacity=0.25]{(0,0,0)}{\r}{0}{360}{}{}
}

% ── Highlighted Cross Section at t=5 ────────────────────────────────────
\pgfmathsetmacro{\tc}{5}
\pgfmathsetmacro{\xc}{1 + 0.4*\tc}
\pgfmathsetmacro{\yc}{1 + 0.2*\tc + 0.03*\tc*\tc}
\pgfmathsetmacro{\rc}{0.12 + 0.09*\tc}
\tdplotdrawarc[tdplot_main_coords, shift={(\xc,\yc,\tc)}, color=ethpetrol, fill=ethpetrol!30, opacity=0.75, thick]{(0,0,0)}{\rc}{0}{360}{}{}

% Tube radius annotation (\epsilon_{\text{gap}}) on disk: pointing towards front-left
\draw[ethpetrol, very thick, ->] (\xc,\yc,\tc) -- (\xc+\rc*0.85, \yc-\rc*0.5, \tc)
    node[midway, below left=2pt, font=\bfseries\small, text=ethpetrol] {$\epsilon_{\text{gap}}$};

% Clearance margin \rho_{\text{sim}} from nominal center to obstacle corner
\draw[ethdarkblue, thick, dashed] (\xc,\yc,\tc) -- (\ox,\oy,\tc)
    node[midway, above=2pt, font=\bfseries\scriptsize, text=ethdarkblue] {$\rho_{\text{sim}}$};

% ── Obstacle front faces ────────────────────────────────────────────────
\draw[harvardcrimson, fill=harvardcrimson!14, opacity=0.32, thick] (\ox+\obstdx,\oy,0) -- (\ox+\obstdx,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox+\obstdx,\oy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!14, opacity=0.32, thick] (\ox,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,0) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox,\oy+\obstdy,\ot) -- cycle;
\draw[harvardcrimson, fill=harvardcrimson!20, opacity=0.45, thick] (\ox,\oy,\ot) -- (\ox+\obstdx,\oy,\ot) -- (\ox+\obstdx,\oy+\obstdy,\ot) -- (\ox,\oy+\obstdy,\ot) -- cycle;

% ── Trajectories ────────────────────────────────────────────────────────
% 1. Simulated trajectory (Nominal s_sim(t) - Blue solid curve)
\pgfmathsetmacro{\prevx}{1}
\pgfmathsetmacro{\prevy}{1}
\pgfmathsetmacro{\prevt}{0}
\foreach \t in {0.1,0.2,...,7.5} {
    \pgfmathsetmacro{\x}{1 + 0.4*\t}
    \pgfmathsetmacro{\y}{1 + 0.2*\t + 0.03*\t*\t}
    \draw[ethdarkblue, very thick] (\prevx, \prevy, \prevt) -- (\x, \y, \t);
    \xdef\prevx{\x}
    \xdef\prevy{\y}
    \xdef\prevt{\t}
}

% 2. Real hardware trial (Hardware execution s_real(t) - Crimson dashed curve)
\pgfmathsetmacro{\prevxr}{1}
\pgfmathsetmacro{\prevyr}{1}
\pgfmathsetmacro{\prevtr}{0}
\foreach \t in {0.1,0.2,...,7.5} {
    \pgfmathsetmacro{\xr}{1 + 0.4*\t + 0.035*\t}
    \pgfmathsetmacro{\yr}{1 + 0.2*\t + 0.03*\t*\t + 0.05*\t}
    \draw[harvardcrimson, thick, dashed] (\prevxr, \prevyr, \prevtr) -- (\xr, \yr, \t);
    \xdef\prevxr{\xr}
    \xdef\prevyr{\yr}
    \xdef\prevtr{\t}
}

% Critical intersection dot at obstacle corner at t=5
\fill[harvardcrimson] (\ox, \oy, \tc) circle (3pt);

% ── Legend Card in upper area (completely clear of all elements) ────────
\begin{scope}[shift={(0.1, 0.1, 5.8)}]
    \node[fill=cardbg, draw=ink!25, rounded corners=3pt, inner sep=6pt, anchor=south west] at (0,0) {
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}cc@{\quad}l@{}}
        \tikz[baseline=-0.5ex]\draw[ethdarkblue, line width=1.8pt] (0,0) -- (0.5,0); && \small\textbf{Simulation} $s_{\text{sim}}(t)$ \\
        \tikz[baseline=-0.5ex]\draw[harvardcrimson, line width=1.4pt, dashed] (0,0) -- (0.5,0); && \small\textbf{Hardware trial} $s_{\text{real}}(t)$ \\
        \tikz[baseline=-0.5ex]\filldraw[fill=ethpetrol!35, draw=ethpetrol, thick] (0.25,0) circle (3pt); && \small\textbf{Robustness tube} ($\epsilon_{\text{gap}}$) \\
        \tikz[baseline=-0.5ex]\filldraw[fill=harvardcrimson!25, draw=harvardcrimson, thick] (0.25,0) circle (3pt); && \small\textbf{Obstacle} ($\rho_{\text{sim}} < \epsilon_{\text{gap}}$)
        \end{tabular}
    };
\end{scope}

% ── Direct Callouts on the Right Side (zero trajectory overlaps) ────────
% Critical intersection callout
\node[harvardcrimson, font=\bfseries\small, anchor=west] (crit_lbl) at (2.5, 4.6, 5.2) {Critical intersection};
\draw[->, harvardcrimson, thick] (crit_lbl.west) to[bend right=15] (\ox+0.05, \oy, \tc);

% Obstacle label (to the right of the prism)
\node[harvardcrimson, font=\bfseries\small, anchor=west] (obs_lbl) at (\ox+\obstdx+0.4, \oy+\obstdy+0.2, 1.8) {Obstacle};
\draw[->, harvardcrimson, thick] (obs_lbl.west) -- (\ox+\obstdx, \oy+\obstdy*0.8, 1.8);

% Tube widening callout at top
\node[ethpetrol, font=\bfseries\small, anchor=west] (tube_lbl) at (4.5, 5.0, 7.5) {Tube widening};
\draw[->, ethpetrol, thick] (tube_lbl.west) to[bend right=10] (4.1, 4.3, 7.5);

\end{tikzpicture}
\end{document}
