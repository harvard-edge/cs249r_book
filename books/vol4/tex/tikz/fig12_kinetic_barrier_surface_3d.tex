\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        width=15cm, height=12cm,
        view={45}{35},
        xlabel={Distance $D$ (m)},
        ylabel={Angular Rate $\omega$ (rad/s)},
        zlabel={Barrier Value $h(D, \omega)$},
        xmin=0, xmax=5,
        ymin=-2, ymax=2,
        zmin=-3, zmax=4,
        grid=major,
        colormap={cbfmap}{
            color=(harvardcrimson!80)
            color=(harvardcrimson!60)
            color=(harvardcrimson!40)
            color=(white)
            color=(emeraldpine!20)
            color=(emeraldpine!40)
            color=(emeraldpine!60)
            color=(emeraldpine!80)
        },
        point meta min=-3, point meta max=4,
        xlabel style={font=\large\bfseries, text=ink},
        ylabel style={font=\large\bfseries, text=ink},
        zlabel style={font=\large\bfseries, text=ink},
        ticklabel style={font=\normalsize, text=ink},
        axis lines=left,
        axis line style={thick, ink},
        z buffer=sort,
        clip=false
    ]
    
    % The z=0 critical boundary plane
    \addplot3[
        surf,
        domain=0:5,
        domain y=-2:2,
        samples=2,
        opacity=0.20,
        fill=gray!25,
        draw=none,
    ] {0};
    
    % The CBF surface: h = D - 1 - 0.75*|omega|
    \addplot3[
        surf,
        domain=0:5,
        domain y=-2:2,
        samples=32,
        samples y=32,
        opacity=0.82,
        faceted color=ink!15!white,
    ] {x - 1 - 0.75*abs(y)};
    
    % Boundary curve h=0 -> D = 1 + 0.75*|omega|
    \addplot3[
        domain=-2:2,
        samples=40,
        ultra thick,
        color=ethdarkblue,
    ] ( {1 + 0.75*abs(x)}, {x}, {0} );

    % Trajectory 1: Approach from initial state to CBF intervention point
    % From D=4.2, omega=1.4 to D=2.3, omega=1.2
    \addplot3[
        domain=0:1,
        samples=25,
        ultra thick,
        ethpetrol,
    ] ( 
        {4.2 - 1.9*x}, 
        {1.4 - 0.2*x}, 
        {(4.2 - 1.9*x) - 1 - 0.75*abs(1.4 - 0.2*x)}
    );

    % Trajectory 1b: CBF active braking deflection (stays safe, h > 0)
    % Smooth deceleration: D goes 2.3 -> 1.7, omega goes 1.2 -> 0.05
    \addplot3[
        domain=0:1,
        samples=25,
        ultra thick,
        ethpetrol,
        ->
    ] ( 
        {2.3 - 0.6*x}, 
        {1.2 - 1.15*x}, 
        {(2.3 - 0.6*x) - 1 - 0.75*abs(1.2 - 1.15*x)}
    );
    
    % Trajectory 2: Uncorrected nominal trajectory (crosses h=0, enters unsafe set h < 0)
    % D goes 2.3 -> 0.4, omega goes 1.2 -> 1.5
    \addplot3[
        domain=0:1,
        samples=25,
        ultra thick,
        dashed,
        harvardcrimson,
        ->
    ] (
        {2.3 - 1.9*x}, 
        {1.2 + 0.3*x}, 
        {(2.3 - 1.9*x) - 1 - 0.75*abs(1.2 + 0.3*x)}
    );

    % Start point marker x0
    \addplot3[only marks, mark=*, mark size=2.5pt, mark options={fill=ethpetrol, draw=ink}]
        coordinates {(4.2, 1.4, 2.15)};

    % Intervention point marker
    \addplot3[only marks, mark=*, mark size=3.5pt, mark options={fill=ethbronze, draw=ink, thick}]
        coordinates {(2.3, 1.2, 0.4)};

    % End of safe trajectory marker
    \addplot3[only marks, mark=*, mark size=2.5pt, mark options={fill=ethpetrol, draw=ink}]
        coordinates {(1.7, 0.05, 0.6625)};

    % ─────────────────────────────────────────────────────────────────────────
    % LEFT HALF: SET GEOMETRY LABELS
    % ─────────────────────────────────────────────────────────────────────────

    % 1. Unsafe Region: on pink surface (clear of z-axis and safe set)
    \node[harvardcrimson, font=\large\bfseries, fill=white, fill opacity=0.92, text opacity=1, rounded corners=3pt, inner sep=3.5pt, draw=harvardcrimson!40] 
        at (axis cs:0.4, -1.4, -1.8) {Unsafe Region ($h < 0$)};

    % 2. Viable Boundary: leader line pointing to the blue boundary curve from clear upper sky
    \draw[<-, thick, ethdarkblue] (axis cs:1.2, -0.27, 0) -- ++(125:1.8cm) 
        node[above left, font=\normalsize\bfseries, text=ethdarkblue, fill=white, fill opacity=0.92, text opacity=1, rounded corners=2pt, inner sep=2.5pt, draw=ethdarkblue!40] 
        {Viable Boundary ($h = 0$)};

    % 3. Safe Set: on foreground green surface with generous clearance
    \node[emeraldpine, font=\large\bfseries, fill=white, fill opacity=0.92, text opacity=1, rounded corners=3pt, inner sep=3.5pt, draw=emeraldpine!40] 
        at (axis cs:3.5, -1.2, 1.3) {Safe Set $\mathcal{C}$ ($h \ge 0$)};

    % ─────────────────────────────────────────────────────────────────────────
    % RIGHT HALF: DYNAMICS AND CBF FILTER LABELS
    % ─────────────────────────────────────────────────────────────────────────

    % 4. Initial state label: to the right of x0
    \node[right, font=\small\bfseries, text=ethpetrol, fill=white, fill opacity=0.92, text opacity=1, rounded corners=2pt, inner sep=2pt, draw=ethpetrol!40]
        at (axis cs:4.25, 1.4, 2.3) {$\mathbf{x}_0$ (Nominal approach)};

    % 5. CBF filter intervenes: leader line straight up into clear upper space
    \draw[<-, thick, ethbronze] (axis cs:2.3, 1.2, 0.4) -- ++(75:2.4cm)
        node[above right, font=\small\bfseries, text=ink, fill=white, fill opacity=0.92, text opacity=1, rounded corners=2pt, inner sep=2.5pt, draw=ethbronze!60]
        {CBF filter intervenes ($h = \delta$)};

    % 6. CBF-deflected trajectory: leader line straight up into clear center-upper sky
    \draw[<-, thick, ethpetrol] (axis cs:1.7, 0.05, 0.6625) -- ++(95:2.0cm)
        node[above, font=\small\bfseries, text=ethpetrol, fill=white, fill opacity=0.92, text opacity=1, rounded corners=2pt, inner sep=2.5pt, draw=ethpetrol!50]
        {CBF-deflected trajectory ($h > 0$)};

    % 7. Uncorrected trajectory: leader line pointing to the dashed red line, going down-right into empty space
    \draw[<-, thick, harvardcrimson] (axis cs:1.2, 1.35, -0.8) -- ++(-25:2.2cm)
        node[right, font=\small\bfseries, text=harvardcrimson, fill=white, fill opacity=0.92, text opacity=1, rounded corners=2pt, inner sep=2.5pt, draw=harvardcrimson!50]
        {Uncorrected nominal trajectory ($h < 0$)};

    \end{axis}
\end{tikzpicture}
\end{document}
