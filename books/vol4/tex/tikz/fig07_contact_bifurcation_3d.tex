% =========================================================================
% LEFT PANEL: 3D Peg-in-Hole Contact Bifurcation
% =========================================================================
\begin{scope}[tdplot_main_coords, shift={(0,0,0)}]
    
    \def\bh{2.2}       % block height
    \def\bd{4.5}       % block depth
    \def\chw{0.45}     % chamfer width
    \def\chh{0.35}     % chamfer height
    \def\slotw{0.90}   % slot gap width
    \def\xleft{1.7}    % left chamfer start (shoulder)
    \def\xslotl{2.15}  % left slot wall (xleft + chw)
    \def\xslotr{3.05}  % right slot wall (xslotl + slotw)
    \def\xright{3.5}   % right chamfer end (xslotr + chw)
    \def\xmax{5.5}     % right block end
    \def\xmin{-0.4}    % left block start
    
    % --- LEFT BLOCK ---
    % Front face (x=xmin to xslotl)
    \draw[fill=ethdarkblue!12, draw=ethdarkblue, thick] 
        (\xmin,0,0) -- (\xslotl,0,0) -- (\xslotl,0,\bh-\chh) -- (\xleft,0,\bh) -- (\xmin,0,\bh) -- cycle;
    % Top flat face (xmin to xleft)
    \draw[fill=ethdarkblue!6, draw=ethdarkblue, thick] 
        (\xmin,0,\bh) -- (\xleft,0,\bh) -- (\xleft,\bd,\bh) -- (\xmin,\bd,\bh) -- cycle;
    % Chamfer face (xleft to xslotl, sloping down)
    \draw[fill=ethbronze!24, draw=ethdarkblue, thick] 
        (\xleft,0,\bh) -- (\xslotl,0,\bh-\chh) -- (\xslotl,\bd,\bh-\chh) -- (\xleft,\bd,\bh) -- cycle;
    % Inner slot vertical face (facing slot)
    \draw[fill=ethdarkblue!20, draw=ethdarkblue, thick] 
        (\xslotl,0,0) -- (\xslotl,\bd,0) -- (\xslotl,\bd,\bh-\chh) -- (\xslotl,0,\bh-\chh) -- cycle;

    % --- RIGHT BLOCK ---
    % Front face
    \draw[fill=ethdarkblue!15, draw=ethdarkblue, thick] 
        (\xslotr,0,0) -- (\xmax,0,0) -- (\xmax,0,\bh) -- (\xright,0,\bh) -- (\xslotr,0,\bh-\chh) -- cycle;
    % Chamfer face
    \draw[fill=ethbronze!24, draw=ethdarkblue, thick] 
        (\xslotr,0,\bh-\chh) -- (\xright,0,\bh) -- (\xright,\bd,\bh) -- (\xslotr,\bd,\bh-\chh) -- cycle;
    % Top flat face
    \draw[fill=ethdarkblue!6, draw=ethdarkblue, thick] 
        (\xright,0,\bh) -- (\xmax,0,\bh) -- (\xmax,\bd,\bh) -- (\xright,\bd,\bh) -- cycle;
    % Inner slot vertical face
    \draw[fill=ethdarkblue!28, draw=ethdarkblue, thick] 
        (\xslotr,0,0) -- (\xslotr,\bd,0) -- (\xslotr,\bd,\bh-\chh) -- (\xslotr,0,\bh-\chh) -- cycle;

    % Slot floor
    \draw[fill=ethdarkblue!35, draw=ethdarkblue] 
        (\xslotl,0,0) -- (\xslotr,0,0) -- (\xslotr,\bd,0) -- (\xslotl,\bd,0) -- cycle;

    % Chamfer highlight lines
    \draw[ethbronze!90!black, line width=1.3pt] (\xleft,0,\bh) -- (\xleft,\bd,\bh);
    \draw[ethbronze!90!black, line width=1.3pt] (\xright,0,\bh) -- (\xright,\bd,\bh);

    % Centerline of the slot
    \def\xc{2.60}  % center of slot
    \def\hy{2.2}   % y-position of trajectory plane

    % Bifurcation height
    \def\bifz{4.2}
    
    % Common approach path down to bifurcation
    \draw[ethpetrol, line width=2.4pt, ->] (\xc, \hy, 6.4) -- (\xc, \hy, \bifz);

    % Trajectory A: Clean insertion straight down into slot
    \draw[emeraldpine, line width=2.4pt, ->] 
        (\xc, \hy, \bifz) -- (\xc, \hy, 0.4);

    % Trajectory B: Perturbed by 0.5 mm to the left (hits chamfer shoulder at x=1.65)
    \def\tBx{1.65} % hits left shoulder
    \draw[harvardcrimson, line width=2.4pt, ->] 
        (\xc, \hy, \bifz) .. controls (\tBx+0.35, \hy, \bifz-0.8) .. (\tBx, \hy, \bh+0.05);

    % Jamming impact flash / starburst
    \coordinate (impact) at (\tBx, \hy, \bh);
    \coordinate (approach_top) at (\xc, \hy, 6.4);
    \coordinate (bifurcation) at (\xc, \hy, \bifz);
    \coordinate (trajB_mid) at ($(\xc, \hy, \bifz)!0.5!(\tBx, \hy, \bh)$);

    \node[starburst, fill=harvardcrimson, draw=harvardcrimson!50!black, line width=0.8pt, inner sep=3.5pt] 
        at (impact) {};

\end{scope}

% Contact force vectors at jamming point (reaction forces on peg, pointing up & right into open space)
\draw[-{Stealth[scale=1.2]}, harvardcrimson, line width=2.2pt] 
    (impact) -- ++(0.9, 1.2) node[above right=-2pt, font=\bfseries\small] {$\vec{F}_N$};
\draw[-{Stealth[scale=1.2]}, harvardcrimson, line width=2.2pt] 
    (impact) -- ++(0.2, 1.5) node[above=2pt, font=\bfseries\small] {$\vec{F}_{\text{jam}}$};

% Nominal Approach label (centered above peg path)
\node[anchor=south, font=\small\bfseries, text=ethpetrol, align=center] at ($(approach_top) + (0, 0.25)$) 
    {Nominal Peg Approach\\[-1pt]\footnotesize (Guided insertion vector)};

% 2D Horizontal Dimension Line for Delta x (clean drafting style with vertical extension line)
\coordinate (d_nom) at ($(approach_top) + (0, -1.1)$);
\coordinate (d_pert) at (impact |- d_nom);
\draw[|<->|, ink, thick] (d_nom) -- (d_pert);
\node[anchor=west, font=\scriptsize\bfseries] at ($(d_pert) + (0.1, 0)$) {$\Delta x = 0.5\text{ mm}$};
\draw[ink, dotted, thick] (d_pert) -- ($(impact) + (0, 0.4)$);

% Trajectory B label and callout (pointing cleanly to jamming impact without crossing force arrows)
\node[anchor=west, font=\small\bfseries, text=harvardcrimson, align=left] at (4.2, 2.3) 
    {\textbf{Trajectory B: Edge collision}\\[-1pt]\footnotesize Non-Lipschitz jamming};
\draw[harvardcrimson, thick, ->] (4.1, 2.3) -- ($(impact) + (0.45, 0.05)$);

% Trajectory A label (placed cleanly on the right below the slot)
\node[anchor=north west, font=\small\bfseries, text=emeraldpine, align=left] at (4.2, 0.2) 
    {\textbf{Trajectory A: Clean insertion}\\[-1pt]\footnotesize Monotonic entry into slot};
\draw[emeraldpine, thick, ->] (4.1, 0.0) -- (2.7, -0.2);

% Chamfer note (cleanly positioned on the lower left pointing to chamfer face)
\node[anchor=north east, font=\scriptsize\bfseries, text=ethbronze!80!black] at (-0.2, -0.3) {Chamfer lead-in};
\draw[ethbronze!80!black, -{Stealth[scale=0.9]}, thick] (-0.1, -0.2) -- (0.8, 0.6);

% =========================================================================
% RIGHT PANEL: Force-Displacement Profile
% =========================================================================

    % Card background

    % Title with ample top clearance
    \node[anchor=north, font=\bfseries\normalsize, text=ink] at (2.85, 5.5) {Force-Displacement Profile};

    % Axes (Force axis limited to 3.9 cm for plenty of room below title)
    \draw[->, thick, ink] (0,0) -- (5.2,0) node[right=2pt, font=\small\bfseries] {Displacement ($z$)};
    \draw[->, thick, ink] (0,0) -- (0,3.9) node[above=2pt, font=\small\bfseries] {Force ($F_z$)};
    
    \node[anchor=north east, font=\small, text=softink] at (-0.05, 0) {0};

    \draw[harvardcrimson, line width=2.2pt] 
        (0, 0.12) -- (1.6, 0.12) -- (1.68, 3.2);
    
    % Curve Labels (placed cleanly in the open interior of the card)
    \node[text=harvardcrimson, font=\small\bfseries, anchor=west, align=left] at (1.9, 2.5) 
        {\textbf{Rigid jamming}\\[-1pt]\scriptsize Spikes to torque limit};
    \draw[harvardcrimson, thin, ->] (1.85, 2.5) -- (1.72, 2.5);

    \node[text=emeraldpine, font=\small\bfseries, anchor=south, align=center] at (3.4, 1.25) 
        {\textbf{Smooth insertion}\\[-1pt]\scriptsize Monotonic low contact force};
    \draw[emeraldpine, thin, ->] (3.4, 1.2) -- (3.4, 0.75);

    % Divergence annotation
    \draw[<->, ink!70, thick] (4.6, 0.90) -- (4.6, 3.1) 
        node[midway, right=2pt, font=\scriptsize\bfseries, text=harvardcrimson, align=left] {Orders-of-\\magnitude\\divergence};

\end{scope}
