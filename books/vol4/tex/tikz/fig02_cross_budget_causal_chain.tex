% Define a macro for a node with header and content
\newcommand{\chainnode}[6]{
    % #1 = node name
    % #2 = color
    % #3 = header text
    % #4 = title text
    % #5 = subtitle text
    % #6 = placement (e.g., below=of previous)
    
    \node[box=#2, #6] (#1) {
        \vspace{0.15cm}
        \textcolor{white}{\textbf{#3}}\\[0.35cm]
        {\small\color{ink}\textbf{#4}\\
        #5}\\[0.25cm]
    };
}
