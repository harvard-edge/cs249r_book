\documentclass[tikz,border=5mm]{standalone}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz-3dplot}
\usepackage{pgfplots}
\usetikzlibrary{decorations.markings, arrows.meta}
\pgfplotsset{compat=1.18}

\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\tdplotsetmaincoords{70}{110}

\begin{tikzpicture}[tdplot_main_coords, >=stealth, font=\sffamily]

% Axes
\draw[->, thick, ink] (0,0,0) -- (6,0,0) node[anchor=north east]{Position $x$ (mm)};
\draw[->, thick, ink] (0,0,0) -- (0,7,0) node[anchor=north west]{Velocity $v$ (mm/s)};
\draw[->, thick, ink] (0,0,0) -- (0,0,5) node[anchor=south]{Friction force $F_f$ (N)};

% Surface (Stribeck Curve)
\foreach \x in {0, 0.5, ..., 5.0} {
    \draw[emeraldpine, opacity=0.3, thin] (\x, 0, 4) 
    \foreach \v in {0.1, 0.2, ..., 6.0} {
        -- (\x, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v})
    };
}
\foreach \v in {0, 0.5, ..., 6.0} {
    \draw[emeraldpine, opacity=0.3, thin] (0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v}) -- (5.0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v});
}

% Highlight the Stribeck curve on x=0 plane
\draw[emeraldpine, very thick] (0, 0, 4) 
\foreach \v in {0.1, 0.2, ..., 6.0} {
    -- (0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v})
};

% Stribeck velocity line
\draw[dashed, ink] (0,0.8,0) -- (0,0.8,{1.5 + 2.5/(1 + 1) + 0.2*0.8});
\node[anchor=north, align=center, ink] at (0, 0.8, -0.2) {Stribeck\\velocity $v_s$};

\node[anchor=east, emeraldpine] at (0, 0, 4) {Static friction $\mu_s$};
\node[anchor=north west, emeraldpine] at (0, 6, {1.5 + 2.5/(1 + (6/0.8)*(6/0.8)) + 0.2*6}) {Coulomb friction $\mu_k$};

% Stick-slip limit cycle (red)
\pgfmathsetmacro{\startx}{2.5 + 1.2*cos(0)}
\pgfmathsetmacro{\startv}{1.0 + 1.0*sin(0)}
\pgfmathsetmacro{\startf}{1.5 + 2.5/(1 + (\startv/0.8)*(\startv/0.8)) + 0.2*\startv}
\draw[harvardcrimson, very thick, decoration={markings, mark=at position 0.6 with {\arrow{stealth}}}, postaction={decorate}] 
    (\startx, \startv, \startf)
    \foreach \t in {5, 10, ..., 355} {
        \pgfextra{
            \pgfmathsetmacro{\xval}{2.5 + 1.2*cos(\t)}
            \pgfmathsetmacro{\vval}{1.0 + 1.0*sin(\t)}
            \pgfmathsetmacro{\fval}{1.5 + 2.5/(1 + (\vval/0.8)*(\vval/0.8)) + 0.2*\vval}
        }
        -- (\xval, \vval, \fval)
    } -- cycle;

\node[harvardcrimson, anchor=north] at (4.0, 1.0, 2) {Stick-slip limit cycle};

% Stable sliding (green)
\pgfmathsetmacro{\gstartx}{2.5 + 0.6*cos(0)}
\pgfmathsetmacro{\gstartv}{4.5 + 0.6*sin(0)}
\pgfmathsetmacro{\gstartf}{1.5 + 2.5/(1 + (\gstartv/0.8)*(\gstartv/0.8)) + 0.2*\gstartv}
\draw[ethpetrol, very thick, decoration={markings, mark=at position 0.6 with {\arrow{stealth}}}, postaction={decorate}] 
    (\gstartx, \gstartv, \gstartf)
    \foreach \t in {5, 10, ..., 355} {
        \pgfextra{
            \pgfmathsetmacro{\xval}{2.5 + 0.6*cos(\t)}
            \pgfmathsetmacro{\vval}{4.5 + 0.6*sin(\t)}
            \pgfmathsetmacro{\fval}{1.5 + 2.5/(1 + (\vval/0.8)*(\vval/0.8)) + 0.2*\vval}
        }
        -- (\xval, \vval, \fval)
    } -- cycle;

\node[ethpetrol, anchor=north east] at (2.5, 4.0, 3.5) {Stable sliding};

\end{tikzpicture}
\end{document}
