\usepackage{pgfplots}

% Official Volume Palette
\definecolor{emeraldpine}{HTML}{1A4D3E}
\definecolor{harvardcrimson}{HTML}{A51C30}
\definecolor{ethdarkblue}{HTML}{1F407A}
\definecolor{ethpetrol}{HTML}{007A87}
\definecolor{ethbronze}{HTML}{B87333}
\definecolor{ink}{HTML}{1A202C}
\definecolor{softink}{HTML}{4A5568}
\definecolor{paleink}{HTML}{A0AEC0}
\definecolor{cardbg}{HTML}{F8FAFC}

\begin{document}
\tdplotsetmaincoords{70}{110}

\begin{tikzpicture}[>=stealth, font=\sffamily]

% =============================================================
% 3D Geometry Scope
% =============================================================
\begin{scope}[tdplot_main_coords]

    % Coordinate Axes with Regular Ticks
    % X Axis: Position x (displacement)
    \draw[->, thick, ink] (0,0,0) -- (6.2,0,0) node[anchor=north east, font=\small\bfseries]{Position $x$ (displacement)};
    \foreach \px in {1, 2, 3, 4, 5} {
        \draw[ink, thin] (\px, 0, 0) -- (\px, 0.12, 0);
    }
    \foreach \px in {2, 4} {
        \node[anchor=south east, font=\tiny, softink] at (\px, 0.15, 0) {\px};
    }

    % Y Axis: Velocity v = \dot{x}
    \draw[->, thick, ink] (0,0,0) -- (0,7.2,0) node[anchor=north west, font=\small\bfseries]{Velocity $v = \dot{x}$ (mm/s)};
    \foreach \vy in {1, 2, 3, 4, 5, 6} {
        \draw[ink, thin] (0, \vy, 0) -- (0.12, \vy, 0);
    }
    \foreach \vy in {2, 4, 6} {
        \node[anchor=north east, font=\tiny, softink] at (0.15, \vy, -0.05) {\vy};
    }

    % Z Axis: Friction force / torque
    \draw[->, thick, ink] (0,0,0) -- (0,0,5.3) node[anchor=south, font=\small\bfseries, align=center]{Friction force $F_f$ / torque $\tau_f$ (N)};
    \foreach \fz in {1, 2, 3} {
        \draw[ink, thin] (0, 0, \fz) -- (-0.12, 0, \fz);
        \node[anchor=east, font=\tiny, softink] at (-0.15, 0, \fz) {\fz};
    }
    % Peak static tick at z=4 (replaces generic '4' to eliminate collision)
    \draw[emeraldpine, thick] (0, 0, 4) -- (-0.15, 0, 4);
    \node[anchor=east, font=\scriptsize\bfseries, emeraldpine] at (-0.18, 0, 4) {Breakaway $F_S$ ($\mu_s$)};

    % =============================================================
    % 3D Stribeck Friction Manifold Surface
    % Formula: F_f(v) = 1.5 + 2.5 / (1 + (v/0.8)^2) + 0.2 * v
    % =============================================================

    % Longitudinal grid curves (along velocity v)
    \foreach \x in {0.5, 1.0, ..., 4.5} {
        \draw[emeraldpine, opacity=0.22, thin] (\x, 0, 4) 
        \foreach \v in {0.1, 0.2, ..., 6.0} {
            -- (\x, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v})
        };
    }

    % Transverse grid lines (along position x)
    \foreach \v in {0.4, 0.8, ..., 6.0} {
        \draw[emeraldpine, opacity=0.22, thin] (0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v}) -- (5.0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v});
    }

    % Back boundary curve on x=0 plane (Primary Stribeck Profile)
    \draw[emeraldpine!90!black, very thick] (0, 0, 4) 
    \foreach \v in {0.1, 0.2, ..., 6.0} {
        -- (0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v})
    };

    % Front boundary curve on x=5.0 plane
    \draw[emeraldpine!60!black, thick] (5.0, 0, 4) 
    \foreach \v in {0.1, 0.2, ..., 6.0} {
        -- (5.0, \v, {1.5 + 2.5/(1 + (\v/0.8)*(\v/0.8)) + 0.2*\v})
    };

    % Static friction peak marker (v = 0)
    \fill[emeraldpine] (0, 0, 4) circle (2pt);

    % Stribeck velocity threshold line (v = v_s = 0.8)
    \pgfmathsetmacro{\fstribeck}{1.5 + 2.5/(1 + 1) + 0.2*0.8}
    \draw[dashed, ethbronze, thick] (0, 0.8, 0) -- (0, 0.8, \fstribeck);
    \fill[ethbronze] (0, 0.8, \fstribeck) circle (2pt);
    \fill[ethbronze] (0, 0.8, 0) circle (1.5pt);
    \node[anchor=north, align=center, font=\scriptsize\bfseries, text=ethbronze] at (0, 0.8, -0.15) {$v_s$\\[-1pt]\tiny (Stribeck)};

    % Coulomb friction level (v -> large)
    \pgfmathsetmacro{\fcoulomb}{1.5 + 2.5/(1 + (6.0/0.8)*(6.0/0.8)) + 0.2*6.0}
    \fill[emeraldpine] (0, 6.0, \fcoulomb) circle (2pt);
    \node[anchor=west, font=\scriptsize\bfseries, emeraldpine] at (0, 6.1, \fcoulomb) {Coulomb + viscous $\mu_k$};

    % 1. Stick-Slip Limit Cycle (Low velocity: v < v_s, negative damping)
    \pgfmathsetmacro{\startx}{2.5 + 1.2*cos(0)}
    \pgfmathsetmacro{\startv}{1.0 + 0.8*sin(0)}
    \pgfmathsetmacro{\startf}{1.5 + 2.5/(1 + (\startv/0.8)*(\startv/0.8)) + 0.2*\startv}

    \draw[harvardcrimson, line width=1.6pt, decoration={markings, mark=at position 0.55 with {\arrow{stealth}}}, postaction={decorate}] 
        (\startx, \startv, \startf)
        \foreach \t in {5, 10, ..., 355} {
            \pgfextra{
                \pgfmathsetmacro{\xval}{2.5 + 1.2*cos(\t)}
                \pgfmathsetmacro{\vval}{1.0 + 0.8*sin(\t)}
                \pgfmathsetmacro{\fval}{1.5 + 2.5/(1 + (\vval/0.8)*(\vval/0.8)) + 0.2*\vval}
            }
            -- (\xval, \vval, \fval)
        } -- cycle;

    % Target coordinate on the lower rim of the orbit
    \coordinate (stick_target) at (2.5, 1.2, 1.8);

    % 2. Stable Steady Sliding (High velocity: v > v_s, positive damping)
    \pgfmathsetmacro{\gstartx}{2.5 + 0.6*cos(0)}
    \pgfmathsetmacro{\gstartv}{4.5 + 0.6*sin(0)}
    \pgfmathsetmacro{\gstartf}{1.5 + 2.5/(1 + (\gstartv/0.8)*(\gstartv/0.8)) + 0.2*\gstartv}

    \draw[ethpetrol, line width=1.6pt, decoration={markings, mark=at position 0.55 with {\arrow{stealth}}}, postaction={decorate}] 
        (\gstartx, \gstartv, \gstartf)
        \foreach \t in {5, 10, ..., 355} {
            \pgfextra{
                \pgfmathsetmacro{\xval}{2.5 + 0.6*cos(\t)}
                \pgfmathsetmacro{\vval}{4.5 + 0.6*sin(\t)}
                \pgfmathsetmacro{\fval}{1.5 + 2.5/(1 + (\vval/0.8)*(\vval/0.8)) + 0.2*\vval}
            }
            -- (\xval, \vval, \fval)
        } -- cycle;

    \coordinate (stable_target) at (2.5, 4.5, {1.5 + 2.5/(1 + (4.5/0.8)*(4.5/0.8)) + 0.2*4.5});

\end{scope}

% =============================================================
% 2D Overlay Scope for Perfectly Placed Annotation Badges
% =============================================================

% Stick-slip Callout Box: Placed in open space between z-axis and x-axis
\node[fill=white, draw=harvardcrimson, thick, rounded corners=3pt, inner sep=4pt, align=center] (stick_callout) at (-2.0, 1.8) {
    \textcolor{harvardcrimson}{\scriptsize\bfseries Stick-Slip Limit Cycle}\\[2pt]
    \tiny Negative damping: $dF_f/dv < 0$\\[1pt]
    \tiny Unstable chatter / limit cycle ($v < v_s$)
};
% Clean leader arrow with white halo crossing the vertical axis
\draw[white, line width=3pt] (stick_callout.east) -- (stick_target);
\draw[->, thick, harvardcrimson] (stick_callout.east) -- (stick_target);

% Stable sliding Callout Box: Placed in wide open space above surface on the right
\node[fill=white, draw=ethpetrol, thick, rounded corners=3pt, inner sep=4pt, align=center] (stable_callout) at (4.5, 3.0) {
    \textcolor{ethpetrol}{\scriptsize\bfseries Stable Sliding Regime}\\[2pt]
    \tiny Positive damping: $dF_f/dv > 0$\\[1pt]
    \tiny Perturbations decay into smooth motion
};
% Clean leader arrow pointing down-left into stable attractor
\draw[->, thick, ethpetrol] (stable_callout.south west) -- (stable_target);

\end{tikzpicture}
