\begin{tikzpicture}[tdplot_main_coords, >=Stealth, font=\sffamily\small, scale=1.5]

% Coordinates
\def\robotW{1.2}
\def\robotH{0.8}
\def\robotL{1.5}

\def\blindL{2.5}
\def\brakeL{3.0}

\def\obsW{1.8}
\def\obsH{1.2}
\def\obsL{0.4}

% Ground grid
\filldraw[fill=shadecolor, draw=paleink, thick] (-2, -3, 0) -- (\blindL+\brakeL+2, -3, 0) -- (\blindL+\brakeL+2, 4, 0) -- (-2, 4, 0) -- cycle;
\draw[paleink, step=1.0] (-2, -3, 0) grid (\blindL+\brakeL+2, 4, 0);

% Draw Robot (-robotL to 0 in X)
\filldraw[fill=ethdarkblue!80, draw=ink, thick] (-\robotL, -\robotW/2, 0) -- (0, -\robotW/2, 0) -- (0, \robotW/2, 0) -- (-\robotL, \robotW/2, 0) -- cycle; % Bottom
\filldraw[fill=ethdarkblue!70, draw=ink, thick] (-\robotL, -\robotW/2, 0) -- (-\robotL, -\robotW/2, \robotH) -- (-\robotL, \robotW/2, \robotH) -- (-\robotL, \robotW/2, 0) -- cycle; % Back (x=min)
\filldraw[fill=ethdarkblue!80, draw=ink, thick] (-\robotL, -\robotW/2, 0) -- (0, -\robotW/2, 0) -- (0, -\robotW/2, \robotH) -- (-\robotL, -\robotW/2, \robotH) -- cycle; % Left (y=min)
\filldraw[fill=ethdarkblue!80, draw=ink, thick] (-\robotL, \robotW/2, 0) -- (0, \robotW/2, 0) -- (0, \robotW/2, \robotH) -- (-\robotL, \robotW/2, \robotH) -- cycle; % Right (y=max)
\filldraw[fill=ethdarkblue!60, draw=ink, thick] (0, -\robotW/2, 0) -- (0, -\robotW/2, \robotH) -- (0, \robotW/2, \robotH) -- (0, \robotW/2, 0) -- cycle; % Front (x=max)
\filldraw[fill=ethdarkblue!90, draw=ink, thick] (-\robotL, -\robotW/2, \robotH) -- (0, -\robotW/2, \robotH) -- (0, \robotW/2, \robotH) -- (-\robotL, \robotW/2, \robotH) -- cycle; % Top

% Blind Travel Volume
\filldraw[fill=ethpetrol, fill opacity=0.3, draw=ethpetrol, thick] (0, -\robotW/2, 0) -- (0, -\robotW/2, \robotH) -- (0, \robotW/2, \robotH) -- (0, \robotW/2, 0) -- cycle; % Back (x=0)
\filldraw[fill=ethpetrol, fill opacity=0.3, draw=ethpetrol, thick] (0, -\robotW/2, 0) -- (\blindL, -\robotW/2, 0) -- (\blindL, -\robotW/2, \robotH) -- (0, -\robotW/2, \robotH) -- cycle; % Left (y=min)
\filldraw[fill=ethpetrol, fill opacity=0.3, draw=ethpetrol, thick] (0, \robotW/2, 0) -- (\blindL, \robotW/2, 0) -- (\blindL, \robotW/2, \robotH) -- (0, \robotW/2, \robotH) -- cycle; % Right (y=max)
\filldraw[fill=ethpetrol, fill opacity=0.4, draw=ethpetrol, thick] (\blindL, -\robotW/2, 0) -- (\blindL, -\robotW/2, \robotH) -- (\blindL, \robotW/2, \robotH) -- (\blindL, \robotW/2, 0) -- cycle; % Front (x=\blindL)
\filldraw[fill=ethpetrol, fill opacity=0.2, draw=ethpetrol, thick] (0, -\robotW/2, \robotH) -- (\blindL, -\robotW/2, \robotH) -- (\blindL, \robotW/2, \robotH) -- (0, \robotW/2, \robotH) -- cycle; % Top

% Braking Volume
\def\bx{\blindL}
\def\ex{\blindL+\brakeL}
\filldraw[fill=emeraldpine, fill opacity=0.3, draw=emeraldpine, thick] (\bx, -\robotW/2, 0) -- (\bx, -\robotW/2, \robotH) -- (\bx, \robotW/2, \robotH) -- (\bx, \robotW/2, 0) -- cycle; % Back
\filldraw[fill=emeraldpine, fill opacity=0.3, draw=emeraldpine, thick] (\bx, -\robotW/2, 0) -- (\ex, -\robotW/2, 0) -- (\bx, -\robotW/2, \robotH) -- cycle; % Left
\filldraw[fill=emeraldpine, fill opacity=0.3, draw=emeraldpine, thick] (\bx, \robotW/2, 0) -- (\ex, \robotW/2, 0) -- (\bx, \robotW/2, \robotH) -- cycle; % Right
\filldraw[fill=emeraldpine, fill opacity=0.2, draw=emeraldpine, thick] (\bx, -\robotW/2, \robotH) -- (\bx, \robotW/2, \robotH) -- (\ex, \robotW/2, 0) -- (\ex, -\robotW/2, 0) -- cycle; % Top slanted

% Obstacle
\def\ox{\blindL+\brakeL}
\filldraw[fill=harvardcrimson!80, draw=ink, thick] (\ox, -\obsW/2, 0) -- (\ox+\obsL, -\obsW/2, 0) -- (\ox+\obsL, \obsW/2, 0) -- (\ox, \obsW/2, 0) -- cycle; % Bottom
\filldraw[fill=harvardcrimson!70, draw=ink, thick] (\ox, -\obsW/2, 0) -- (\ox, -\obsW/2, \obsH) -- (\ox, \obsW/2, \obsH) -- (\ox, \obsW/2, 0) -- cycle; % Back (x=min)
\filldraw[fill=harvardcrimson!80, draw=ink, thick] (\ox, -\obsW/2, 0) -- (\ox+\obsL, -\obsW/2, 0) -- (\ox+\obsL, -\obsW/2, \obsH) -- (\ox, -\obsW/2, \obsH) -- cycle; % Left (y=min)
\filldraw[fill=harvardcrimson!80, draw=ink, thick] (\ox, \obsW/2, 0) -- (\ox+\obsL, \obsW/2, 0) -- (\ox+\obsL, \obsW/2, \obsH) -- (\ox, \obsW/2, \obsH) -- cycle; % Right (y=max)
\filldraw[fill=harvardcrimson!60, draw=ink, thick] (\ox+\obsL, -\obsW/2, 0) -- (\ox+\obsL, -\obsW/2, \obsH) -- (\ox+\obsL, \obsW/2, \obsH) -- (\ox+\obsL, \obsW/2, 0) -- cycle; % Front (x=max)
\filldraw[fill=harvardcrimson!90, draw=ink, thick] (\ox, -\obsW/2, \obsH) -- (\ox+\obsL, -\obsW/2, \obsH) -- (\ox+\obsL, \obsW/2, \obsH) -- (\ox, \obsW/2, \obsH) -- cycle; % Top

% Dimensions on ground plane, in foreground (y > 0)
\def\dimYBlind{\robotW/2 + 0.6}
\def\dimYBrake{\robotW/2 + 1.6}
\def\dimYTotal{\robotW/2 + 2.8}

% Projection lines from object to dimension lines
\draw[dashed, ink!50, thick] (0, \robotW/2, 0) -- (0, \dimYTotal, 0);
\draw[dashed, ink!50, thick] (\blindL, \robotW/2, 0) -- (\blindL, \dimYBrake, 0);
\draw[dashed, ink!50, thick] (\ox, \obsW/2, 0) -- (\ox, \dimYTotal, 0);

% Blind Travel
\draw[thick, ink] (0, \dimYBlind-0.2, 0) -- (0, \dimYBlind+0.2, 0);
\draw[thick, ink] (\blindL, \dimYBlind-0.2, 0) -- (\blindL, \dimYBlind+0.2, 0);
\draw[<->, thick, ethpetrol] (0, \dimYBlind, 0) -- (\blindL, \dimYBlind, 0) node[midway, sloped, above, font=\sffamily\bfseries] {Blind Travel ($v \cdot \tau_{\text{delay}}$)};

% Braking Travel
\draw[thick, ink] (\blindL, \dimYBrake-0.2, 0) -- (\blindL, \dimYBrake+0.2, 0);
\draw[thick, ink] (\ox, \dimYBrake-0.2, 0) -- (\ox, \dimYBrake+0.2, 0);
\draw[<->, thick, emeraldpine] (\blindL, \dimYBrake, 0) -- (\ox, \dimYBrake, 0) node[midway, sloped, above, font=\sffamily\bfseries] {Braking Travel ($v^2 / 2a_{\text{max}}$)};

% Total stopping envelope
\draw[thick, ink] (0, \dimYTotal-0.2, 0) -- (0, \dimYTotal+0.2, 0);
\draw[thick, ink] (\ox, \dimYTotal-0.2, 0) -- (\ox, \dimYTotal+0.2, 0);
\draw[<->, thick, ink] (0, \dimYTotal, 0) -- (\ox, \dimYTotal, 0) node[midway, sloped, above, font=\sffamily\bfseries] {Total Stopping Envelope ($d_{\text{stop}}$)};
