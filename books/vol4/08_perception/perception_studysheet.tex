\begin{bigidea}
A camera can name an obstacle correctly yet place it where it stood before the robot moved.
Each observation must state \hl{when it was captured, where it was measured from, and how far to trust it}.
The permission path can refuse only what it can check, and stale evidence that crosses the
\hl{causal boundary} becomes motion that cannot be recalled.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Choose the sensor by the quantity.}{Camera: bearing and class; LiDAR: metric range. A qualified
  range return vetoes motion, whatever the label says.}
\idea{Date at capture, not arrival.}{Use the exposure midpoint; the arrival
  stamp (DMA completion) is \val{arrival}\,ms late.}
\idea{Age is paid in travel.}{The base covers $\SpeedAtDetection\ObservationAge$ before anyone reads the frame;
  no later stage returns it.}
\idea{Fix geometry before interpreting it.}{Timing errors grow with speed, mount errors with range.}
\idea{An inset rests on measured residuals.}{A covariance or a network's spread is a belief; the
  inset uses a measured p99 residual, bias and outliers included.}
\idea{Silent failures are the dangerous ones.}{A misdated frame passes every check and
  narrows its spread around the wrong spot.}
\end{ideas}

\vspace{9pt}
\sheetsec{How each error grows}
\begin{sheettable}{@{}l l L@{}}
\thead{Error} & \thead{Size} & \thead{Grows with}\\ \midrule
Age travel & $\SpeedAtDetection\ObservationAge$ & speed and every stage\\
Rolling-shutter skew & $\SpeedAtDetection\FrameReadoutTime$ & speed; removed by IMU row poses\\
Misdated frame & $\SpeedAtDetection\,\FiniteChange\TimeInstant$ & speed, along travel; zero at rest\\
Mount angle & $\CameraFrameDepth\ExtrinsicMisalignmentAngle$ & range, sideways; present at rest\\
Stereo depth & $\DepthStandardDeviation$ & range squared\\
\end{sheettable}
\par\vspace{2pt}{\fontsize{8.6}{11}\selectfont\color{muted}$\FiniteChange\TimeInstant$: stamp error; $\CameraFrameDepth$: range; $\ExtrinsicMisalignmentAngle$: mount angle.}\par

\global\firstcardtrue
\fcard{Stereo depth and the clearance inset}%
  {\DepthStandardDeviation \approx \frac{\CameraFrameDepth^2\DisparityStandardDeviation}{\FocalLength\StereoBaseline},
   \qquad \ClearanceInset = \RequiredClearance + \lVert\SpatialErrorVector\rVert_{\text{p99}}}%

   baseline $\StereoBaseline$; set it equal to an allowance to find that range. The inset adds the
   measured p99 residual to the required clearance $\RequiredClearance$.}%
  {One standard deviation misses tail, bias and outliers.}{}

\sheetsec{Before reading the payload}
\begin{steps}
\item Checksum and health bits clean (sync, saturation, out-of-distribution).
\item Clock conversion valid: bound $\ClockConversionBound$ present, mapping unexpired.
\item Current age, on the consumer's own clock, within this use's limit.
\item Frame id and calibration revision valid.
\end{steps}
\vspace{2pt}Any failure: refuse and fall back.\par

\begin{answers}
\ans{Finish}{\val{speed}\,m/s $\times$ \val{batched}\,ms = \res{\val{batchtravel}\,mm}. \val{batched} + \val{sync} = \val{batchedclk}\,ms with the clock bound exceeds the
  \val{threshold}\,ms threshold: \res{refused at dispatch}. Its extra \val{extra}\,mm takes
  \val{extrashare}\% of the remainder (age card).}
\ans{1}{\val{slow}\,m/s $\times$ \val{readout}\,ms = \res{\val{skewslow}\,mm}, about \val{skewshare}\% of $\LocalizationBound$.
  It still spends a quarter of the estimate's allowance on an error IMU row poses remove (error table).}
\ans{2}{(a) \val{rack}$^2$ $\times$ \val{disptwo} $\div$ \val{fb} = \val{sigmeter}\,m = \res{\val{sigracktwo}\,mm}.
  (b) $\sqrt{\val{locm} \times \val{fb} \div \val{disptwo}}$ = \res{\val{zloctwo}\,m}. (c) $\DepthStandardDeviation$ is one
  standard deviation; the inset needs the measured p99 residual (inset card).}
\columnbreak
\ans{3}{Along travel, scaling with speed, gone at rest: timing; a mount error is sideways and stays parked. \val{gap}\,mm $\div$ \val{qspeed}\,m/s = \res{\val{gaptime}\,ms}, the arrival offset: consistent with DMA-completion stamping (error table).}
\ans{4}{\val{age} + \val{admit} + \val{sync} = \res{\val{current}\,ms} at most. No: that threshold is dispatch's,
  not admission's. The permission path recomputes age from the capture stamp on its own
  clock; the transit's \val{admextra}\,mm comes out of the remainder ($\LocalizationBound$ holds the clock bound) (age card).}
\end{answers}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{How old is the frame, and does the stop fit?}
Base at \val{speed}\,m/s. Camera stages (ms): half exposure \val{expohalf}, readout \val{readout}, transport
\val{transport}, DMA \val{dma}, ISP \val{isp}, backbone \val{backbone}, IPC dispatch \val{ipc}. Clock bound \val{sync}\,ms;
braking, overhead and lease travel need \val{before} of \val{clearmm}\,mm.\par\vspace{5pt}

  \node[small, anchor=south] at ({(\val{pisp}+\val{pdisp})/2},13) {backbone + IPC};
  \draw[dim, dashed] (\val{pdisp},-4) -- (\val{pdisp},24);
  \node[small, anchor=south] at (\val{pdisp},24) {dispatch};
  \node[note, anchor=north] at (103,-4) {to scale, from the exposure midpoint};
\end{tikzpicture}\par}\vspace{4pt}
\begin{steps}
\item Sum the stages: $\ObservationAge$ = \res{\val{age}\,ms}.
\item Travel: \val{speed}\,m/s $\times$ \val{age}\,ms = \res{\val{travel}\,mm}.
\item Dispatch threshold: \val{age} + \val{sync} = \res{\val{threshold}\,ms}, met with no spare.
\item Partial stop: \val{before} + \val{travel} = \res{\val{after}\,mm}; the \val{remain}\,mm left must absorb
  admission delay, jerk-limited braking and the inset.
\end{steps}
\takeaway{Exposure and readout spend half the age before computing starts.}
\end{practicebox}

\begin{practicebox}\boxtitle{Same pipeline, batched frames}
Perception batches \val{batch} frames at \val{fps}\,Hz.\par\vspace{4pt}
\textbf{Given:} the oldest frame waits three frame periods (\val{wait}\,ms), so its age is
\val{age} + \val{wait} = \val{batched}\,ms.\par
\textbf{You:} find its travel, admit or refuse at dispatch, and the extra travel's share of the \val{remain}\,mm remainder.\par\vspace{30pt}
\end{practicebox}

\begin{rightcol}
\sheetsec{Common mistakes}
\mistake{``The scene model is confident the patch ahead is a shadow, so the LiDAR return there can be dropped.''}%
  {Confidence describes appearance at a past exposure, not clearance. A qualified LiDAR return inside the
   stopping envelope vetoes: slow or halt.}{}
\mistake{``The driver reports no errors, so the frame is good.''}%
  {Stamped at arrival, the frame is \val{arrival}\,ms late: at \val{speed}\,m/s the obstacle sits \val{arrivaltravel}\,mm
   off, with growing confidence. A dropped frame at least ages honestly, by \val{frameperiod}\,ms.}{}

  Upper bound on its age? Refuse it for exceeding \val{threshold}\,ms? Who computes the age, and how?
\end{qs}
