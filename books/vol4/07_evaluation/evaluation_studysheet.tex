\frontside
\begin{bigidea}
Twenty clean door openings are evidence about twenty runs, not readiness. Evaluation turns a finite
record into a \hl{bounded claim}. Work across the \hl{causal boundary} cannot be recalled, so each
untested condition falls to the permission path.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{A success rate belongs to a population.}{A new floor or unit is a new population.}
\idea{Replay cannot see the loop.}{Inputs reset to the log, so bias never builds; on the
  machine it drifts.}
\idea{Credit only what the proposer did.}{Completion, autonomy and interventions are separate;
  every run counts; a stop credits the safeguard.}
\idea{Regimes trade blind spots.}{Not a fidelity ladder: each question goes to the regime that
  exposes it.}
\idea{Each added nine costs about ten times the trials.}{The claim fixes the count; early stops need a
  set boundary.}
\idea{Past the exposure wall, architecture carries the claim.}{Untested conditions become
  monitors or hard limits.}
\end{ideas}

  {ACR over all $\TrialCount$ runs; MTBI over every clip, stop and takeover.}%
  {A stop fails autonomy; whether it would have crashed is censored.}{}

\fcard{Zero-failure bound}%
  {\SuccessLowerConfidenceLimit = \HypothesisTestSignificanceLevel^{1/\ZeroFailureTrialCount},
   \quad \FailureUpperConfidenceLimit \approx \frac{3}{\ZeroFailureTrialCount},
   \quad \ZeroFailureTrialCount = \left\lceil \frac{\ln(1/\HypothesisTestSignificanceLevel)}{\ln(1/\TargetTrialSuccessProbability)} \right\rceil}%
  {Lower limit after $\ZeroFailureTrialCount$ clean runs at confidence $1-\HypothesisTestSignificanceLevel$;
   $\ln(1/\HypothesisTestSignificanceLevel)$ = \val{lninv} at \val{conf}\%.}%
  {Independent draws, one cell, full denominator. Never restart the counter after a failure: binomial
   lower limit, \val{nwithfail} of \val{nclean} $\rightarrow$ \val{pfailpct}\%, not \val{plowpct}\%.}%

\fcard{Sequential test (SPRT)}%
  {\begin{gathered}\SequentialLogLikelihoodRatio = \ObservedTrialFailureCount\ln\frac{\HypothesizedTrialFailureProbability_1}{\HypothesizedTrialFailureProbability_0}
     + (\SequentialTrialCount-\ObservedTrialFailureCount)\ln\frac{1-\HypothesizedTrialFailureProbability_1}{1-\HypothesizedTrialFailureProbability_0}\\[2pt]
   \SprtUpperBoundary \approx \ln\frac{1-\StatisticalFalseNegativeRate}{\HypothesisTestSignificanceLevel},
   \qquad \SprtLowerBoundary \approx \ln\frac{\StatisticalFalseNegativeRate}{1-\HypothesisTestSignificanceLevel}\end{gathered}}%
  {$\SequentialTrialCount$ runs, $\ObservedTrialFailureCount$ failures; $\HypothesizedTrialFailureProbability_0$,
   $\HypothesizedTrialFailureProbability_1$: unacceptable, acceptable failure probabilities; $\StatisticalFalseNegativeRate$,
   $\HypothesisTestSignificanceLevel$: allowed false rejection, false acceptance. Here a clean run adds \val{incs}, a failure subtracts \val{incf}; accept at
   $+$\val{bup}, reject at $-$\val{blo}.}%
  {Declare it before the first run; never stop on a good streak.}{}
\fcard{Exposure wall}%
  {\FailureObservationInterval \ge \frac{\ln(1/\HypothesisTestSignificanceLevel)}{\ConstantHazardRate} \approx \frac{3}{\ConstantHazardRate}}%
  {Failure-free hours to bound hazard rate $\ConstantHazardRate$ per hour.}%
  {Constant rate. Rigs add hours, not diversity.}{}

\begin{answers}
\ans{Finish}{$\lceil\val{lninv}/\val{ylnn}\rceil$ = \res{\val{yruns} runs}, near the rule of three's \val{yrule}; \val{yruns}
  $\times$ \val{tcycle}\,s = \val{ysec}\,s = \res{\val{yhours}\,h}. One failure rejects the zero-failure target; a fixed-count record with failures takes the
  binomial limit (\val{nwithfail} of \val{nclean}: \val{pfailpct}\%), never a restarted counter (zero-failure card).}
\ans{1}{\val{qdone}/\val{qruns} = \res{\val{qdonepct}\%} done; \val{qunass}/\val{qruns} = \res{\val{qacrpct}\%} ACR;
  \val{qruns} $\times$ \val{runtime}\,s = \val{qactive}\,min over \val{qevents} events = \res{\val{qmtbi}\,min} MTBI. No: each
  stop is an autonomy failure, a censored crash (channel metrics).}
\ans{2}{(a) Shadow: replay first for compute; then shadow, hot on real buses. (b) Simulation, a screen only. (c) Constrained
  testing: only real contact shows the force (regimes).}
\columnbreak
\ans{3}{\val{qclean} $\times$ \val{incs} $-$ \val{qfirst} = \res{$-$\val{qscore}}, between the boundaries: continue.
  (\val{bup} $+$ \val{qscore}) / \val{incs} = \val{qratio}, so \res{\val{qmore} clean runs}. Stopping on a good streak
  inflates false acceptance (SPRT card).}
\ans{4}{\val{lninv} / $\val{qlammath}$ = \res{\val{qhours}\,h}, or \val{qyears} rig-years: \res{\val{qcal} years} with
  \val{qrigs} rigs, valid only for conditions they share. The permission path and safeguards carry it; the record lists it as a gap with its monitor (record).}
\end{answers}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{What do twenty clean runs support, and what does \val{tgtninetynine}\% cost?}
The latch policy opens the cage door \val{nclean} times, no tripwire crossing, no intervention. Confidence \val{conf}\% ($\HypothesisTestSignificanceLevel$ = \val{alpha}); each trial is a
\val{runtime}\,s run plus a \val{treset}\,s reset.
\begin{steps}
\item $\ln(1/\HypothesisTestSignificanceLevel)$ = \val{lninv}; $\div$ \val{nclean} = \val{wexpo};
  $\SuccessLowerConfidenceLimit = e^{-\val{wexpo}}$ = \res{\val{plowpct}\%}.
\item Failure upper limit \res{\val{failuppct}\%} (rule of three: \val{rulethree}).
\item For $\TargetTrialSuccessProbability$ = \val{tgtninetynine}\%: $\lceil\val{lninv}/\val{wlnnn}\rceil = \lceil\val{wratio}\rceil$
  = \res{\val{runsninetynine} clean runs}.
\item Cell time: \val{runsninetynine} $\times$ \val{tcycle}\,s = \val{wsec}\,s = \res{\val{whours}\,h}, one cell only.
\end{steps}
\takeaway{This bounds one cell at \val{plowpct}\%; a policy failing \val{failuppct}\% of runs still fits.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``Its angular error is only $\val{msemath}$\,\textup{rad$^2$} on the logs, so it drives straight.''}%
  {A \val{headdeg}\textdegree{} heading bias scores that on replay, where inputs return to the centerline; at
   \val{vdrive}\,m/s it spends the \val{sideclear}\,m side clearance in \val{timpact}\,s, \val{ximpact}\,m into a
   \val{aislelen}\,m aisle.}{}
\mistake{``A thousand clean runs prove a \val{tgtthreenines}\% policy.''}%
  {$\val{alpha}^{1/\val{nthousand}}$ = \val{pthousandpct}\%. The claim needs \val{runsthreenines} clean runs
   (\val{hthreenines}\,h of cell time).}{}
\mistake{``The candidate went \val{armb} of \val{armruns} against \val{armruns} of \val{armruns}: it is worse.''}%
  {Fisher's two-sided exact p-value = \val{fisherp}, above \val{alpha}. Separating \val{phigh}\% from \val{plow}\% at
   \val{powerpct}\% power needs about \val{perarm} interleaved, blinded runs per arm.}{}
