\begin{bigidea}
A machine's actions choose the states it sees next, and every demonstration is paid for in
\hl{cell time and component life}. An archive of successes says nothing about the states a first
mistake reaches, so physical data is engineered, down to the
\hl{known absences} it records.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{The policy's actions choose its data.}{One error can carry it into states no demonstration
  covers; more nominal runs only densify the corridor.}
\idea{Cloning error compounds.}{Worst case grows with the horizon squared; labeling the
  learner's states makes it linear while mistakes can be undone.}
\idea{One command, four taps.}{Requested, mapped, enforced, measured: imitate the mapped
  command only where the permission path left it unchanged.}
\idea{Date the frame at its exposure midpoint.}{A timing error becomes a position error at the
  speed of motion. The operator's delay is recorded, not erased.}
\idea{Coverage is a join, not a count.}{Join occupancy against a scenario table declared from
  the operating domain; the latch archive fills \val{supported} of \val{cells} cells.}
\idea{A label belongs to its body.}{Opening = normalized command $\times$ stroke, so a shorter stroke
  shrinks it. Gate borrowed labels in physical units: pass, relabel, or reject.}
\end{ideas}

\begin{answers}
\ans{Finish}{\val{tattempts} $\times$ \val{retention} = \res{\val{tkept} kept} per session. Attempts stay
  \val{campaign}, so \val{campaign} $\div$ \val{tattempts} = \res{\val{thours}\,h}, \val{tsaved}\,h less; cable life used
  stays \res{\val{cablepct}\%}. Only the cell-time bill moved (budget card).}
\ans{1}{\val{qspeed}\,m/s $\times$ \val{lag}\,ms = \res{\val{qdx}\,mm} \textgreater{} \val{clearance}\,mm: misfit. Ceiling
  \val{clearance} $\div$ \val{qspeed} = \res{\val{qceil}\,ms}. Stamped at midpoint, top-to-bottom skew alone gives \val{qskewdx}\,mm across the frame:
  fits, with almost no margin (pairing card).}
\ans{2}{$\BehaviorCloningHorizon$ = \val{qhorizon}; no information at or above \res{\val{qvacuous}}. Bound \res{\val{qclone}} of
  \val{qhorizon}; aggregation \val{qagg}. A bent latch cannot be undone: $\DeviationDisadvantage$ grows with $\BehaviorCloningHorizon$ and the
  advantage is lost; the permission path must refuse it (bound card).}
\columnbreak
\ans{3}{Barrier trips: intercepted, the permission path's job, not a policy target. No trip: check the authority
  record for a takeover (curate), exclusion codes and raw recordings for censoring (fix the pipeline); if
  all clean, never visited (collect) (table).}
\ans{4}{\val{gcommand} $\times$ \val{stroketarget}\,mm = \res{\val{opentarget}\,mm} \textless{} \val{part}\,mm: fingers hit the part.
  \val{opendonor}\,mm fits the \val{stroketarget}\,mm stroke, so relabel in millimeters (normalized \val{relabel}); force, rate and sensing
  still need checking (key idea on bodies).}
\end{answers}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{What does the latch campaign cost?}
Each scheduled \val{sched}\,min session leaves \val{motion}\,min of motion. An opening takes \val{attempt}\,s and the
fixture reset \val{reset}\,s; \val{failpct}\% of attempts fail and filtering drops \val{qapct}\% more. Keep
\val{target} demonstrations. Per attempt the wrist cable flexes \val{cableper} times (rated \val{cablelife}) and the
gear reverses \val{gearper} times (rated \val{gearlife}).
\begin{steps}
\item Attempts: $\lfloor$\val{motionsec}\,s $\div$ \val{cycle}\,s$\rfloor$ = \res{\val{attempts}} per session.
\item Retention: (1 $-$ \val{failshare})(1 $-$ \val{qashare}) = \val{retention}; kept \res{\val{kept}} per session.
\item Campaign: $\lceil$\val{target} $\div$ \val{retention}$\rceil$ = \val{campaign} attempts, $\div$ \val{attempts} =
  \res{\val{hours}\,h} of cell time.
\item Wear: \val{campaign} $\times$ \val{cableper} = \val{cablecycles} bends, \res{\val{cablepct}\%} of the cable;
  \val{gearcycles} reversals, \val{gearpct}\% of the gear.
\end{steps}
\takeaway{Under a quarter of each hour is kept, and the cable, not the gear, binds the campaign.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``We log \val{records} million records an hour, so coverage is not our problem.''}%
  {At \val{lograte}\,kHz adjacent records correlate at \val{rho}: about \val{effective} independent samples of a
   slowly varying statistic, all on the corridor. A faster logger never fills an empty cell.}{}
\mistake{``Held-out loss is tiny, so the policy is ready.''}%
  {A cautious archive scores \val{cautiousloss}\,rad against \val{exploratoryloss}\,rad for an exploratory one,
   because it covers \val{cautiouscells} of \val{gridcells} cells (\val{cautiouspct}\%) against \val{exploratorycells}.
   A split from the same archive shares its gaps.}{}
