\begin{bigidea}
A spoken request names an outcome, not a motion. A learned target is safe to pursue only as an
\hl{intent lease}: bounded in space, effort and time, granted by the permission path and revoked on
its own clock. Motion toward a wrong target crosses the \hl{causal boundary} and cannot be taken back.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{A label is not a target.}{Grounding yields a feature pose in the base frame, approach, aperture
  and wrench ceiling, each tied to dated sensor evidence.}
\idea{The goal set is a specification.}{The task profile sets the tolerance; the proposal's error
  envelope must fit inside it, never stretch it.}
\idea{A score is not uniqueness.}{Detectors score each box alone. Refuse unless the calibrated set
  holds exactly one candidate.}
\idea{Admission checks physics, not meaning.}{A confident wrong object can pass every coordinate,
  reach and force check.}
\idea{The lease grants a target, never a path.}{It ends on the safety controller's clock with no
  message from the host; only a newer capture renews it.}
\idea{Time prices reach.}{A target inside the workspace still fails if the move outlasts its evidence.}
\end{ideas}

\fcard{Lease expiry: earliest of three}%
  {\begin{gathered}\IntentLeaseExpiry = \min\!\left(\MonotonicClockInstant_0 + \EvidenceHorizon,\;
     \LeaseAdmissionInstant + \IntentAdministrativeLease,\; \StoppingDeadline\right)\\[2pt]
   \StoppingDeadline = \LeaseAdmissionInstant + \frac{\ClearDistance - \StoppingDistance}{\SpeedAtDetection}\\[2pt]
   \StoppingDistance = \SpeedAtDetection\PostRevocationDelay + \frac{\SpeedAtDetection^2}{2\CredibleBrakingDeceleration}\end{gathered}}%
  {Evidence decay from capture $\MonotonicClockInstant_0$, host silence from admission
   $\LeaseAdmissionInstant$, and the last instant a stop still fits clearance $\ClearDistance$.
   $\PostRevocationDelay$: expiry to braking onset.}%
  {Expiry is not rest.}{}
\fcard{Goal set and error box}%
  {\sqrt{\left(\tfrac{\XPositionErrorEnvelope}{\PositionalTaskTolerance}\right)^2
   + \left(\tfrac{\YPositionErrorEnvelope}{\PositionalTaskTolerance}\right)^2
   + \left(\tfrac{\ZPositionErrorEnvelope}{\PositionalTaskTolerance}\right)^2} \le 1}%
  {The error box's corner must lie inside the tolerance ellipsoid
   (same $\PositionalTaskTolerance$ on every axis).}%
  {Each axis passing alone is necessary, not sufficient.}{}
\fcard{Calibrated referent set}%
  {\ConformalPredictionSet = \{\, \CandidateIndex : \NormalizedCandidateScore_\CandidateIndex \ge 1 - \ConformalScoreQuantile \,\}}%
  {Normalize scores to sum to one; keep every candidate the calibrated threshold cannot
   exclude. One member: proceed; otherwise refuse.}%
  {A station depth count above one still refuses.}{}
\fcard{Reach and occupancy screens}%
  {\begin{gathered}\frac{\TargetTravelDistance}{\SpeedBound} \le \IntentLeaseExpiry - \CurrentMonotonicTime\\[2pt]
   \lVert\CartesianPosition - \EstimatedPosition_\ManifestTrackIndex\rVert \le \ActorBodyRadius_\ManifestTrackIndex
   + \HumanWalkingSpeedBound\,(\ObservationAge + \FiniteChange\MonotonicClockInstant)
   + \OccupancyCoverageMultiplier\TrackPositionDeviation\end{gathered}}%
  {Transit floor at speed ceiling $\SpeedBound$ against the window left. Person $\ManifestTrackIndex$'s
   set grows at walking bound $\HumanWalkingSpeedBound$ from the track's age $\ObservationAge$.}%
  {Necessary only. A set wider than the aisle refuses every corridor (the freezing robot); a long move needs staged, renewed segments.}{}
\end{rightcol}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{When does the untracked mug lease expire?}
The handle has $\PositionalTaskTolerance$ = \val{rtol}\,mm and $\InitialTrackingError$ = \val{ezero}\,mm;
the belt runs at \val{vbelt}\,mm/s. Administrative lease \val{tadmin}\,ms. The arm moves at its
\val{vtcp}\,m/s ceiling with clearance \val{dclear}\,mm, $\PostRevocationDelay$ = \val{trev}\,ms,
$\CredibleBrakingDeceleration$ = \val{abrake}\,m/s$^2$. All countdowns start at capture.\par\vspace{5pt}
{\centering
\begin{tikzpicture}[sheet]
  \draw[nbox] (0,0) rectangle (\val{sreact},12);
  \draw[accbox] (\val{sreact},0) rectangle ({\val{sreact}+\val{sbrake}},12);
  \node[small, anchor=south] at ({\val{sreact}/2},13) {onset};
  \node[small, anchor=south] at ({\val{sreact}+\val{sbrake}/2},13) {braking};
  \draw[dim, dashed] (\val{sclear},-4) -- (\val{sclear},24);
  \node[small, anchor=south] at (\val{sclear},24) {clearance};
  \node[note, anchor=north] at ({\val{sclear}/2},-4) {to scale; spare \val{sspare}\,mm (\val{tstop}\,ms at \val{vtcp}\,m/s)};
\end{tikzpicture}\par}\vspace{4pt}
\begin{steps}
\item Evidence: (\val{rtol} $-$ \val{ezero})\,mm $\div$ \val{vbelt}\,mm/s = \res{\val{tauu}\,ms}.
\item Stop: \val{vtcp}\,m/s $\times$ \val{trev}\,ms + (\val{vtcp}\,m/s)$^2/(2\times\val{abrake}$\,m/s$^2$) = \val{dreact}\,mm + \val{dbrake}\,mm = \res{\val{dstop}\,mm}.
\item Deadline: (\val{dclear} $-$ \val{dstop})\,mm $\div$ \val{vtcp}\,m/s = \res{\val{tstop}\,ms}.
\item Expiry: $\min(\val{tauu}, \val{tadmin}, \val{tstop})$ = \res{\val{tauu}\,ms}; evidence binds.
\end{steps}
\takeaway{The \val{infer}\,ms intent inference cannot renew a \val{tauu}\,ms target: track it, slow the belt, or refuse.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``Renewals keep arriving with new sequence numbers, so the target is fresh.''}%
  {Authority dates from capture, not arrival: a re-sent frame renews nothing. The lease still ends
   \val{tauu}\,ms after capture; a \val{stall}\,ms stall grows the error to \val{erru}\,mm against \val{rtol}\,mm.}{}
\mistake{``The detector is confident, so skip the ambiguity check.''}%
  {Mug \val{smug} and jar \val{sjar} normalize to \val{cmug} and \val{cjar}; both clear
   \val{cthr}, so the set holds two and the producer refuses.}{}
\mistake{``The tripwire is \val{ftrip}\,N, so the latch never feels more.''}%
  {Force keeps rising through the \val{tcontact}\,ms contact-loop response: \val{ftrip}\,N + \val{klatch}\,N/mm
   $\times$ \val{vlatch}\,m/s $\times$ \val{tcontact}\,ms = \val{fbrake}\,N at brake command. The arm has only \val{xrem}\,mm
   of compression left before the \val{flimit}\,N limit.}{}
