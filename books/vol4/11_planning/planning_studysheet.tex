\begin{bigidea}
A target is not a motion. Planning gives a proposal \hl{timing}, a smooth join to the motion under way,
and a \hl{latched stop that runs without the proposer}. Work across the causal boundary cannot be
recalled, so a plan is admitted only while that stop fits the clearance ahead.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Generators only propose.}{Search, optimization and learned chunks hand over candidates; the Safety
  MCU decides.}
\idea{Replacement outlasts renewal.}{Capture to admission outlasts one renewal period, so requests
  overlap; the mission's handoff count sets the tail.}
\idea{Tool limits are not joint limits.}{Near a straight elbow a modest tool speed needs huge joint
  rates; damping caps it, costing tool accuracy.}
\idea{Seams are C$^2$, planned from the projected state.}{An acceleration step times inertia is a torque
  step. A pose sampled at request lags by joint speed times delay.}
\idea{Smooth stops are longer.}{A jerk limit acts like extra reaction time.}
\idea{Two deadlines, two stops.}{A lapsed lease stops the arm from its measured state; with no successor
  by the cutoff, the latched stop starts at commitment.}
\end{ideas}

\fcard{Joint rate near a singularity}%
  {\begin{gathered}\|\GeneralizedVelocities\| = \frac{\ToolSpeedLimit}{\SmallestJacobianSingularNumber}\\[2pt]
   \text{damped: } \|\GeneralizedVelocities\| = \frac{\SmallestJacobianSingularNumber\,\ToolSpeedLimit}{\SmallestJacobianSingularNumber^2+\InverseKinematicsDamping^2}
   \le \frac{\ToolSpeedLimit}{2\InverseKinematicsDamping}\end{gathered}}%
  {$\ToolSpeedLimit$: tool speed in the weakest direction; $\SmallestJacobianSingularNumber$: smallest
   singular value of the Jacobian. With two joints, one turns at least $\|\GeneralizedVelocities\|/\sqrt{2}$.}%
  {Bounded is not feasible: the joints allow \val{qdmax}\,rad/s.}{}
\fcard{Seam torque and the C$^2$ bridge}%
  {\begin{gathered}\SeamStepTorque = \EffectiveJointInertia\,\frac{\FiniteChange\JointVelocity}{\SamplingPeriod}\\[2pt]
   \JointAcceleration_{\text{peak}} = \frac{1.5\,\FiniteChange\JointVelocity}{\SeamBridgeDuration}, \qquad
   |\JointJerk|_{\text{peak}} = \frac{6\,\FiniteChange\JointVelocity}{\SeamBridgeDuration^2}\end{gathered}}%
  {Gap $\FiniteChange\JointVelocity$ closed in one tick
   $\SamplingPeriod$ or by a symmetric quintic bridge. $\EffectiveJointInertia = \LoadInertia + \GearReductionRatio^2\RotorInertia$:
   gearing multiplies rotor inertia by $\GearReductionRatio^2$. Torque rate $\CommandedJointTorqueSlew = \EffectiveJointInertia\JointJerk \le \TorqueRateLimit$.
   Seam joint: $\EffectiveJointInertia$ = \val{jeff}\,kg$\cdot$m$^2$ (rotor term alone), rated \val{rated}\,N$\cdot$m,
   $\TorqueRateLimit$ = \val{taudmax}\,N$\cdot$m/s.}%
  {Torque rate grows as $1/\SeamBridgeDuration^2$.}{}
\fcard{Jerk-limited stop (S-curve)}%
  {\begin{gathered}\StoppingDistance = \frac{\SpeedAtDetection^2}{2\ProfileAccelerationBound} + \frac{\SpeedAtDetection\,\ProfileAccelerationBound}{2\JerkLimit}\\[2pt]
   \SpeedBound = -\ProfileAccelerationBound\EffectiveBrakingDelay + \sqrt{(\ProfileAccelerationBound\EffectiveBrakingDelay)^2 + 2\ProfileAccelerationBound\ClearDistance}\end{gathered}}%
  {Braking plus half a jerk ramp at speed; holds when
   $\SpeedAtDetection \ge \ProfileAccelerationBound^2/\JerkLimit$. $\SpeedBound$: fastest speed whose response
   travel and stop fit $\ClearDistance$; $\EffectiveBrakingDelay = \PrebrakeDelay + \ProfileAccelerationBound/2\JerkLimit$,
   $\PrebrakeDelay$ = lease + tick + bus + onset.}%
  {}{}

\end{rightcol}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{Can the arm stop if the proposer stalls?}
Arm at its \val{wv}\,m/s tool limit; no replacement arrives. Lease
\val{leasems}\,ms; tick \val{wtick}\,ms, bus \val{wbus}\,ms, brake onset \val{wonset}\,ms. Stop bounds
$\ProfileAccelerationBound$ = \val{wa}\,m/s$^2$, $\JerkLimit$ = \val{wj}\,m/s$^3$; clearance \val{wclear}\,mm.\par\vspace{5pt}
{\centering
\begin{tikzpicture}[sheet]
  \draw[nbox] (0,0) rectangle (\val{ptravel},12);
  \draw[accbox] (\val{ptravel},0) rectangle ({\val{ptravel}+\val{pstop}},12);
  \node[small, anchor=south] at ({\val{ptravel}/2},13) {response};
  \node[small, anchor=south] at ({\val{ptravel}+\val{pstop}/2},13) {S-curve stop};
  \draw[dim, dashed] (\val{pclear},-4) -- (\val{pclear},24);
  \node[small, anchor=south] at (\val{pclear},24) {obstacle at \val{wclear}\,mm};
  \node[note, anchor=north] at (93,-4) {to scale; spare \val{wmargin}\,mm};
\end{tikzpicture}\par}\vspace{4pt}
\begin{steps}
\item Response: \val{leasems} + \val{wtick} + \val{wbus} + \val{wonset} = \val{wresp}\,ms, so \res{\val{wtravel}\,mm} at \val{wv}\,m/s.
\item Branch: $\ProfileAccelerationBound^2/\JerkLimit$ = \val{wthresh}\,m/s, below \val{wv}\,m/s, so the stop reaches $\ProfileAccelerationBound$.
\item Stop: $\val{wv}^2/(2\times\val{wa})$ + $\val{wv}\times\val{wa}/(2\times\val{wj})$ = \val{wbrake} + \val{wramp} = \res{\val{wstop}\,mm}.
\item Total \res{\val{wtotal}\,mm} of \val{wclear}\,mm: admit, \val{wmargin}\,mm spare.
\end{steps}
\takeaway{At \val{wv}\,m/s, each millisecond of lease costs a millimeter of runway.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``The one-tick step asks only \val{stepnm}\,N$\cdot$m, under the \val{rated}\,N$\cdot$m rating, so the seam is fine.''}%
  {That torque must arrive within \val{tickms}\,ms: \val{steprate}\,N$\cdot$m/s, \val{stepratio}$\times$ the
   \val{taudmax}\,N$\cdot$m/s limit.}{}
\mistake{``Constant deceleration is close enough to size the stop.''}%
  {Base at \val{bv}\,m/s, \val{ba}\,m/s$^2$: braking \val{bbrake}\,mm plus \val{bother}\,mm response
   and allowances = \val{bconst}\,mm, inside \val{bclear}\,m. The jerk limit (\val{bj}\,m/s$^3$) adds \val{bramp}\,mm: \val{bover}\,mm over.}{}
\mistake{``Inference is nearly done; admit it just after the cutoff.''}%
  {Braking is due at the \val{commitms}\,ms commitment and takes \val{commitresp}\,ms to start, so admission closes at
   \val{cutoffms}\,ms. Each late millisecond costs a millimeter at \val{vtcp}\,m/s; the arm can enter an inevitable collision state.}{}
