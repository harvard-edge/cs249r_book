% Part openers use unnumbered chapters inside the current numbered chapter.
% KOMA-Script 3.49 resets dependent counters even for \chapter*. Preserve the
% existing object sequence there; a numbered chapter still resets it normally.
\makeatletter
\newcommand{\mlsysSaveChapterObjectCounters}{%
  \def\mlsys@restore@chapter@objects{}%
  \@for\mlsys@object:=table,figure,equation,algorithm,codelisting,lstlisting\do{%
    \@ifundefined{c@\mlsys@object}{}{%
      \edef\mlsys@restore@chapter@objects{%
        \unexpanded\expandafter{\mlsys@restore@chapter@objects}%
        \noexpand\setcounter{\mlsys@object}{\number\value{\mlsys@object}}%
      }%
    }%
  }%
}
\newcommand{\mlsysRestoreChapterObjectCounters}{\mlsys@restore@chapter@objects}
\makeatother
