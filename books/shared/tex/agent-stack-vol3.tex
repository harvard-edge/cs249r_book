% ─── Stack Layer Helper ──────────────────────────────────────────────────────
% #1: y-position  #2: title  #3: intensity (0-100)  #4: pic-icon  #5: subtitle
\newcommand{\agentstacklayer}[5]{%
    \ifnum#3=0
        \def\fillcol{black!5}\def\textcol{black!35}\def\subcol{black!25}%
    \else
        \def\fillcol{amethyst!#3}%
        \ifnum#3>50
            \def\textcol{white}\def\subcol{white!85}%
        \else
            \def\textcol{amethyst}\def\subcol{amethyst!80!black}%
        \fi
    \fi
    \node[
        draw=white, line width=0.5pt, fill=\fillcol,
        minimum width=2.6cm, minimum height=0.74cm,
        anchor=south west, rounded corners=2pt, inner sep=0pt
    ] (box) at (0,#1) {};
    \node[anchor=west, text=\textcol, font=\usefont{T1}{phv}{b}{n}\fontsize{5.4}{6.0}\selectfont]
        at ($(box.west)+(0.12,0.11)$) {#2};
    \node[anchor=west, text=\subcol, font=\usefont{T1}{phv}{m}{n}\fontsize{4.0}{4.6}\selectfont]
        at ($(box.west)+(0.12,-0.12)$) {#5};
    \path (box.east) +(-0.35,0) pic[scale=0.72, draw=\textcol, fill=\textcol, text=\textcol] {#4};
}
