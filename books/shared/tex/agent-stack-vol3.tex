\newcommand{\agentstacklayer}[6]{
    \ifnum#3=0
        \def\fillcol{black!5}
        \def\textcol{black!40}
        \colorlet{tabcol}{black!12}
    \else
        \def\fillcol{amethyst!#3!white}
        \ifnum#3=100
            \colorlet{tabcol}{amethyst!85!black}
        \else
            \colorlet{tabcol}{amethyst!\the\numexpr#3+15\relax!white}
        \fi
        \ifnum#3>50
            \def\textcol{white}
        \else
            \def\textcol{amethyst}
        \fi
    \fi
    
    % Main Box
    \node[
        draw=none, fill=\fillcol,
        minimum width=4.6cm, minimum height=1.1cm,
        anchor=south west, rounded corners=3pt, inner sep=0pt
    ] (box) at (0,#1) {};
    
    % Left Tab
    \begin{scope}
        \clip (box.south west) rectangle ($(box.north west)+(0.6,0)$);
        \fill[tabcol] (box.south west) rectangle ($(box.north west)+(0.8,0)$);
    \end{scope}
    
    % Tab Separation Line
    \draw[\textcol, opacity=0.25, line width=0.8pt] ($(box.south west)+(0.6,0)$) -- ($(box.north west)+(0.6,0)$);
    
    % Rotated Main Label (Classical)
    \node[rotate=90, text=\textcol, font=\sffamily\fontsize{4.6}{5.6}\selectfont\bfseries]
        at ($(box.west)+(0.3,0)$) {#2};
        
    % Agentic Inside Title
    \node[anchor=west, text=\textcol, font=\sffamily\fontsize{7.5}{8.5}\selectfont\bfseries]
        at ($(box.west)+(0.8, 0.18)$) {#5};
        
    % Agentic Inside Tech Pieces
    \node[anchor=west, text=\textcol, font=\sffamily\fontsize{5.5}{6.5}\selectfont]
        at ($(box.west)+(0.8, -0.25)$) {#6};
        
    % Icon
    \path (box.east) +(-0.45,0)
        pic[scale=0.85, draw=\textcol, fill=\textcol, text=\textcol] {#4};
}
