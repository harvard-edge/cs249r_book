% ─── Stack layer helper ───────────────────────────────────────────────────
\newcommand{\agentstacklayer}[5]{
    \ifnum#3=0
        \def\fillcol{black!5}
        \def\textcol{black!30}
    \else
        \def\fillcol{amethyst!#3!white}
        \ifnum#3>50
            \def\textcol{white}
        \else
            \def\textcol{amethyst}
        \fi
    \fi
    \node[
        draw=white, line width=0.5pt, fill=\fillcol,
        minimum width=3.8cm, minimum height=0.9cm,
        anchor=south west, rounded corners=3pt, inner sep=0pt
    ] (box) at (0,#1+0.05) {};
    
    \node[anchor=west, text=\textcol, font=\sffamily\scriptsize\bfseries]
        at ($(box.west)+(0.12,0.10)$) {#2};
    \node[anchor=west, text=\textcol, font=\sffamily\fontsize{5}{6}\selectfont]
        at ($(box.west)+(0.12,-0.23)$) {#5};
        
    \path (box.east) +(-0.55,0)
        pic[scale=0.85, draw=\textcol, fill=\textcol, text=\textcol] {#4};
}

% ─── mlagentstack: smooth gradient trajectory bar + layer-colored wires ──────────
\newcommand{\mlagentstack}[6]{
    \begin{tikzpicture}
        \agentstacklayer{5.0}{Distributed Fleet}{#1}{icon_fleet}{coordination, tokenomics}
        \agentstacklayer{4.0}{Policy Compiler}{#2}{icon_policy}{SFT, RLVR, optimization}
        \agentstacklayer{3.0}{Kernel \& Control}{#3}{icon_kernel}{interrupts, scheduling}
        \agentstacklayer{2.0}{I/O Peripherals}{#4}{icon_io}{actuation, sandboxing}
        \agentstacklayer{1.0}{Memory \& Storage}{#5}{icon_memory}{context, KV paging, disk}
        \agentstacklayer{0.0}{Stoch. Processor}{#6}{icon_processor}{foundation model}

        % Trajectory vertical spine gradient
        \pgfmathtruncatemacro{\dlone}{#6}
        \pgfmathtruncatemacro{\dltwo}{#5}
        \pgfmathtruncatemacro{\dlthree}{#4}
        \pgfmathtruncatemacro{\dlfour}{#3}
        \pgfmathtruncatemacro{\dlfive}{#2}
        \pgfmathtruncatemacro{\dlsix}{#1}

        \colorlet{stop@l1}{amethyst!\dlone!white}
        \colorlet{stop@l2}{amethyst!\dltwo!white}
        \colorlet{stop@l3}{amethyst!\dlthree!white}
        \colorlet{stop@l4}{amethyst!\dlfour!white}
        \colorlet{stop@l5}{amethyst!\dlfive!white}
        \colorlet{stop@l6}{amethyst!\dlsix!white}

        \pgfdeclareverticalshading{trajbarshading}{100bp}{%
            color(0bp)=(stop@l1);
            color(10bp)=(stop@l1);
            color(26bp)=(stop@l2);
            color(42bp)=(stop@l3);
            color(58bp)=(stop@l4);
            color(74bp)=(stop@l5);
            color(90bp)=(stop@l6);
            color(100bp)=(stop@l6)
        }

        \shade[shading=trajbarshading, rounded corners=1.5pt]
            (4.05, -0.1) rectangle (4.35, 6.10);

        % Wires
        \foreach \layerint/\ypos in {#6/0.5, #5/1.5, #4/2.5, #3/3.5, #2/4.5, #1/5.5} {
            \pgfmathtruncatemacro{\lint}{\layerint}
            \ifnum\lint=0
                \def\wirecol{black!8}
            \else
                \edef\wirecol{amethyst!\lint}
            \fi
            \draw[\wirecol, line width=1.5pt, rounded corners=3pt] (3.8, \ypos) -- (3.95, \ypos) -- (4.05, \ypos);
            \fill[\wirecol] (4.05, \ypos) circle (0.06);
        }

        \pgfmathtruncatemacro{\maxint}{max(max(max(#1,#2),max(#3,#4)),max(#5,#6))}
        \ifnum\maxint>50
            \def\datatextcol{white}
        \else
            \ifnum\maxint<10
                \def\datatextcol{black!25}
            \else
                \def\datatextcol{amethyst}
            \fi
        \fi
        
        \node[rotate=90, text=\datatextcol, font=\sffamily\scriptsize\bfseries]
            at (4.20, 3.0) {Trajectory};
    \end{tikzpicture}
}
