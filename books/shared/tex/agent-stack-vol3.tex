\newcommand{\agentstacklayer}[7]{%
  \pgfmathtruncatemacro{\lint}{#3}%
  \ifnum\lint=0
    \def\bgcol{white}\def\tabcol{black!4}%
    \def\tcol{black!85}\def\subcol{black!52}\def\bdrcol{black!18}\def\rulecol{black!14}%
    \def\wcol{black!20}\def\icocol{black!60}%
  \else\ifnum\lint>44
    \pgfmathtruncatemacro{\cint}{min(100, 75 + \lint/4)}%
    \edef\bgcol{amethyst!\cint!black}\edef\tabcol{amethyst!\cint!black!82!white}%
    \def\tcol{white}\def\subcol{white!88}\edef\bdrcol{amethyst!\cint!black}\def\rulecol{white!35}%
    \edef\wcol{amethyst!\cint}\def\icocol{white}%
  \else
    \pgfmathtruncatemacro{\bgtint}{12 + \lint}%
    \pgfmathtruncatemacro{\tabtint}{22 + \lint*2}%
    \edef\bgcol{amethyst!\bgtint!white!20!white}\edef\tabcol{amethyst!\tabtint!white!45!white}%
    \def\tcol{amethyst!95!black}\def\subcol{amethyst!82!black}\edef\bdrcol{amethyst!\tabtint!black!30!white}\def\rulecol{amethyst!\tabtint!white!65!white}%
    \edef\wcol{amethyst!\tabtint}\def\icocol{amethyst!90!black}%
  \fi\fi
  \begin{scope}[shift={(0,#1)}]
    \draw[\bdrcol, line width=0.7pt, fill=\bgcol, rounded corners=3.5pt] (0,0) rectangle (5.25, 1.72);
    \clip[rounded corners=3.5pt] (0,0) rectangle (5.25, 1.72);
