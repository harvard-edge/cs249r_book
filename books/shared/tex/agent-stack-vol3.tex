\newcommand{\agentstacklayer}[5]{
    \ifnum#3=0
        \def\fillcol{black!5}
        \def\textcol{black!40}
        \colorlet{innerbg}{white}
    \else
        \def\fillcol{amethyst!#3!white}
        \ifnum#3>50
            \def\textcol{white}
            \colorlet{innerbg}{black!15!\fillcol}
        \else
            \def\textcol{amethyst}
            \colorlet{innerbg}{white}
        \fi
    \fi
    
    % Main Box
    \node[
        draw=none, fill=\fillcol,
        minimum width=4.6cm, minimum height=1.4cm,
        anchor=south west, rounded corners=3pt, inner sep=0pt
    ] (box) at (0,#1) {};
    
    % Tab Separation Line
    \draw[\textcol, opacity=0.3, line width=0.8pt] ($(box.south west)+(0.55,0)$) -- ($(box.north west)+(0.55,0)$);
    
    % Rotated Main Label
    \node[rotate=90, text=\textcol, font=\sffamily\scriptsize\bfseries]
        at ($(box.west)+(0.27,0)$) {#2};
        
    % Icon
    \path (box.east) +(-0.45,0)
        pic[scale=0.85, draw=\textcol, fill=\textcol, text=\textcol] {#4};
        
    % Sub-components (Innards)
    \begin{scope}[shift={(box.south west)}]
        #5
    \end{scope}
}
