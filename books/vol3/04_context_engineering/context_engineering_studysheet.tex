\frontside
\begin{bigidea}
A model carries no state between calls: the runtime rebuilds every request, and fitting is not the goal.
Each call needs a \hl{selected working set}: the evidence the next decision needs, bound to the revision it
describes, laid out for prefix reuse, and withdrawn when the workspace changes.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Admitted is not usable.}{The window sets what fits; what the model can use is measured per model and
  workload. Each staged key competes with the evidence for attention.}
\idea{Select for the next decision.}{Relevance, not recency. Bulky payloads stay outside behind a handle,
  re-read by a checked tool call.}
\idea{Order by change rate.}{Root, trunk, leaf. Prefix reuse ends at the first changed token, but a
  stale trunk is never kept to save a cache hit.}
\idea{Every observation keeps its scope.}{It supports claims only for the revision and run conditions
  that produced it. Labels guide the model; host gates enforce.}
\idea{Compact from raw records, then audit.}{Protected fields must survive. A model-written record is a
  candidate, checked against the task contract and the event log.}
\idea{Invalidate, do not delete.}{A write marks bound observations historical and flags the conclusions
  built on them. A missing edge never proves freshness.}
\end{ideas}

\begin{sheettable}{@{}l L L@{}}

\begin{answers}

  results, overflow at turn \val{yoverflow}; digests $\le$ \val{ybudget} $\div$ \val{turns} = \val{ydigest} tokens.
  Change the per-turn observation (digest plus handle), not the window (input budget).}
\ans{1}{(a) Nothing reused: \val{total} uncached. (b) Cold call, rewrites prefill trunk and leaf, the rest the leaf:
  \val{request} + \val{rewrites} $\times$ \val{suffix} + \val{warm} $\times$ \val{leaf} = \val{stable}; (a) is
  \val{unstableratio}$\times$ (b) (root, trunk, leaf table).}

\columnbreak

  then read \texttt{r18}. Keep the failure as \texttt{r17} history; flag the hypothesis for reassessment (three choices).}

  by pass to where it vanished. Present: the model did not comply. An executed deploy means a host gate was missing.}

\begin{steps}
\item Input limit \val{window} $-$ \val{reserve} = \val{limit}; less \val{fixed} fixed leaves
  \res{\val{budget}} for observations.
\item \val{budget} $\div$ \val{perturn} = \res{\val{fit} results}: the request overflows at turn \val{overflow}.
  To last \val{turns} turns, digest each to \val{budget} $\div$ \val{turns} = \res{\val{digest}}.
\item Even with a \val{finalneed}-token window, no reuse: append \val{turns} $\times$ \val{first} + \val{perturn}
  $\times$ \val{pairs} = \res{\val{appendbill}}; select \val{turns} $\times$ \val{requestsize} = \val{selectbill}.
  Appending sends \res{\val{billratio}$\times$} the input.
\end{steps}
\takeaway{Per-turn observation size is the lever.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``If every token fits, every fact is usable.''}%
  {Fixed logits (target \val{ztarget}, others \val{zback}): the target holds \val{shortweight}\% of the weight
   among \val{shortkeys} keys, \val{longweight}\% among \val{longkeys}, though all fit.}{}
\mistake{``The record passes the schema, so the constraints survived.''}%
  {Schemas check presence. At \val{rho}\% loss per pass a constraint survives \val{roundslong} passes with
   \val{survivelong}\%, every record still valid.}{}
