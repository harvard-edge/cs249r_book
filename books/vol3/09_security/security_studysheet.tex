\begin{bigidea}
An agent inside its sandbox, calling only permitted tools, can still work for an attacker: text it
\hl{reads} steers what it proposes, executed with the task's \hl{authority}. So the runtime tracks where
every value came from and where it may go, and checks each proposal at \hl{dispatch}.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Injection is structural.}{Instructions, schemas, request and observations share one token
  sequence; no position is privileged, so delimiters are only advice.}
\idea{Callable is not permitted.}{Injected, the runtime is a confused deputy with a legitimate identity.
  Judge this request's data and recipient, not the tool name.}
\idea{Two labels, two checks.}{Integrity provenance (what influenced a value) and confidentiality (who
  may receive it) vary, and fail their checks, independently.}
\idea{Well-formed is not true.}{A quarantined reader keeps raw text from the planner, yet its
  schema-valid field is still the document's claim; only the runner's record is evidence.}
\idea{Approval binds an action.}{Effective permission = requester $\cap$ service principal $\cap$ task grant
  $\cap$ policy. Approval signs a digest of the exact action, recomputed at dispatch; tool replies never approve.}
\idea{Every turn, every surface.}{Summaries and memory inherit labels; every dispatch is rechecked. A
  shared prefix cache leaks by timing: \val{probehit} ms hit, \val{probemiss} ms miss.}
\end{ideas}

\begin{practicebox}\boxtitle{Does the release check pay?}
Policy A checks tool name and destination; B adds a release check. Each runs
\val{attacked} attacked and \val{benign} benign repairs at \$\val{attempt} per attempt. Confirmed
disclosures, unresolved, benign accepted: A \val{confa}, \val{unresa}, \val{benigna}; B \val{confb}, \val{unresb}, \val{benignb}.\par\vspace{5pt}

  \node[note, anchor=south] at ({44+(\val{lowa}+\val{upa})*\sc/2},29) {+ unresolved};

\begin{steps}
\item Observed: \val{confa} $\div$ \val{attacked} = \val{lowa}\%, \val{confb} $\div$ \val{attacked} = \val{lowb}\%;
  reduction $1 - \val{lowb} \div \val{lowa}$ = \res{\val{redobserved}\%}.
\item Unresolved as disclosures: \val{upa}\% and \val{upb}\%: $1 - \val{upb} \div \val{upa}$ = \res{\val{redupper}\%};
  least favorable $1 - \val{upb} \div \val{lowa}$ = \res{\val{redleast}\%}.
\item Price: \val{attempt} $\div$ \val{acca} = \$\val{ceffa} vs.\ \val{attempt} $\div$ \val{accb} = \$\val{ceffb};
  ratio \val{acca} $\div$ \val{accb} = \val{costratio}: \res{\val{costrise}\% more} per accepted repair.
\end{steps}
\takeaway{Report a defense as a range and a price.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``A guardrail classifier in front of the agent stops injection.''}%
  {Attackers optimize against it. Missing only \val{missrate}\% of payloads, it still leaves
   \val{screenedresearch}\% of research tasks (\val{readsresearch} reads at \val{payload}\%) exposed.}{}
\mistake{``No breaches in \val{zeron} attacked episodes: the defense works.''}%
  {Zero observed bounds the rate; the \val{conf}\% bound is still
   \val{zeroub}\% ($\approx 3 \div \val{zeron}$).}{}
\mistake{``We deleted the poisoned README, so the agent is clean.''}%
  {Summaries, lessons and queued proposals derived from it remain: retract them all.}{}
