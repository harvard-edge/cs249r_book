\begin{bigidea}
A completed run is one draw from a stochastic process, judged by checks that see only what they
cover. Evaluation turns trials into a claim about a \hl{configuration}: credit only on \hl{evidence},
sample \hl{tasks}, compare on \hl{paired} tasks, and charge every failed attempt to the accepted ones.
Unresolved trials stay in the denominator, unaccepted, reported apart from failures.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Acceptance is a contract.}{Accept when every mandatory criterion passes on the artifact;
  a mandatory failure fails. Pending: underway; unknown: missing at cutoff.}
\idea{A check covers only what it exercises.}{Sealed tests sit beyond the agent's reach, bound to the
  artifact. Retire a held-out suite once it steers changes.}
\idea{Calibrate graders on independent reference.}{False acceptance $\Pr(\text{Acc}\mid\ReferenceAssessedFailure)$
  needs reference failures; an accepted-only audit gives $\Pr(\ReferenceAssessedFailure\mid\text{Acc})$.}
\idea{Sample tasks, not runs.}{New tasks shrink both variance terms; repeats shrink only the run term.}
\idea{Compare on paired tasks.}{Run both configurations on each task; discordant pairs carry the gain.
  Gate the interval's lower end on a tolerance declared in advance.}
\idea{The policy picks the metric.}{A checker before delivery: pass@$\IndependentAttemptCount$. None,
  as for an answer a customer reads: pass$^{\IndependentAttemptCount}$.}
\end{ideas}

\vspace{8pt}
\fcard{Paired comparison and the release gate}%
  {\EstimatedPairedAcceptanceDifference = \frac{\CandidateOnlyAcceptedTaskCount-\BaselineOnlyAcceptedTaskCount}{\EvaluationSampleCount},
   \quad \PairedDifferenceLowerLimit,\PairedDifferenceUpperLimit = \EstimatedPairedAcceptanceDifference \mp \ConfidenceCriticalQuantile
   \sqrt{\frac{\DiscordantPairFraction-\EstimatedPairedAcceptanceDifference^{2}}{\EvaluationSampleCount}}}%
  {$\EvaluationSampleCount$ paired tasks; $\BaselineOnlyAcceptedTaskCount$, $\CandidateOnlyAcceptedTaskCount$: tasks only the
   baseline, only the candidate accepts; $\DiscordantPairFraction = (\BaselineOnlyAcceptedTaskCount+\CandidateOnlyAcceptedTaskCount)/\EvaluationSampleCount$.
   Noninferior if $\PairedDifferenceLowerLimit > -\NoninferiorityTolerance$ (declared tolerance); superior if $\PairedDifferenceLowerLimit > 0$.}%
  {$\ConfidenceCriticalQuantile = \val{zcrit}$; width shrinks as $1/\sqrt{\EvaluationSampleCount}$.}{}
\end{leftcol}%
\begin{rightcol}
\fcard{Tasks, repeats, and a clean record}%
  {\operatorname{Var}(\EstimatedAcceptanceProbability) = \frac{\BetweenTaskStandardDeviation^{2}}{\EvaluationSampleCount}
   + \frac{\WithinTaskRunStandardDeviation^{2}}{\EvaluationSampleCount\PerTaskRunCount},
   \qquad \AcceptanceIntervalUpperLimit = \frac{\ConfidenceCriticalQuantile^{2}}{\EvaluationSampleCount+\ConfidenceCriticalQuantile^{2}}
   \ \text{at}\ \AcceptedSampleCount = 0}%
  {$\EvaluationSampleCount$ tasks, $\PerTaskRunCount$ runs each: repeats shrink only the run term. $\AcceptanceIntervalUpperLimit$:
   Wilson upper limit after $\AcceptedSampleCount = 0$ events in $\EvaluationSampleCount$ independent units;
   $\ConfidenceCriticalQuantile^{2} = \val{zsq}$.}%
  {Wald collapses to zero width at $\AcceptedSampleCount = 0$. Once $\PerTaskRunCount$ exceeds
   $\WithinTaskRunStandardDeviation^{2}/\BetweenTaskStandardDeviation^{2}$, the task term dominates.}{}
\fcard{Availability against consistency}%

\fcard{Cost per accepted task, two ways to select}%
  {\SequentialCostPerAcceptedTask = \frac{\CandidateGenerationCost+\CandidateCheckingCost}{\AttemptPassProbability},
   \qquad \BatchCostPerAcceptedTask = \frac{\IndependentAttemptCount(\CandidateGenerationCost+\CandidateCheckingCost)}{\CandidateAvailabilityProbability}}%
  {$\CandidateGenerationCost$, $\CandidateCheckingCost$: dollars to generate, to check one candidate (exact checker);

\begin{practicebox}\boxtitle{Does candidate\_v2 clear the release gate?}
On \val{pairs} sampled repair tasks, both accept \val{both}, neither \val{neither}; only the baseline accepts
$\BaselineOnlyAcceptedTaskCount = \val{baselineonly}$, only the candidate $\CandidateOnlyAcceptedTaskCount = \val{candidateonly}$.
Declared tolerance: \val{tolpt} points.\par\vspace{5pt}

\begin{steps}
\item $\EstimatedPairedAcceptanceDifference = (\val{candidateonly}-\val{baselineonly})/\val{pairs} = \val{dhat}$;
  $\DiscordantPairFraction = \val{disc}/\val{pairs} = \val{psi}$.
\item Var $= (\val{psi} - \val{dhatsq})/\val{pairs} = \val{var}$; SE $=$ \val{se}.
\item Half-width $\val{zcrit} \times \val{se} = \val{hw}$: normal-approximation interval \res{\val{lopt} to \val{hipt} points}.
\item Gate: \val{lopt} $< -\val{tolpt}$, so \res{hold}; zero is inside, so not superior either.
\end{steps}
\takeaway{More acceptances is not a release claim; the planned number of pairs is the knob.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``It passes \val{prun}\% of runs, so it is reliable.''}%
  {A task recurring \val{krec} times passes every time in only \val{powk}\% of cases. Holding \val{target}\% needs
   \val{pneed}\% per run, \val{failcut}$\times$ fewer failures.}{}
\mistake{``Compare each configuration's own interval; overlap means no difference.''}%
  {Separate intervals throw away the pairing. In the chapter's comparison, pairing keeps \val{pairedshare}\% of the
   variance; unpaired arms need about \val{indepmult}$\times$ the tasks.}{}
\mistake{``We tuned against the held-out suite, but it is still a fair test.''}%
  {Once a held-out result steers a change, the suite is no longer independent; repeated tuning
   makes its score measure fit, and no sandbox reset detects it. Retire it into development;
   decide on a fresh sample.}{}
