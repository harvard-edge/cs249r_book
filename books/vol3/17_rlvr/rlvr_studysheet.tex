\frontside
\begin{bigidea}
A programmatic check scores each sampled attempt, and the score, compared within a group, weights the
update. The check is now the \hl{objective}: the optimizer searches every gap between what it measures and what the task needs.
The policy improves only as far as the check's \hl{scope and protection} allow.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Learning needs contrast.}{Only rollouts the reward tells apart move the policy; a never-solved task
  gives zero signal.}
\idea{A reward is a measurement.}{A weakened assertion or a patched test runner earns full reward and
  is often likelier than a real fix.}
\idea{Isolate the verifier.}{Grade a source-only diff in a fresh workspace whose tests and verdict the
  candidate cannot reach. An instruction not to hack protects nothing.}
\idea{A terminal reward credits every turn alike.}{Separating good and bad steps costs a critic,
  a gameable step score, or extra forked rollouts.}
\idea{Record what was sampled.}{Per call: delivered prefix, sampled token IDs, behavior log
  probabilities. A preempted rollout is unresolved, not zero.}
\idea{Judge by outcomes.}{Rising reward or longer self-checking proves nothing. Lost action
  diversity on unsolved tasks shows up as tied groups.}
\end{ideas}

\vspace{8pt}
\fcard{Reward ordering}%
  {\TrajectoryReward = \CorrectnessReward + \FormatRewardWeight\FormatComplianceScore - \LengthPenaltyWeight\,\text{len},
   \qquad \FormatRewardWeight + \LengthPenaltyWeight < 1}%
  {Pass $\CorrectnessReward=1$, fail $0$; format $\FormatComplianceScore$ and len in $[0,1]$. A pass earns
   at least $1-\LengthPenaltyWeight$, a failure at most $\FormatRewardWeight$, so passes outrank failures.}%
  {Needs a length cap the runtime enforces.}{}
\fcard{False-accept share of the reward}%
  {\Pr(\ReferenceAssessedFailure \mid \VerifierVerdict^{+}) =
   \frac{(1-\ValidCandidateRate)\,\VerifierFalseAcceptanceRate}{\ValidCandidateRate\,\VerifierTrueAcceptanceRate + (1-\ValidCandidateRate)\,\VerifierFalseAcceptanceRate}}%
  {Share of rewards going to true failures. True pass rate $\ValidCandidateRate$; the verifier accepts a
   failure with probability $\VerifierFalseAcceptanceRate$, a success with $\VerifierTrueAcceptanceRate$.}%
  {}{}
\end{leftcol}%
\begin{rightcol}
\fcard{Group-relative coefficient (GRPO)}%
  {\GroupRelativeAdvantage_i = \frac{\TrajectoryReward_i - \GroupMeanReward}{\GroupRewardStandardDeviation + \AdvantageStabilizer}}%
  {$\GroupSize$ rollouts of one task, no critic: $\GroupMeanReward$ is their mean reward,
   $\GroupRewardStandardDeviation$ their standard deviation (divide by $\GroupSize$), $\AdvantageStabilizer$ a
   negligible stabilizer. Rewards
   $[0,1,0]$ give $\val{grponeg}, \val{grpopos}, \val{grponeg}$; a lone pass gets $\sqrt{\GroupSize-1}$, sound or tampered.}%
  {Initially each of the $\RolloutPolicyTokenCount_i$ tokens of rollout $i$ carries
   $\GroupRelativeAdvantage_i/\RolloutPolicyTokenCount_i$: a long failure is pushed down less per token.
   A divisor shared by the group removes this length bias.}{}
\fcard{Reward contrast}%
  {\GroupContrastProbability = 1 - \AttemptPassProbability^{\GroupSize} - (1-\AttemptPassProbability)^{\GroupSize}}%
  {Chance a group holds a pass and a failure; independent pass/fail runs at rate $\AttemptPassProbability$. A tied group
   has every $\GroupRelativeAdvantage_i=0$: full cost, no reward gradient. Rollouts per informative group:
   $\GroupSize/\GroupContrastProbability$.}%
  {}%
  {\begin{tikzpicture}[sheet]
     \foreach \yv/\yl in {0/0, 50/0.5, 100/1}{\draw[gridline] (30,{\yv*0.5}) -- (230,{\yv*0.5});
       \node[small, anchor=east] at (27,{\yv*0.5}) {\yl};}

     \filldraw[fill=accent, draw=accent] ({30+200*\val{ptargetfrac}},{50*\val{contrasttarget}}) circle (2pt);
     \node[lab, anchor=west] at ({36+200*\val{phardfrac}},{50*\val{contrasthard}-2}) {\val{phard}\% pass: \val{contrasthardshort}};
     \node[lab, anchor=north] at ({30+200*\val{ptargetfrac}},{50*\val{contrasttarget}-4}) {\val{ptarget}\%: \val{contrasttargetshort}};
     \node[note, anchor=north] at (130,-12) {$\GroupContrastProbability$ by pass rate, groups of \val{group}};
   \end{tikzpicture}}
\fcard{Rollout slot cycle}%
  {\EnvironmentCycleTime = \TurnHorizon(\ModelTime + \ToolTime) + \RolloutAssessmentTime + \EnvironmentResetTime}%
  {A serial slot: $\TurnHorizon$ turns, then assessment and reset. Capacity is slots $\div$ $\EnvironmentCycleTime$.}%
  {Amdahl: speeding one stage saves at most its share of the cycle.}{}
\end{rightcol}

\begin{answers}
\ans{Finish}{$\val{missmodfrac}^{\val{group}} = \val{allfailmod}$, so $\GroupContrastProbability = \val{contrastmod}$ and
  $\val{group} \div \val{contrastmod}$ = \val{permod} rollouts. Groups of \val{groupbig}: $\val{missmodfrac}^{\val{groupbig}} = \val{allfailmodbig}$,
  $\val{groupbig} \div \val{contrastmodbig}$ = \val{permodbig}. Doubling still costs more; contrast is already high, so
  retargeting to \val{ptarget}\% saves only \val{modvstarget}$\times$ (contrast card).}
\ans{Q1}{Mean \val{qmean}; spread $\sqrt{\val{qvar}} = \val{qstd}$. Pass $\val{qcenteredpass} \div \val{qstd} = \val{qpass}$
  ($\sqrt{\val{qroot}}$), each failure $\val{qcenteredfail} \div \val{qstd} = \val{qfail}$; sum zero. Tampered: still \val{qpass}, the
  largest push; the coefficient sees only reward. Isolate the verifier (GRPO card).}
\ans{Q2}{A pass with no format credit at full length scores \val{lowpass}, below the best failure \val{wformat}: a tidy failure can outrank a pass.
  The ordering rule needs $\LengthPenaltyWeight < \val{wlengthmax}$ (reward ordering card).}
\columnbreak
\ans{Q3}{Exploit or false accepts: audit rewarded artifacts for edits to tests, runner or conftest, and
  regrade them in an isolated fresh workspace. Length bias: compare lengths by outcome; if failures grow,
  retrain fixed tasks with a shared divisor and compare (key ideas, GRPO card).}
\ans{Q4}{Work $\val{supturns}\times(\val{supmodel}+\val{suptool}) + \val{supassess} = \val{supwork}$ s. Cycles \val{supcoldcycle} s and \val{supsnapcycle} s:
  \val{supgain}$\times$ capacity: reset was \val{supshare}\% of the cycle (repair: \val{repairshare}\%). A short horizon
  keeps $\TurnHorizon(\ModelTime+\ToolTime)$ small (cycle card).}
\end{answers}

\begin{steps}
\item All fail: $\val{missfrac}^{\val{group}} = \val{allfail}$ (all pass is negligible), so $\GroupContrastProbability = \val{contrasthard}$;
  $\val{group} \div \val{contrasthard}$ = \res{\val{perhard}} rollouts.
\item Double: $\val{missfrac}^{\val{groupbig}} = \val{allfailbig}$, $\GroupContrastProbability = \val{contrastbig}$;
  $\val{groupbig} \div \val{contrastbig}$ = \res{\val{perbig}} rollouts.
\item Target: ties $2\times\val{ptargetfrac}^{\val{group}} = \val{tietarget}$, $\GroupContrastProbability = \val{contrasttarget}$;
  $\val{group} \div \val{contrasttarget}$ = \res{\val{pertarget}} rollouts.
\item Targeting wins: \val{perhard} $\div$ \val{pertarget} = \res{\val{saving}$\times$} fewer rollouts.
\end{steps}
\takeaway{Difficulty moves contrast more than group size; log the changed task mix.}
\end{practicebox}

\begin{rightcol}
\sheetsec{Common mistakes}
\mistake{``The verifier wrongly accepts only \val{faeps}\% of failures, so the rewards are sound.''}%
  {It accepts every pass. At a \val{fapass}\% pass rate: $(1-\val{fapassfrac})\times\val{faepsfrac}$ = \val{fanum} false against \val{fapassfrac} true,
   so \val{fashare}\% of rewards go to failures; at \val{falowpass}\% pass, \val{falowshare}\%.}{}
\mistake{``A faster environment reset speeds training in proportion.''}%
  {Reset from \val{slotcold} s to \val{slotsnap} ms is \val{resetratio}$\times$ faster, but
   \val{slotturns} turns of \val{slotmodel} + \val{slottool} ms plus \val{slotassess} s of assessment stay:
   \val{slotcoldcycle} s $\div$ \val{slotsnapcycle} s = \val{capacityratio}$\times$ capacity.}{}
\mistake{``Longer self-correction text shows the policy reasons better.''}%
  {Per-rollout averaging pushes a \val{shorttokens}-token failure down \val{lengthratio}$\times$ harder per token
   than a \val{longtokens}-token one; failures can lengthen with no extra checking.}{}
