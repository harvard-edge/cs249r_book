\begin{bigidea}
In production every exposure arrives at once: the \hl{state} a failure leaves, the \hl{horizon} before a
monitor decides, the \hl{authority} a rollback cannot recall. Decide in advance what to \hl{record}, how to
\hl{test} an explanation, how much live work to \hl{expose}, and what a release must show.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Three records, three guarantees.}{A trace may sample or drop; the recovery log must persist;
  the accounting record lists every admitted trial, unsampled.}
\idea{A span's status is not a verdict.}{A test can exit cleanly with failed assertions;
  acceptance comes from independent checks.}
\idea{Elapsed time follows the critical path.}{Overlaps are not summed; shortening a
  branch helps only until the next-longest takes over.}
\idea{Retention waits for the verdict.}{Keep failed and pending trials plus a random control
  sample. Redact at capture; masked bytes no longer replay exactly.}
\idea{Test an explanation by intervening.}{Harness replay checks code on captured outputs, no model
  call. Only branches from a saved prefix, all else fixed, test a cause, and only for that fixture.}
\idea{Release in stages; a rollback settles nothing.}{Offline gate, shadow, canary. A hold stops
  new admission; in-flight operations keep their owners until recovery settles them.}
\end{ideas}

\fcard{Head sampling misses rare failures}%
  {\Pr(\text{no trace kept}) = (1-\HeadSamplingProbability)^{\FailureModeOccurrenceCount}}%
  {Each trial kept with probability $\HeadSamplingProbability$ before its outcome is known.}%
  {Tail selection keeps every failed and pending trial plus a control fraction of ordinary ones;
   rates come from the accounting record.}{}
\fcard{Sizing a canary}%
  {\begin{gathered}
     \SprtUpperLogBoundary = \ln\frac{1-\HypothesisTestFalseNegativeRate}{\StatisticalSignificanceLevel},\qquad
     \SprtLowerLogBoundary = \ln\frac{\HypothesisTestFalseNegativeRate}{1-\StatisticalSignificanceLevel}\\[3pt]
     \SprtExpectedLogIncrement_1 = \HypothesizedAcceptanceProbability_1\ln\frac{\HypothesizedAcceptanceProbability_1}{\HypothesizedAcceptanceProbability_0}
       + (1-\HypothesizedAcceptanceProbability_1)\ln\frac{1-\HypothesizedAcceptanceProbability_1}{1-\HypothesizedAcceptanceProbability_0}\\[3pt]
     \mathbb{E}_1[\SprtStoppingSampleCount] \approx \frac{(1-\HypothesisTestFalseNegativeRate)\SprtUpperLogBoundary
       + \HypothesisTestFalseNegativeRate\SprtLowerLogBoundary}{\SprtExpectedLogIncrement_1}\\[3pt]
     \text{time} \approx \frac{\mathbb{E}_1[\SprtStoppingSampleCount]}{\CanaryAllocationFraction\ArrivalRate}
       + \CanaryLabelDelay\\[3pt]
     \text{excess failures} \approx \bigl(\mathbb{E}_1[\SprtStoppingSampleCount]
       + \CanaryAllocationFraction\ArrivalRate\CanaryLabelDelay\bigr)
       (\HypothesizedAcceptanceProbability_0-\HypothesizedAcceptanceProbability_1)
   \end{gathered}}%
  {Acceptance $\HypothesizedAcceptanceProbability_0$ against degradation
   $\HypothesizedAcceptanceProbability_1$, error targets $\StatisticalSignificanceLevel$,
   $\HypothesisTestFalseNegativeRate$; $\SprtExpectedLogIncrement_1$: expected step per label. Share
   $\CanaryAllocationFraction$ of arrivals $\ArrivalRate$; labels lag $\CanaryLabelDelay$. At
   $\SprtUpperLogBoundary$: hold; at $\SprtLowerLogBoundary$: reject this degradation; between: keep sampling.}%
  {$\CanaryAllocationFraction$ sets calendar time and unmeasured harm, not the count. Tripwires stop
   admission at once.}{}
\fcard{The release gate is a conjunction}%
  {\begin{gathered}
     \ShipDecision = \IndicatorFunction[\PairedDifferenceLowerLimit > -\NoninferiorityTolerance] \wedge
     \IndicatorFunction[\CandidateTrajectoryGoodput \ge (1-\GoodputRelativeTolerance)\BaselineTrajectoryGoodput]\\
     \wedge\ \IndicatorFunction[\CandidateTailCompletionTime \le \CompletionDeadline] \wedge
     \IndicatorFunction[\CandidateViolationCount = 0]
   \end{gathered}}%
  {Paired lower limit, accepted tasks per hour, p99 completion, verified violations: all declared
   before the data.}%
  {Nothing offsets a failed term; nothing outside the four is checked. Tool-call drift routes to review.}{}
\end{rightcol}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{When does the canary decide, and at what cost?}
$\HypothesizedAcceptanceProbability_0 = \val{pzero}$ against $\HypothesizedAcceptanceProbability_1 = \val{pone}$,
$\StatisticalSignificanceLevel = \val{alpha}$, $\HypothesisTestFalseNegativeRate = \val{beta}$; \val{arrivals}
tasks/h, $\CanaryAllocationFraction = \val{rho}$\%, labels \val{delay} h after admission.\par\vspace{2pt}
\begin{steps}
\item $\SprtUpperLogBoundary = \ln(\val{betac}/\val{alpha}) = \val{upper}$;
  $\SprtLowerLogBoundary = \ln(\val{beta}/\val{alphac}) = \val{lower}$.
\item $\SprtExpectedLogIncrement_1 = \val{drift}$: $\mathbb{E}_1[\SprtStoppingSampleCount] \approx
  (\val{betac} \times \val{upper} - \val{beta} \times \val{lowerabs}) / \val{drift}$ = \res{\val{en}} labeled tasks.
\item $\CanaryAllocationFraction\ArrivalRate = \val{rate}$/h: \val{en} $\div$ \val{rate} $\approx$ \val{admit} h, plus
  \val{delay} h $=$ \res{\val{tdec} h}.
\item Pending \val{rate} $\times$ \val{delay} = \val{pend}: (\val{en} + \val{pend}) $\times$ \val{gapacc}
  $\approx$ \res{\val{excess} excess failures}; \val{excessfull} at full allocation (\val{pendfull} pending).
\end{steps}
\takeaway{The sample count is fixed; a small canary waits longer and trims only the tasks awaiting labels.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``The tool span returned cleanly, so the step succeeded.''}%
  {Span status records how one operation ended. Acceptance comes from the accounting record.}{}
\mistake{``A \val{theta}\% head sample keeps any failure that matters.''}%
  {A mode seen \val{rarecount} times leaves no trace with $\val{thetafrac}^{\val{rarecount}} \approx \val{missrare}$\%
   probability; only at \val{commoncount} does the miss fall to \val{misscommon}\%.}{}

\mistake{``The fix passed every rerun and we rolled back: incident closed.''}%
  {\val{cleanruns} of \val{cleanruns} clean runs from one prefix still allow a failure rate up to \val{wilsonup}\% (\val{conf}\% Wilson bound). Rollback
   recalls no request already sent.}{}
