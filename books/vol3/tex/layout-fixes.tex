% Vol3 layout fixes (2026-09-29).
%
% longtable starts with \global\@colroom\@colht and never restores it. When a
% top float already sits on the current page, LaTeX then believes the page has
% a full column free, admits a second float, and the page overflows (text ran
% off the bottom of a page in the vol3 PDF). The kernel keeps \vsize equal to
% \@colroom, and longtable restores \vsize itself, so resynchronizing after the
% environment is correct whether or not the table broke across pages.
\makeatletter
\AfterEndEnvironment{longtable}{\global\@colroom\vsize}
\makeatother
