\frontside
\begin{bigidea}
A durable log records intent; it cannot change the world. Over a long \hl{horizon}, each effect its
\hl{authority} produced becomes an obligation: cleanup registered before dispatch, outcomes kept
\hl{unknown} until evidence settles them, and bounds enforced at the admission gate, not by a prompt warning.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{A timeout settles nothing.}{Outcomes are established, established absent, or unknown. Unknown keeps
  its reservation and owner, gets probed, and blocks steps that presume either.}
\idea{Cleanup is owed before the effect.}{Register compensation before dispatch; bind the resource ID on
  evidence. Compensation is a new operation restoring declared postconditions; it can fail.}
\idea{Cleanup is a graph, not a stack.}{Release waits until the test stops and logs are saved; an
  unresolved node blocks only its dependents.}
\idea{Check before the pivot.}{The one irreversible step of its region, under this grant.
  Hoist every check that can precede it; cleanup survives it.}
\idea{Watch progress, not liveness.}{A loop stays busy and green. Compare state digests, verifier results
  and repeated action--outcome pairs; enforce with an admission hold.}
\idea{Stopping is not recovering.}{An incident hold stops model-directed work at each dispatch boundary;
  preregistered cleanup runs under a narrow grant.}
\end{ideas}

\fcard{Repair first or unwind?}%
  {\begin{gathered}\mathbb{E}[\RepairFirstCost] = \RepairAttemptCost + (1-\RepairSuccessProbability)\FallbackCost\\[2pt]
   \text{repair first if } \RepairSuccessProbability > \RepairSuccessProbability^* = \RepairAttemptCost/\FallbackCost\end{gathered}}%
  {$\RepairAttemptCost$: one bounded attempt; $\FallbackCost$: compensate and rebuild to the \emph{same}
   endpoint; $\RepairSuccessProbability$: measured success rate. Seconds and dollars: separate thresholds.}%
  {A failed repair must leave the fallback intact.}{}
\fcard{Expected irreversible loss}%
  {\ExpectedIrreversibleLoss = \DefectiveActionProbability \cdot \text{consequence}(\PlanAction)\cdot(1-\PrePivotCatchRate\RevertFidelity)}%

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{Repair the configuration, or tear down and rebuild?}
Validation rejects a test configuration before dispatch; the environment is up. Fallback: unwind
\val{unwind} s, rebuild \val{rebuild} s with \val{rebin} in, \val{rebout} out tokens (\$\val{pin}, \$\val{pout} per
million). Repair: \val{repin} in, \val{repout} out, \val{startup} s start, \val{tps} tokens/s, \val{valid} s check;
\val{passume}\% succeed.\par\vspace{5pt}
{\centering
\begin{tikzpicture}[sheet]
  \pgfmathsetmacro\sc{130/\val{fbs}}
  \node[lab, anchor=east] at (78,23) {unwind now};
  \draw[nbox] (82,17) rectangle ({82+\val{fbs}*\sc},29);
  \node[note, anchor=west] at ({86+\val{fbs}*\sc},23) {\val{fbs} s};
  \node[lab, anchor=east] at (78,6) {repair first};
  \draw[hotbox] (82,0) rectangle ({82+\val{exps}*\sc},12);
  \node[note, anchor=west] at ({86+\val{exps}*\sc},6) {\val{exps} s expected};
\end{tikzpicture}\par}\vspace{4pt}
\begin{steps}
\item $\FallbackCost$: \val{unwind} + \val{rebuild} = \res{\val{fbs} s}; (\val{rebin} $\times$ \val{pin} + \val{rebout}
  $\times$ \val{pout})/$10^6$ = \res{\$\val{fbusd}}.
\item $\RepairAttemptCost$: \val{startup} + \val{repout}/\val{tps} + \val{valid} = \res{\val{reps} s};
  (\val{repin} $\times$ \val{pin} + \val{repout} $\times$ \val{pout})/$10^6$ = \res{\$\val{repusd}}.
\item $\RepairSuccessProbability^*$: \val{reps}/\val{fbs} = \res{\val{pstars}\%} (time); \val{repusd}/\val{fbusd} = \res{\val{pstarusd}\%} (dollars).
\item At \val{passume}\%: \val{reps} + \val{compassume} $\times$ \val{fbs} = \res{\val{exps} s}, \val{saves}\% less;
  \res{\$\val{expusd}} vs.\ \$\val{fbusd}.
\end{steps}
\takeaway{A costly fallback makes unlikely repairs pay.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``The provisioning call timed out, so it failed; send a new one.''}%
  {It may still complete: two billed environments, one unowned. Keep it unknown and probe; repeat
   only under the repeat contract.}{}
\mistake{``The \val{loopcap}-turn budget ceiling will catch a looping agent.''}%
  {Late: \$\val{loopusdcap} at the ceiling vs.\ \$\val{loopusdstop} stopped after turn \val{loopstop},
   about \val{loopratio}$\times$. Detection latency sets the cost.}{}
