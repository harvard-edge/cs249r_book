\begin{bigidea}
Splitting a task across agents replaces nothing a capable single controller already has; it adds
\hl{boundaries} between separately owned trajectories. Each must carry explicitly what one trajectory
carried implicitly: another completion to await (\hl{horizon}), an
interface version to revalidate (\hl{state}), a grant to bound (\hl{authority}). Delegation pays only by beating that \hl{single controller} on
the same budget.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Delegation buys separation, not speed.}{One controller already runs tools concurrently; a
  worker adds its own context, grant and lifecycle.}
\idea{Three structures, kept apart.}{Dependencies: what finishes first. Topology: who messages whom.
  Supervision: who owns and cancels.}
\idea{Envelope in, receipt out.}{Fixed inputs, grant and budget; receipts split evidence from
  claims. A full window of $\CreditWindowCapacity$ credits: mean residence $\CreditWindowCapacity/\ServiceRate$ at drain rate $\ServiceRate$.}
\idea{A clean merge is not a valid change.}{Revalidate against the current head, check the exact
  candidate, publish only if the head held.}
\idea{Agreement is one correlated signal.}{Shared difficulty floors majority false approval; add an
  independent, protected check.}
\idea{Spawning is admission; canceling is not settling.}{Copied remainders multiply liability: reserve each
  child's share atomically or deny it; grants only narrow.}
\end{ideas}

\fcard{Does a token multiple pay?}%
  {\frac{\CostPerAcceptedTask^{\text{multi}}}{\CostPerAcceptedTask^{\text{single}}}
   = \DelegationTokenMultiple\,\frac{\SingleAgentAcceptanceProbability}{\MultiAgentAcceptanceProbability}}%
  {Delegation uses $\DelegationTokenMultiple$ times the tokens per attempt. It lowers cost per accepted task
   only if acceptance rises more than $\DelegationTokenMultiple$-fold, possible only when
   $\SingleAgentAcceptanceProbability < 1/\DelegationTokenMultiple$.}%
  {Otherwise it must buy time the objective values. Compare against a matched single controller.}{}
\fcard{Slots or a shorter path?}%
  {\ScheduleFinishTime \ge \max\!\left(\frac{\AggregateSlotWork}{\WorkerCount},\ \CriticalPathDuration\right)}%
  {$\AggregateSlotWork$: total slot-work over $\WorkerCount$ worker slots; $\CriticalPathDuration$: longest
   dependency chain. Once a schedule reaches $\CriticalPathDuration$, only shortening that chain helps.}%
  {}{}

\fcard{Barrier or quorum}%
  {\begin{gathered}\Pr(\text{all by } \TimeInstant) = \WorkerCompletionDistribution(\TimeInstant)^{\WorkerCount}\\[2pt]
   \Pr(\ge \RequiredResultCount \text{ by } \TimeInstant) = \sum_{\FinishedWorkerCount=\RequiredResultCount}^{\WorkerCount}
   \binom{\WorkerCount}{\FinishedWorkerCount}\WorkerCompletionDistribution^{\FinishedWorkerCount}(1-\WorkerCompletionDistribution)^{\WorkerCount-\FinishedWorkerCount}\end{gathered}}%
  {Each worker finishes by $\TimeInstant$ with probability $\WorkerCompletionDistribution(\TimeInstant)$,
   independently. Waiting for all decays geometrically in $\WorkerCount$.}%
  {A quorum of $\RequiredResultCount$ needs interchangeable results; a required part has no stand-in.}{}
\fcard{Shared writes collide early}%
  {\Pr(\text{no shared write}) \approx \exp\!\left(-\frac{\WorkerCount(\WorkerCount-1)\WorkerWrittenFileCount^2}{2\RepositoryFileCount}\right)}%

   Pairs drive the exponent; past \val{lntwo} a collision is likely: with \val{files} files and \val{writes} writes
   each, at \val{evenodds} workers.}%
  {}{}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{How many workers should the migration get?}
A fixed partition has $\ParallelFraction = \val{fpar}$ and $\WorkerOverheadSlope = \val{gamma}$ per added worker;
the ceiling is \val{maxn} workers. Which count minimizes elapsed time, as a share of the serial run?\par\vspace{2pt}
\begin{steps}
\item Near $\sqrt{\val{fpar}/\val{gamma}} \approx \val{sqrtbest}$, so compare the neighbors.
\item $\WorkerCount = \val{best}$: \val{serial} + \val{osix} + \val{gsix} = \res{\val{tsix}}, a
  \res{\val{bestspeed}$\times$} speedup.
\item $\WorkerCount = \val{nextcount}$: \val{serial} + \val{oseven} + \val{gseven} = \val{tseven}: overhead now outgrows the saving.
\item Cap: $1/\val{serial}$ = \res{\val{cap}$\times$}. Halving $\WorkerOverheadSlope$ to \val{ghalf}: only
  \val{speedhalf}$\times$, at \val{besthalf} workers.
\end{steps}
\takeaway{The serial fraction sets the ceiling; the overhead sets the useful fleet size.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``The two patches merged cleanly, so they work together.''}%
  {Two workers writing \val{writes} of \val{files} files touch disjoint files about \val{disjointtwo}\% of the time:
   a clean merge is normal and says nothing about the interface.}{}
\mistake{``A majority of five reviewers approved, so the patch is correct.''}%
  {On invalid patches with \val{revp}\% false approval each, an independent majority errs \val{indfive}\%. The
   chapter's correlated model (correlation $\CorrelationCoefficient = \val{revrho}$) approaches a \val{revfloor}\% floor as
   panels grow.}{}
