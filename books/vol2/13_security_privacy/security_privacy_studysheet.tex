\begin{bigidea}
A machine learning service can run exactly as designed and still leak its training data or be
subverted through interfaces it exposes on purpose. Each control guards \hl{one boundary} and pays a
C\textsuperscript{3} price in \hl{compute, communication or coordination}.
\end{bigidea}

\begin{leftcol}
\sheetsec{Key ideas}
\begin{ideas}
\idea{Write the threat model in four fields.}{Asset, boundary, adversary, control. Authentication
  says who sent an update, not that it is benign.}
\idea{Learned behavior is an attack surface.}{Theft copies behavior, poisoning needs write access to
  training data, and adversarial inputs need only an account.}
\idea{Correct outputs can leak training data.}{Membership inference, inversion and verbatim extraction
  read only authorized answers; risk grows with overfitting.}
\idea{Model output is untrusted input.}{Injected text rides normal channels: separate instructions
  from retrieved content; authorize actions outside the model.}
\idea{Privacy is an accounted budget.}{Every data-dependent release spends $\PrivacyBudget$; querying a
  fixed DP-trained model is post-processing and spends none.}
\idea{Provenance bounds trust.}{Trust a model only as far as records of its data, code and weights
  reach. Breach: isolate, keep evidence, roll back, fix the boundary.}
\end{ideas}

\sheetsec{Formulas you will use}
\fcard{Extraction break-even}%
  {\ExtractionQueryCount \cdot \QueryPrice \geq \TrainingCost}%
  {Copying costs more than training when the $\ExtractionQueryCount$ queries a surrogate needs, at
   $\QueryPrice$ each, cost at least the training cost $\TrainingCost$.}%
  {Holds only if every query is billed; assumes an equal-quality copy.}{}
\fcard{Laplace mechanism}%
  {\LaplaceNoiseScale = \frac{\QuerySensitivity\DatasetQuery}{\PrivacyBudget}}%
  {Noise scale $\LaplaceNoiseScale$ added to a sum $\DatasetQuery$ that one record can change by at most
   $\QuerySensitivity\DatasetQuery$; dividing by a public count $\DatasetSize$ gives a mean that errs by about
   $\LaplaceNoiseScale/\DatasetSize$. Chapter: salaries \$0 to \$\val{salrange}, $\PrivacyBudget = \val{saleps}$,
   \val{salgroup} people: $\LaplaceNoiseScale$ = \$\val{salscale}, error \$\val{salerr}.}%
  {$\ComposedMechanismCount$ fresh releases at $\PrivacyBudget_0$ total at most $\ComposedMechanismCount\PrivacyBudget_0$.
   DP-SGD: a tested accountant takes clipping norm, noise multiplier, sampling rate, steps, $\PrivacyAdditiveSlack$,
   protected unit.}{}
\fcard{Membership ceiling}%
  {\text{members flagged} \leq e^{\PrivacyBudget} \times \text{nonmembers flagged}}%
  {Under pure $\PrivacyBudget$-DP no membership test beats this: an upper bound on leakage. An audit attack
   measures it from below.}%
  {$e^{\PrivacyBudget}$ = \val{eone} at $\PrivacyBudget = \val{epsone}$; this branch binds while nonmembers flagged
   $< 1/(1+e^{\PrivacyBudget})$ = \val{knee}.}%

\begin{answers}
\ans{Finish}{\val{fqueriesM}M $\div$ \val{fquota} a day = \val{fdays} days; \$\val{ffee} $\times$ \val{fdays}
  $\div$ \val{wperiod} = \$\val{fbill}, about \val{fpct}\% of \$\val{ftrainM}M. Bill every query past a lifetime
  allowance at about \$\val{fbreak} (break-even card).}
\ans{Q-Laplace}{(a) $\LaplaceNoiseScale$ = \$\val{salrange} $\div$ \val{qeps} = \$\val{qscale}; $\div$
  \val{qgroup} = \$\val{qerr} expected error, \val{qratio}$\times$ the chapter's \$\val{salerr} (half the budget,
  half the people). (b) \val{qmonths} $\times$ \val{qeps} = \val{qtotal} under basic composition. (c) $\PrivacyAdditiveSlack$,
  the protected unit, the accountant and its inputs (Laplace card).}
\ans{Q-Fleet}{(a) Half of \val{qdp}: \val{qgatherratio}$\times$ the ring traffic. (b) Effective
  $\LinkBandwidth$ halves: the bandwidth term doubles unless overlapped. (c) Coordination, $\SynchronizationTime$:
  fresh evidence and the key come before rejoining (price table).}
\columnbreak
\ans{Q-Audit}{Ceiling \val{eone} $\times$ \val{auditfpr}\% = \val{ceilpct}\% $<$ \val{audittpr}\%: if the rates are
  precise, the release breaks $\PrivacyBudget = \val{epsone}$: accounting or implementation is wrong.
  Below the ceiling, an audit proves nothing (membership card).}
\ans{Q-Rollback}{Pull it from serving, revoke its credentials, keep an isolated copy as evidence;
  roll back to a release whose record (data, dependencies, image) verifies; repair the boundary (provenance).}
\end{answers}

\sheetsec{Worked example}
\begin{practicebox}\boxtitle{Does a daily quota stop a copier?}
A ResNet-50 API cost \$\val{wtrain} to train; a copy needs \val{wqueriesM}M answers for
\val{wfidelity}\% fidelity. Billing: \$\val{wprice} a query; a \$\val{wtier}-a-month tier capped at
\val{wquota} a day; or, on the tier, a lifetime allowance of \val{wallowK}K per principal, then per query.\par\vspace{5pt}
{\centering
\begin{tikzpicture}[sheet]
  \pgfmathsetmacro\sc{140/\val{drawtrain}}
  \foreach \v/\t/\y/\s/\k in {\val{drawpaygo}/per query/28/nbox/\val{wpaygoK},\val{drawlife}/lifetime/14/accbox/\val{wlifeK},\val{drawtier}/daily quota/0/hotbox/\val{wtierK}}{
    \node[small, anchor=east] at (56,{\y+5}) {\t};
    \draw[\s] (60,\y) rectangle ({60+\v*\sc},{\y+10});
    \pgfmathsetmacro\lx{\v > 0.8*\val{drawtrain} ? 64+\val{drawtrain}*\sc : 64+\v*\sc}
    \node[lab, anchor=west] at (\lx,{\y+5}) {\$\k k};}
  \draw[accent, dashed, line width=0.6pt] ({60+\val{drawtrain}*\sc},-3) -- ({60+\val{drawtrain}*\sc},41);
  \node[note, anchor=south] at ({60+\val{drawtrain}*\sc},42) {training \$\val{wtrainK}k};
\end{tikzpicture}\par}\vspace{6pt}
\begin{steps}
\item Per query: \val{wqueriesM}M $\times$ \$\val{wprice} = \res{\$\val{wpaygo}}, the training cost: break-even.
\item Daily quota: \val{wqueriesM}M $\div$ \val{wquota} = \val{wdays} days; \$\val{wtier} $\times$ \val{wdays}
  $\div$ \val{wperiod} = \res{\$\val{wtiercost}}, \val{wtierpct}\% of training.
\item Lifetime: (\val{wqueriesM}M $-$ \val{wallowK}K) $\times$ \$\val{wprice} = \$\val{woverage}, plus \$\val{wallowfee}
  of tier fees (\val{wallowdays} days) = \res{\$\val{wlife}}, \val{wlifepct}\% of training.
\end{steps}
\takeaway{A daily quota buys time, not cost; a lifetime allowance prices the copy like training.}
\end{practicebox}

\sheetsec{Common mistakes}
\mistake{``An attested worker sends honest gradients.''}%
  {Attestation shows which code booted, not that its update is benign. One unbounded gradient moves
   the mean for all \val{gpus} GPUs; robust aggregation or update checks address it.}{}
\mistake{``A strict system prompt stops prompt injection.''}%
  {Roles and delimiters guide the model but authorize nothing; retrieved text can still steer it.
   A policy point outside the model checks the user's permission for each tool call and holds
   irreversible ones for approval.}{}
\mistake{``Federated learning keeps the data private.''}%
  {Updates can reconstruct examples. Secure aggregation hides each update (given at least
   \val{noncolluding} noncolluding clients); only accounted DP bounds what the released model reveals.}{}
