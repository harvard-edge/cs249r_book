\begin{frame}{One neuron is a dot product followed by an activation}

      \end{tikzpicture}
    \end{column}
    \begin{column}{0.42\textwidth}
      \normalsize
      \uncover<1->{$z_j = \sum_i x_i\, w_{ij} + b_j$}\\[0.35cm]
      \uncover<2->{$\hat{y}_j = f(z_j)$}\\[0.5cm]
      \uncover<3->{\textbf{Cost:} $n$ inputs $\rightarrow$ $n$ MACs\\[0.1cm]
      {\small\textcolor{midgray}{784 for a first-layer unit}}}
    \end{column}
  \end{columns}
  \note{%
    \mlsysnotemeta{1.5 min}{Core}%
    Rosenblatt's perceptron (1958) computes two steps. Step 1 forms the weighted sum plus bias, $z_j = \sum_i x_i w_{ij} + b_j$. Step 2 applies a nonlinear activation $\hat{y}_j = f(z_j)$. Step 3 counts the work: $n$ inputs cost $n$ MACs, so a first-layer MNIST unit costs 784 MACs. Without data reuse, every MAC fetches both an input and a weight, making memory traffic contend with arithmetic.

    \notecue{Check}{Ask how many MACs all 128 first-layer units perform together: $784 \times 128 = 100{,}352$ MACs.}

    \notecue{Watch for}{Treat the biological neuron strictly as a historical metaphor rather than a physical model.}

    \notecue{Next}{Show mathematically why networks require nonlinear activations using the classic XOR problem.}
  }
\end{frame}

\begin{frame}{Takeaways}
  \normalsize
  \begin{itemize}\setlength{\itemsep}{0.25cm}

    \item Backpropagation reads saved forward values; a step is 2 to 3 forward passes.
