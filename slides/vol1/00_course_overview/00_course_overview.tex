\mlsysnote{[2 min] The iron law as three factors, one per D·A·M axis. Samples per task is the data term, FLOPs per sample the algorithm term, seconds per FLOP the machine term. Every optimization in this course reduces one factor; Lecture 1 develops it.}
\mlsysfocus{The Iron Law of ML Systems}{%
  $\text{Time} = \underbrace{\text{samples}}_{\text{data}} \times
   \underbrace{\text{FLOPs per sample}}_{\text{algorithm}} \times
   \underbrace{\text{seconds per FLOP}}_{\text{machine}}$%
}


\begin{frame}{Diagnose: which axis is the problem?}
  \normalsize
  \begin{tabular}{@{}P{0.66\textwidth}c@{}}
    \toprule
    \textbf{Symptom} & \textbf{Axis} \\
    \midrule
    Recall falls as the user population changes. & \uncover<2->{\textbf{\textcolor{GreenLine}{Data}}} \\[0.15cm]
    Each new token waits for every weight to be read from memory. & \uncover<3->{\textbf{\textcolor{BlueLine}{Machine}}} \\[0.15cm]
    A larger model is needed, and it doubles the FLOPs per sample. & \uncover<4->{\textbf{\textcolor{OrangeLine}{Algorithm}}} \\
    \bottomrule
  \end{tabular}
\note{[4 min] ENGAGE: students write D, A, or M for each before any reveal. Step 2: data, the distribution moved (Lecture 14). Step 3: machine, memory bandwidth bounds language-model decoding (Lectures 11 and 13). Step 4: algorithm, the architecture sets the arithmetic (Lecture 6). Note that a fix on one axis can expose another.}
\end{frame}
