\begin{frame}{Methods that turn computation into skill beat encoded knowledge}
  \centering
  {\small\itshape ``General methods that leverage computation are ultimately the most effective,\\and by a large margin.''}\\[0.05cm]
  {\footnotesize\textcolor{midgray}{Richard Sutton, ``The Bitter Lesson,'' 2019}}\\[0.15cm]
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering\safeimg[width=\textwidth,height=2.5cm,keepaspectratio]{deep-blue.jpg}\\[0.05cm]
      {\footnotesize\textbf{Deep Blue, 1997}\\ custom chips, encoded chess knowledge}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering\safeimg[width=\textwidth,height=2.5cm,keepaspectratio]{go-board.jpg}\\[0.05cm]
      {\footnotesize\textbf{AlphaGo Zero, 2017}\\ self-play, no human games}
    \end{column}
  \end{columns}
  \mlsyscredit{Photos: James the photographer (CC BY 2.0); Dietmar Rabich (CC BY-SA 4.0); Wikimedia Commons. Facts: Introduction, \S1.4.}
  \note{%
    \mlsysnotemeta{1.5 min}{Core}%
    Richard Sutton's 2019 essay argues that general methods of search and learning consistently overtake hand-crafted domain knowledge as available compute scales. Deep Blue defeated Kasparov in 1997 using 480 custom chips evaluating 100 to 200 million positions per second with encoded chess heuristics; twenty years later, AlphaGo Zero learned Go purely through self-play and beat the earlier champion version 100 games to 0 after three days.

\begin{frame}{The machine factor splits into memory, compute, and overhead}
  \vspace{-0.15cm}
  \centering

    \node[st, draw=midgray, fill=vlightgray!35, right=0.55cm of c] (o) {\textbf{3. Fixed overhead} ($\Llat$)};
    \draw[ar] (m) -- (c); \draw[ar] (c) -- (o);
  \end{tikzpicture}\\[0.10cm]}
  \uncover<4->{\small $\Dvol$ counts bytes moved; one decoded token moves every weight.}

  \note{%
