## fuse
>>> conv, bn = nn.Conv2d(16, 16, 3), nn.BatchNorm2d(16)
>>> _ = bn.running_var.uniform_(0.5, 2)
>>> m = nn.Sequential(conv, bn, nn.ReLU()).eval()
>>> [type(x).__name__ for x in m]
['Conv2d', 'BatchNorm2d', 'ReLU']
>>> f = fuse_modules(m, [['0', '1', '2']])
>>> [type(x).__name__ for x in f]
['ConvReLU2d', 'Identity', 'Identity']
>>> x = torch.randn(1, 16, 28, 28)
>>> (f(x) - m(x)).abs().max().item()
1.6689300537109375e-06
