\begin{frame}{Each accepted task carries $(1-\alpha)/\alpha$ rejected ones}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \centering\safeimg[width=\textwidth,height=5.3cm,keepaspectratio]{multiplier.pdf}
    \end{column}
    \begin{column}{0.47\textwidth}
      \small
      $\Ceff = \dfrac{\sum_i C_{\text{task},i}}{N_{\text{acc}}} = \bar C_{\text{acc}} + \dfrac{1-\alpha}{\alpha}\,\bar C_{\text{rej}}$\\[0.3cm]
      {\footnotesize $\alpha$: share of initiated tasks accepted under one contract by a cutoff. A retry on one ticket adds no accepted task.}\\[0.3cm]
      \begin{mlsyscard}{mlsysaccent}\small A token cut shrinks one term. Acceptance divides \textbf{every} term.\end{mlsyscard}
    \end{column}
  \end{columns}
  \mlsyscredit{Agent Economics, \S19.2.2 (cohort cost per accepted task) and its rejected-per-accepted figure, redrawn at slide size.}
  \note{%
    \mlsysnotemeta{2.5 min}{Core}%
    Cohort cost per accepted task (\S19.2.2) turns Lecture 1's planning ratio into an exact accounting identity with no distributional assumptions: dividing total initiated spend (failed, timed-out, and canceled work included) by distinct accepted tasks under one contract by cutoff gives $C_{\text{eff}} = \bar{C}_{\text{acc}} + \frac{1-\alpha}{\alpha}\bar{C}_{\text{rej}}$. At 25 percent acceptance, each accepted task carries 3 rejected attempts; at 75 percent, only 0.33 (on Slide 8, Run A's 168 resolved runs carried $332/168 \approx 2$ unresolved runs each), and below 30 percent the multiplier climbs steeply. Discounting tokens shrinks one numerator term; raising acceptance divides every term.

    \notecue{Watch for}{Three successful retries on the same ticket still count as one accepted task in the denominator.}

    \notecue{Next}{Slide 10 tests three accounting edge cases on canceled children, pending verdicts, and zero acceptances.}
  }
\end{frame}

\begin{frame}{Compaction pays only if enough calls remain to repay it}
  \centering
  \safeimg[width=0.92\textwidth,height=4.15cm,keepaspectratio]{payback.pdf}\\[0.02cm]
  {\small $n^\star = 1 + \left\lceil \dfrac{p_{\text{out}}K_{\text{sum}} + p_{\text{in}}S_c}{p_{\text{cache}}(S - S_c)} \right\rceil$}\enspace{\footnotesize ($p$: \$15/\$3/\$0.30/M;\ $K_{\text{sum}}{=}1.5\text{k}$;\ $S{=}30.5\text{k}{\to}S_c{=}12\text{k}$)}\\[0.02cm]

  \mlsyscredit{Agent Economics, \S19.3.2 and Napkin Math 19.3 (hypothetical; tariff of Napkin Math 19.2). Curves and payback (derived).}
  \note{%
    \mlsysnotemeta{3 min}{Core}%
    Compacting a trajectory rewrites the prefix and forces a cache miss on the next call (Napkin Math 19.3).
