  \begin{columns}[c]
    \begin{column}{0.58\textwidth}

    \end{column}
    \begin{column}{0.40\textwidth}
      \small
      Tick $=\val{unloaded_us}\,[1+\varphi(S-1)]$\,µs\\[0.1cm]
