  "file:NotoColorEmoji.ttf:mode=harf;"

% Use the existing outer-margin space for wide technical diagrams. The box
% retains the text-column width so Pandoc does not shrink it a second time.
% Explicit per-image widths remain authoritative for tall diagrams.
\usepackage[strict]{changepage}
\usepackage{xstring}
\newsavebox{\TTDiagramBox}
\renewcommand{\includegraphics}[2][]{%
  \IfSubStr{#2}{assets/images/diagrams/}{%
    \sbox{\TTDiagramBox}{%
      \TTOrigIncludeGraphics[%
        keepaspectratio,%
        width=\dimexpr\textwidth+\marginparsep+\marginparwidth\relax,%
        #1,%
        max width=\dimexpr\textwidth+\marginparsep+\marginparwidth\relax,%
        max height=0.8\textheight%
      ]{#2}%
    }%
    \checkoddpage%
    \ifoddpage%
      \makebox[\linewidth][l]{\usebox{\TTDiagramBox}}%
    \else%
      \makebox[\linewidth][r]{\usebox{\TTDiagramBox}}%
    \fi%
  }{%
    \TTOrigIncludeGraphics[keepaspectratio,#1,max width=\linewidth,max height=0.55\textheight]{#2}%
  }%
}

% A float is already indivisible. A breakable tcolorbox nested in a listing
% float can split its paint layer without splitting the float, hiding code
% and leaking syntax colors. Keep only these nested boxes unbreakable.
\AtBeginEnvironment{codelisting}{%
  \renewenvironment{Shaded}{%
    \begin{tcolorbox}[borderline west={3pt}{0pt}{shadecolor},
      boxrule=0pt, colback=codebgcolor, enhanced,
      frame hidden, sharp corners, breakable=false]%
  }{\end{tcolorbox}}%
}
