// -----------------------------------------------------------------------------
// Category 1: Math and Arithmetic Primitives
// -----------------------------------------------------------------------------

int add(int a, int b) {
    return a + b;
}

int multiply(int a, int b) {
    return a * b;
}

int is_even(int n) {
    return n % 2 == 0;
}

int is_odd(int n) {
    return n % 2 != 0;
}

float clamp(float x, float min_val, float max_val) {
    if (x < min_val) {
        return min_val;
    }
    if (x > max_val) {
        return max_val;
    }
    return x;
}

int factorial(int n) {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

int is_prime(int n) {
    if (n < 2) {
        return 0;
    }
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return 0;
        }
    }
    return 1;
}

int gcd(int a, int b) {
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int power(int base, int exp) {
    int result = 1;
    while (exp > 0) {
        if (exp % 2 == 1) {
            result *= base;
        }
        base *= base;
        exp /= 2;
    }
    return result;
}

// -----------------------------------------------------------------------------
// Category 2: Machine Learning & Systems Primitives
// -----------------------------------------------------------------------------

float relu(float x) {
    if (x > 0.0f) {
        return x;
    }
    return 0.0f;
}

float relu_backward(float grad_output, float x) {
    if (x > 0.0f) {
        return grad_output;
    }
    return 0.0f;
}

float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}

float mse_loss(const float* pred, const float* target, int n) {
    float total = 0.0f;
    for (int i = 0; i < n; i++) {
        float diff = pred[i] - target[i];
        total += diff * diff;
    }
    return total / (float)n;
}

float dot_product(const float* a, const float* b, int n) {
    float sum = 0.0f;
    for (int i = 0; i < n; i++) {
        sum += a[i] * b[i];
    }
    return sum;
}

void vector_add(const float* a, const float* b, float* out, int n) {
    for (int i = 0; i < n; i++) {
        out[i] = a[i] + b[i];
    }
}

void vector_scale(const float* a, float scalar, float* out, int n) {
    for (int i = 0; i < n; i++) {
        out[i] = a[i] * scalar;
    }
}

void linear_forward(const float* x, const float* w, const float* b, float* out, int in_dim, int out_dim) {
    for (int o = 0; o < out_dim; o++) {
        float sum = b[o];
        for (int i = 0; i < in_dim; i++) {
            sum += x[i] * w[o * in_dim + i];
        }
        out[o] = sum;
    }
}

void softmax(const float* input, float* output, int n) {
    float max_val = input[0];
    for (int i = 1; i < n; i++) {
        if (input[i] > max_val) {
            max_val = input[i];
        }
    }
    float sum = 0.0f;
    for (int i = 0; i < n; i++) {
        output[i] = expf(input[i] - max_val);
        sum += output[i];
    }
    for (int i = 0; i < n; i++) {
        output[i] /= sum;
    }
}

int linear_search(const int* arr, int n, int target) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == target) {
            return i;
        }
    }
    return -1;
}

int binary_search(const int* arr, int n, int target) {
    int low = 0;
    int high = n - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        if (arr[mid] == target) {
            return mid;
        }
        if (arr[mid] < target) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    return -1;
}

void bubble_sort(int* arr, int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

struct Node {
    int data;
    struct Node* next;
};

struct Stack {
    int items[64];
    int top;
};

void stack_init(struct Stack* s) {
    s->top = -1;
}

int stack_push(struct Stack* s, int val) {
    if (s->top >= 63) {
        return 0;
    }
    s->items[++(s->top)] = val;
    return 1;
}

int stack_pop(struct Stack* s, int* out_val) {
    if (s->top < 0) {
        return 0;
    }
    *out_val = s->items[(s->top)--];
    return 1;
}
