% ===========================================================================
% ROUND 4 - built on a CORRECTED measurement and the print-typography review.
%   "Nervous System" in \rmfamily\scriptsize\bfseries = 59.70pt = 2.098cm,
%   against 2.480cm of usable rung. It fits with 3.8mm to spare. The 2.55cm
%   figure quoted in rounds 1-3 was an eyeball estimate, not a measurement,
%   so the boxed vol1 geometry is available to vol4 after all.
% Typography fixes applied:
%   labels set in real Helvetica via \usefont{T1}{phv}{..}{n}; \sffamily is
%     unreliable here and falls through to a serif
%   secondary line 7pt, never 5.4pt (a Palatino en-dash rule at 5.4pt is
%     ~0.24pt, under the uncoated press floor)
%   tints restricted to 0 / 22 / 90; the 31-84 band has no legible text colour
%   no transparency, no screened sub-1pt rules
% ===========================================================================
\newcommand{\sflab}{\usefont{T1}{phv}{b}{n}\fontsize{8}{9}\selectfont}
\newcommand{\sfsub}{\usefont{T1}{phv}{m}{n}\fontsize{7}{8}\selectfont}
\definecolor{ctxtint}{HTML}{D3DBD7}   % emerald at ~22, flattened to solid ink
\definecolor{ctxink}{HTML}{5A6560}

% ---- boxed, vol1 geometry (2.6cm), binary tint --------------------------
% #1 y #2 label #3 rate #4 active #5 showrate
\newcommand{\bRow}[5]{%
  \ifnum#4=1 \def\bg{emeraldpine}\def\tx{white}\else\def\bg{ctxtint}\def\tx{ctxink}\fi
  \node[draw=white,line width=1pt,fill=\bg,minimum width=2.6cm,minimum height=1.0cm,
        anchor=south west,inner sep=0pt](b) at (0,#1){};
  \ifnum#5=1
    \node[anchor=west,text=\tx,font=\sflab] at ($(b.west)+(0.12,0.13)$) {#2};
    \node[anchor=west,text=\tx,font=\sfsub] at ($(b.west)+(0.12,-0.16)$) {#3};
  \else
    \node[anchor=west,text=\tx,font=\sflab] at ($(b.west)+(0.12,0)$) {#2};
  \fi}
\newcommand{\bStack}[1]{\begin{tikzpicture}
  \bRow{3.0}{Governance}{0.1--1 Hz}{0}{#1}
  \bRow{2.0}{Brain}{1--50 Hz}{1}{#1}
  \bRow{1.0}{Nervous System}{1000 Hz}{0}{#1}
  \bRow{0.0}{Body}{continuous}{0}{#1}
\end{tikzpicture}}

% ---- gapped, mark outside (the round-3 winner), solid ink ----------------
\newcommand{\gRow}[5]{%
  \ifnum#4=1
    \fill[emeraldpine] (0,#1) rectangle (0.32,#1+0.88);
    \def\tx{emeraldpine}\def\sx{emeraldpine}
  \else
    \fill[ctxink!55] (0.13,#1+0.06) rectangle (0.20,#1+0.82);
    \def\tx{ctxink}\def\sx{ctxink}
  \fi
  \ifnum#5=1
    \node[anchor=west,text=\tx,font=\sflab] at (0.46,#1+0.58) {#2};
    \node[anchor=west,text=\sx,font=\sfsub] at (0.46,#1+0.27) {#3};
  \else
    \node[anchor=west,text=\tx,font=\sflab] at (0.46,#1+0.44) {#2};
  \fi}
