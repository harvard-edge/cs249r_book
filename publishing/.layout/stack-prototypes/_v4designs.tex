% --- D2  Outline ladder: only the active tier carries ink -----------------
\newcommand{\dTwoRung}[5]{%
  \ifnum#4=1
    \node[draw=\vfour,line width=0.5pt,fill=\vfour,minimum width=3.05cm,
          minimum height=#2 cm,anchor=south west,inner sep=0pt](b) at (0,#1){};
    \node[anchor=west,text=white,font=\rmfamily\scriptsize\bfseries]
          at ($(b.west)+(0.14,0.11)$) {#3};
    \node[anchor=west,text=white,opacity=0.85,
          font=\rmfamily\fontsize{5.6}{6.2}\selectfont] at ($(b.west)+(0.14,-0.14)$) {#5};
  \else
    \node[draw=\vfour!35,line width=0.5pt,fill=white,minimum width=3.05cm,
          minimum height=#2 cm,anchor=south west,inner sep=0pt](b) at (0,#1){};
    \node[anchor=west,text=\vfour!70,font=\rmfamily\scriptsize]
          at ($(b.west)+(0.14,0.11)$) {#3};
    \node[anchor=west,text=\vfour!45,
          font=\rmfamily\fontsize{5.6}{6.2}\selectfont] at ($(b.west)+(0.14,-0.14)$) {#5};
  \fi}
