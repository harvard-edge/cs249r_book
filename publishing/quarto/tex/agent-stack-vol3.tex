% ML AGENT STACK DIAGRAM (VOLUME III)
% =============================================================================
% Renders the Volume III Agent Stack with configurable layer shading (0-100).
% The interface mirrors Volume 1's \mlsysstack and Volume 2's \mlfleetstack.
% The primary chapter focus uses 90-100, neighboring context rows use lower
% intensities (15-30), and unengaged rows use 5-10.
%
% Usage:
%   \mlagentstack{<fleet>}{<control>}{<policy>}{<serving>}{<tools>}{<sandbox>}
%
% Stack (top to bottom):
%   Layer 6: Fleet & Assurance (Part V: Multi-Agent, Telemetry, Replay)
%   Layer 5: Control Plane & Descriptors (Part I: ACB, Topologies, Circuit Breakers)
%   Layer 4: Policy & Reasoning (Part II: SFT, RLVR / GRPO, Deliberate Search)
%   Layer 3: Serving & Memory (Part III: Radix Attention, KV Paging, Scheduling)
%   Layer 2: Tool Actuation & ABI (Part IV: MCP, Typed Contracts, Grammar Decoding)
%   Layer 1: Sandbox & Environment (Part IV: MicroVMs, gVisor, Protection Domains)
%
% Vertical Spine (Right):
%   TRAJECTORY (The stateful closed loop binding all layers into Invariant Closure)
% =============================================================================

% --- Row helper --------------------------------------------------------------
% Fixes two defects found in a print-typography review (2026-09-10):
%
%   1. The label font was \sffamily. Under this font stack that does not
%      resolve to Helvetica. With fontspec on LuaLaTeX the default encoding is
%      TU, phv carries no TU font, and NFSS silently substitutes a serif, so
%      every vol3 label was setting in Latin Modern Roman while vol1 and vol2
%      set in the intended sans. \usefont{T1}{phv}{..}{n} names the font
%      directly and resolves correctly. Helvetica is also narrower than the
%      fallback at the same size, so this cannot overflow a rung.
%
%   2. The label colour was hardcoded text=black. On the active tier that put
%      black on amethyst!95, a contrast ratio of 2.18:1 at 4.8pt, which prints
%      as a smear. The colour now flips to white once the tier is dark enough
%      to need it.
%
% Deliberately NOT changed here, because both would alter the rendered result
% in all fifteen vol3 chapters and are design calls rather than defects:
% the 4.8pt label size (below the 7pt floor recommended for uncoated stock),
% and the context tints, four of whose five distinct values carry no
% information the reader can resolve at 26mm.
%
% #1 y-bottom  #2 y-top  #3 label  #4 intensity (0-100)
\newcommand{\agentrow}[4]{%
  \fill[amethyst!#4!white] (0,#1) rectangle (\stackw,#2);
  \draw[agentlayer] (0,#1) rectangle (\stackw,#2);
  \ifnum#4>55 \def\agentink{white}\else \def\agentink{black}\fi
  \pgfmathsetmacro{\agentmid}{(#1+#2)/2}
  \node[anchor=west,font=\usefont{T1}{phv}{b}{n}\fontsize{4.8}{5.2}\selectfont,
        text=\agentink] at (1.2,\agentmid) {#3};}

\newcommand{\mlagentstack}[6]{%
  \begin{tikzpicture}[x=1mm,y=1mm,baseline=(current bounding box.north)]
    \def\stackw{26.0}
    \def\stackh{6.2}
    \tikzset{agentlayer/.style={draw=amethyst!70, line width=0.3pt, rounded corners=0.5pt}}

    % Layer 6 (Top): Fleet & Assurance
    \agentrow{35.0}{41.2}{Fleet \& Assurance}{#1}

    % Layer 5: Control Plane & Descriptors
    \agentrow{28.0}{34.2}{Control Plane}{#2}

    % Layer 4: Policy & Reasoning
    \agentrow{21.0}{27.2}{Policy \& Reasoning}{#3}

    % Layer 3: Serving & Memory
    \agentrow{14.0}{20.2}{Serving \& Memory}{#4}

    % Layer 2: Tool Actuation & ABI
    \agentrow{7.0}{13.2}{Tool Actuation}{#5}

    % Layer 1 (Bottom): Sandbox & Environment
    \agentrow{0}{6.2}{Sandbox \& Env}{#6}

    % Trajectory Vertical Spine on Right
    \fill[amethyst!88] (27.2,0) rectangle (31.0,41.2);
    \node[white,rotate=90,font=\usefont{T1}{phv}{b}{n}\fontsize{4.6}{5}\selectfont] at (29.1,20.6) {TRAJECTORY};
  \end{tikzpicture}%
}
